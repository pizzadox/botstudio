   108→
   109→export default function ChannelsView({
   110→  bot,
   111→  onBack,
   112→  onOpenEditor,
   113→}: {
   114→  bot: { id: string; name: string; status: string };
   115→  onBack: () => void;
   116→  onOpenEditor: () => void;
   117→}) {
   118→  const { toast } = useToast();
   119→  const [channels, setChannels] = useState<ChannelItem[]>([]);
   120→  const [loading, setLoading] = useState(true);
   121→  const [addOpen, setAddOpen] = useState(false);
   122→  const [type, setType] = useState<ChannelItem['type']>('telegram');
   123→  const [title, setTitle] = useState('');
   124→  const [token, setToken] = useState('');
   125→  const [phone, setPhone] = useState('');
   126→  const [busy, setBusy] = useState(false);
   127→  const [demoChannel, setDemoChannel] = useState<ChannelItem | null>(null);
   128→  const [origin, setOrigin] = useState('');
   129→  // Подтверждение удаления (AlertDialog): state-target + open={!!target}
   130→  const [deleteTarget, setDeleteTarget] = useState<ChannelItem | null>(null);
   131→
   132→  const load = useCallback(async () => {
   133→    try {
   134→      const d = await api<{ channels: ChannelItem[] }>(`/api/bots/${bot.id}/channels`);
   135→      setChannels(d.channels);
   136→    } catch {
   137→      toast({ title: 'Не удалось загрузить каналы', variant: 'destructive' });
   138→    } finally {
   139→      setLoading(false);
   140→    }
   141→  }, [bot.id, toast]);
   142→
   143→  useEffect(() => {
   144→    setOrigin(window.location.origin);
   145→    load();
   146→  }, [load]);
   147→
   148→  const createChannel = async () => {
   149→    setBusy(true);
   150→    try {
   151→      await api(`/api/bots/${bot.id}/channels`, {
   152→        method: 'POST',
   153→        body: JSON.stringify({ type, title, token, phone }),
   154→      });
   155→      setAddOpen(false);
   156→      setTitle('');
   157→      setToken('');
   158→      setPhone('');
   159→      toast({ title: 'Канал добавлен' });
   160→      load();
   161→    } catch (e) {
   162→      toast({ title: 'Ошибка', description: e instanceof Error ? e.message : '', variant: 'destructive' });
   163→    } finally {
   164→      setBusy(false);
   165→    }
   166→  };
   167→
   168→  const toggleActive = async (ch: ChannelItem, active: boolean) => {
   169→    setChannels((cs) => cs.map((c) => (c.id === ch.id ? { ...c, active } : c)));
   170→    try {
   171→      await api(`/api/channels/${ch.id}`, { method: 'PUT', body: JSON.stringify({ active }) });
   172→    } catch {
   173→      load();
   174→    }
   175→  };
   176→
   177→  const testChannel = async (ch: ChannelItem) => {
   178→    toast({ title: 'Проверяем подключение…' });
   179→    try {
   180→      const d = await api<{ ok: boolean; message: string }>(`/api/channels/${ch.id}/test`, {
   181→        method: 'POST',
   182→      });
   183→      toast({ title: d.message, variant: d.ok ? 'default' : 'destructive' });
   184→      load();
   185→    } catch {
   186→      toast({ title: 'Ошибка проверки', variant: 'destructive' });
   187→    }
   188→  };
   189→
   190→  const deleteChannel = async (ch: ChannelItem) => {
   191→    try {
   192→      await api(`/api/channels/${ch.id}`, { method: 'DELETE' });
   193→      toast({ title: 'Канал удалён' });
   194→      load();
   195→    } catch {
   196→      toast({ title: 'Не удалось удалить', variant: 'destructive' });
   197→    }
   198→  };
   199→
   200→  const copy = async (text: string) => {
   201→    try {
   202→      await navigator.clipboard.writeText(text);
   203→      toast({ title: 'Скопировано в буфер' });
   204→    } catch {
   205→      toast({ title: 'Не удалось скопировать', variant: 'destructive' });
   206→    }
   207→  };
   208→
   209→  const webhookUrl = (ch: ChannelItem) => `${origin}/api/webhook/${ch.type}/${ch.secret}`;
   210→
   211→  return (
   212→    <div className="flex-1 min-h-0 overflow-y-auto space-y-6 p-4 sm:p-6 lg:p-8">
   213→      <div className="flex flex-wrap items-center justify-between gap-4">
   214→        <div className="flex items-center gap-3">
   215→          <Button variant="ghost" size="icon" className="h-8 w-8" onClick={onBack} aria-label="Назад">
   216→            <ChevronLeft className="h-4 w-4" />
   217→          </Button>
   218→          <div>
   219→            <h1 className="flex items-center gap-2 text-lg font-bold tracking-tight sm:text-2xl">
   220→              <Plug className="h-6 w-6 text-primary" /> Каналы
   221→            </h1>
   222→            <p className="text-sm text-muted-foreground">Подключение бота «{bot.name}» к мессенджерам</p>
   223→          </div>
   224→        </div>
   225→        <Button onClick={() => { setType('telegram'); setAddOpen(true); }}>
   226→          <Plus className="h-4 w-4" /> Подключить канал
   227→        </Button>
   228→      </div>
   229→
   230→      {bot.status !== 'published' && (
   231→        <Alert className="border-amber-200 bg-amber-50 dark:border-amber-500/30 dark:bg-amber-500/10">
   232→          <Info className="h-4 w-4 text-amber-600 dark:text-amber-400" />
   233→          <AlertTitle className="text-amber-800 dark:text-amber-300">Бот ещё не опубликован</AlertTitle>
   234→          <AlertDescription className="text-amber-700 dark:text-amber-300/80">
   235→            Каналы сохранят настройки, но отвечать они начнут после публикации бота в{' '}
   236→            <button className="font-medium underline underline-offset-2" onClick={onOpenEditor}>
   237→              конструкторе
   238→            </button>
   239→            .
   240→          </AlertDescription>
   241→        </Alert>
   242→      )}
   243→
   244→      {loading ? (
   245→        <div className="space-y-4" role="status" aria-live="polite">
   246→          <p className="sr-only">Загрузка каналов…</p>
   247→          <div className="grid gap-4 md:grid-cols-2 xl:grid-cols-3">
   248→            {[0, 1, 2].map((i) => (
   249→              <div key={i} className="rounded-xl border bg-card p-4">
   250→                <div className="flex items-center gap-3">
   251→                  <Skeleton className="h-10 w-10 rounded-xl" />
   252→                  <div className="min-w-0 flex-1 space-y-2">
   253→                    <Skeleton className="h-4 w-2/3" />
   254→                    <Skeleton className="h-3 w-1/3" />
   255→                  </div>
   256→                </div>
   257→                <Skeleton className="mt-4 h-9 w-full" />
   258→                <div className="mt-3 flex gap-2">
   259→                  <Skeleton className="h-8 flex-1" />
   260→                  <Skeleton className="h-8 w-9" />
   261→                </div>
   262→              </div>
   263→            ))}
   264→          </div>
   265→        </div>
   266→      ) : channels.length === 0 ? (
   267→        <Card className="border-dashed">
   268→          <CardContent className="flex flex-col items-center gap-3 py-12 text-center">
   269→            <div className="flex h-14 w-14 items-center justify-center rounded-2xl bg-primary/10 text-primary">
   270→              <Plug className="h-7 w-7" />
   271→            </div>
   272→            <div className="font-semibold">Ни один канал не подключён</div>
   273→            <p className="max-w-md text-sm text-muted-foreground">
   274→              Подключите Telegram, WhatsApp, MAX — или создайте демо-чат, чтобы общаться с ботом
   275→              прямо на сайте.
   276→            </p>
   277→            <Button onClick={() => setAddOpen(true)}>
   278→              <Plus className="h-4 w-4" /> Подключить канал
   279→            </Button>
   280→          </CardContent>
   281→        </Card>
   282→      ) : (
   283→        <div className="grid gap-4 md:grid-cols-2 xl:grid-cols-3">
   284→          {channels.map((ch) => {
   285→            const meta = CHANNEL_META[ch.type];
   286→            const Icon = meta.icon;
   287→            return (
   288→              <Card key={ch.id}>
   289→                <CardHeader className="pb-3">
   290→                  <div className="flex min-w-0 items-start justify-between gap-2">
   291→                    <div className="flex items-center gap-3 min-w-0">
   292→                      <div className={cn('flex h-10 w-10 shrink-0 items-center justify-center rounded-xl', meta.color)}>
   293→                        <Icon className="h-5 w-5" />
   294→                      </div>
   295→                      <div className="min-w-0">
   296→                        <CardTitle className="truncate text-base">{ch.title}</CardTitle>
   297→                        <CardDescription>{meta.label}</CardDescription>
   298→                      </div>
   299→                    </div>
   300→                    <div className="flex items-center gap-2">
   301→                      <Switch
   302→                        checked={ch.active}
   303→                        onCheckedChange={(v) => toggleActive(ch, v)}
   304→                        aria-label={ch.active ? `Выключить канал «${ch.title}»` : `Включить канал «${ch.title}»`}
   305→                      />
   306→                    </div>
   307→                  </div>
   308→                </CardHeader>
   309→                <CardContent className="space-y-3">
   310→                  {ch.type === 'max' ? (
   311→                    <div className="rounded-lg border bg-muted/40 p-2.5 text-[11px] leading-relaxed text-muted-foreground">
   312→                      <span className="font-medium text-foreground">Вебхук не нужен.</span> Сообщения из MAX
   313→                      принимаются автоматически (long polling). Нажмите «Проверить» — студия свяжется с MAX и
   314→                      покажет имя бота.
   315→                    </div>
   316→                  ) : (
   317→                    <div className="flex items-center gap-2 rounded-lg border bg-muted/40 p-2">
   318→                      <code className="min-w-0 flex-1 truncate text-[11px] text-muted-foreground">
   319→                        {webhookUrl(ch)}
   320→                      </code>
   321→                      <Button
   322→                        variant="ghost"
   323→                        size="icon"
   324→                        className="h-7 w-7 shrink-0"
   325→                        onClick={() => copy(webhookUrl(ch))}
   326→                        aria-label="Скопировать URL вебхука"
   327→                      >
   328→                        <Copy className="h-3.5 w-3.5" />
   329→                      </Button>
   330→                    </div>
   331→                  )}
   332→                  {ch.type === 'web' && (
   333→                    <Button variant="outline" className="w-full" onClick={() => setDemoChannel(ch)}>
   334→                      <Bot className="h-4 w-4" /> Открыть демо-чат
   335→                    </Button>
   336→                  )}
   337→                  {ch.lastStatus && (
   338→                    <div role="status" aria-live="polite" className="min-w-0">
   339→                      <span
   340→                        className={cn(
   341→                          'inline-flex max-w-full items-center gap-1.5 rounded-full border px-2 py-0.5 text-[11px] font-medium',
   342→                          ch.lastStatus.startsWith('OK')
   343→                            ? 'border-emerald-200 bg-emerald-100 text-emerald-800 dark:border-emerald-500/30 dark:bg-emerald-500/15 dark:text-emerald-300'
   344→                            : ch.lastStatus.startsWith('Ошибка')
   345→                              ? 'border-rose-200 bg-rose-100 text-rose-800 dark:border-rose-500/30 dark:bg-rose-500/15 dark:text-rose-300'
   346→                              : 'border-slate-200 bg-slate-100 text-slate-700 dark:border-slate-500/30 dark:bg-slate-500/15 dark:text-slate-300'
   347→                        )}
   348→                      >
   349→                        <span
   350→                          className={cn(
   351→                            'h-1.5 w-1.5 shrink-0 rounded-full',
   352→                            ch.lastStatus.startsWith('OK')
   353→                              ? 'bg-emerald-500'
   354→                              : ch.lastStatus.startsWith('Ошибка')
   355→                                ? 'bg-rose-500'
   356→                                : 'bg-slate-400 dark:bg-slate-500'
   357→                          )}
   358→                          aria-hidden
   359→                        />
   360→                        <span className="sr-only">Статус: </span>
   361→                        <span className="truncate" title={ch.lastStatus}>
   362→                          {ch.lastStatus}
   363→                        </span>
   364→                      </span>
   365→                    </div>
   366→                  )}
   367→                  <div className="flex gap-2">
   368→                    <Button size="sm" variant="outline" className="flex-1" onClick={() => testChannel(ch)}>
   369→                      <Zap className="h-3.5 w-3.5" /> Проверить
   370→                    </Button>
   371→                    <Button
   372→                      size="sm"
   373→                      variant="outline"
   374→                      className="text-destructive hover:bg-destructive/10 hover:text-destructive"
   375→                      onClick={() => setDeleteTarget(ch)}
   376→                      aria-label={`Удалить канал «${ch.title}»`}
   377→                    >
   378→                      <Trash2 className="h-3.5 w-3.5" />
   379→                    </Button>
   380→                  </div>
   381→                  <details className="group">
   382→                    <summary className="flex cursor-pointer list-none items-center gap-1 text-xs font-medium text-primary [&::-webkit-details-marker]:hidden">
   383→                      Как подключить
   384→                      <ChevronDown
   385→                        className="h-3.5 w-3.5 transition-transform group-open:rotate-180"
   386→                        aria-hidden
   387→                      />
   388→                    </summary>
   389→                    <p className="mt-2 whitespace-pre-line rounded-lg border bg-muted/40 p-2.5 text-[11px] leading-relaxed text-muted-foreground">
   390→                      {meta.hint}
   391→                    </p>
   392→                  </details>
   393→                </CardContent>
   394→              </Card>
   395→            );
   396→          })}
   397→        </div>
   398→      )}
   399→
   400→      {/* Интеграция MyTKO (mytko.ru — «Чистая логистика») */}
   401→      <MytkoCard botId={bot.id} />
   402→
   403→      {/* Диалог добавления канала */}
   404→      <Dialog open={addOpen} onOpenChange={setAddOpen}>
   405→        <DialogContent>
   406→          <DialogHeader>
   407→            <DialogTitle>Подключить канал</DialogTitle>
   408→            <DialogDescription>
   409→              Выберите мессенджер и вставьте данные доступа.
   410→            </DialogDescription>
   411→          </DialogHeader>
   412→          <div className="space-y-4">
   413→            <div className="grid grid-cols-2 gap-2 sm:grid-cols-4">
   414→              {(Object.keys(CHANNEL_META) as ChannelItem['type'][]).map((t) => {
   415→                const meta = CHANNEL_META[t];
   416→                const Icon = meta.icon;
   417→                return (
   418→                  <button
   419→                    key={t}
   420→                    onClick={() => setType(t)}
   421→                    aria-pressed={type === t}
   422→                    className={cn(
   423→                      'flex flex-col items-center gap-1.5 rounded-xl border p-3 text-xs font-medium transition-colors focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-primary/40',
   424→                      type === t
   425→                        ? 'border-primary bg-primary/5 shadow-sm ring-1 ring-primary/30'
   426→                        : 'hover:border-primary/40 hover:bg-muted/50'
   427→                    )}
   428→                  >
   429→                    <div className={cn('flex h-8 w-8 items-center justify-center rounded-lg', meta.color)}>
   430→                      <Icon className="h-4 w-4" />
   431→                    </div>
   432→                    {meta.label}
   433→                  </button>
   434→                );
   435→              })}
   436→            </div>
   437→            <div className="space-y-2">
   438→              <Label htmlFor="ch-title" className="text-xs font-medium">Название канала</Label>
   439→              <Input
   440→                id="ch-title"
   441→                placeholder={`${CHANNEL_META[type].label} — основной`}
   442→                value={title}
   443→                onChange={(e) => setTitle(e.target.value)}
   444→              />
   445→            </div>
   446→            {type !== 'web' && (
   447→              <div className="space-y-2">
   448→                <Label htmlFor="ch-token" className="text-xs font-medium">{CHANNEL_META[type].tokenLabel}</Label>
   449→                <Input
   450→                  id="ch-token"
   451→                  type="password"
   452→                  placeholder="вставьте токен"
   453→                  value={token}
   454→                  onChange={(e) => setToken(e.target.value)}
   455→                />
   456→              </div>
   457→            )}
   458→            {type === 'whatsapp' && (
   459→              <div className="space-y-2">
   460→                <Label htmlFor="ch-phone" className="text-xs font-medium">Phone Number ID</Label>
   461→                <Input
   462→                  id="ch-phone"
   463→                  placeholder="например 123456789012345"
   464→                  value={phone}
   465→                  onChange={(e) => setPhone(e.target.value)}
   466→                />
   467→              </div>
   468→            )}
   469→            <p className="whitespace-pre-line rounded-lg border bg-muted/40 p-2.5 text-[11px] leading-relaxed text-muted-foreground">
   470→              {CHANNEL_META[type].hint}
   471→            </p>
   472→          </div>
   473→          <DialogFooter>
   474→            <Button variant="outline" onClick={() => setAddOpen(false)}>
   475→              Отмена
   476→            </Button>
   477→            <Button onClick={createChannel} disabled={busy || (type !== 'web' && !token.trim())}>
   478→              {busy && <Loader2 className="h-4 w-4 animate-spin" />} Подключить
   479→            </Button>
   480→          </DialogFooter>
   481→        </DialogContent>
   482→      </Dialog>
   483→
   484→      {/* Демо-чат */}
   485→      <Dialog open={!!demoChannel} onOpenChange={(o) => !o && setDemoChannel(null)}>
   486→        <DialogContent className="gap-0 p-0 sm:max-w-md">
   487→          <VisuallyHidden>
   488→            <DialogTitle>Демо-чат</DialogTitle>

   528→function DemoChat({ secret }: { secret: string }) {
   529→  const [messages, setMessages] = useState<
   530→    { id: string; role: string; text: string; buttons?: { id: string; text: string }[] }[]
   531→  >([]);
   532→  const [input, setInput] = useState('');
   533→  const [sending, setSending] = useState(false);
   534→  const convRef = useRef<string | null>(null);
   535→  const visitorRef = useRef<string>('');
   536→  const scrollRef = useRef<HTMLDivElement>(null);
   537→  const storageKey = `botstudio_demo_conv_${secret}`;
   538→  const visitorKey = 'botstudio_visitor';
   539→
   540→  // Стабильный id посетителя: один гость = одно обращение,
   541→  // даже если conversationId был потерян
   542→  useEffect(() => {
   543→    try {
   544→      let v = window.localStorage.getItem(visitorKey);
   545→      if (!v) {
   546→        v =
   547→          typeof crypto !== 'undefined' && 'randomUUID' in crypto
   548→            ? crypto.randomUUID().replace(/-/g, '')
   549→            : Math.random().toString(36).slice(2) + Date.now().toString(36);
   550→        window.localStorage.setItem(visitorKey, v);
   551→      }
   552→      visitorRef.current = v;
   553→    } catch {
   554→      visitorRef.current = Math.random().toString(36).slice(2) + Date.now().toString(36);
   555→    }
   556→  }, [visitorKey]);
   557→
   558→  // Восстановить диалог посетителя (как реальный виджет сайта)
   559→  useEffect(() => {
   560→    try {
   561→      const saved = window.localStorage.getItem(storageKey);
   562→      if (saved) {
   563→        convRef.current = saved;
   564→        fetch(`/api/webhook/demo/${secret}?conversationId=${saved}`)
   565→          .then((r) => r.json())
   566→          .then((d) => {
   567→            if (d.messages?.length) setMessages(d.messages);
   568→          })
   569→          .catch(() => {});
   570→      }
   571→    } catch {
   572→      /* ignore */
   573→    }
   574→  }, [secret, storageKey]);
   575→
   576→  const scrollToBottom = () => {
   577→    requestAnimationFrame(() => {
   578→      scrollRef.current?.scrollTo({ top: scrollRef.current.scrollHeight });
   579→    });
   580→  };
   581→
   582→  useEffect(() => {
   583→    scrollToBottom();
   584→  }, [messages]);
   585→
   586→  const poll = useCallback(async () => {
   587→    if (!convRef.current) return;
   588→    try {
   589→      const d = await api<{
   590→        messages: { id: string; role: string; text: string; buttons?: { id: string; text: string }[] }[];
   591→      }>(`/api/webhook/demo/${secret}?conversationId=${convRef.current}`);
   592→      setMessages((prev) => (d.messages.length > prev.length ? d.messages : prev));
   593→    } catch {
   594→      /* ignore */
   595→    }
   596→  }, [secret]);
   597→
   598→  useEffect(() => {
   599→    const t = setInterval(poll, 4000);
   600→    return () => clearInterval(t);
   601→  }, [poll]);
   602→
   603→  const send = async (text: string) => {
   604→    const t = text.trim();
   605→    if (!t || sending) return;
   606→    setSending(true);
   607→    setInput('');
   608→    setMessages((prev) => [...prev, { id: `local-${Date.now()}`, role: 'user', text: t }]);
   609→    scrollToBottom();
   610→    try {
   611→      const d = await api<{
   612→        conversationId: string;
   613→        messages: { id: string; role: string; text: string; buttons?: { id: string; text: string }[] }[];
   614→      }>(`/api/webhook/demo/${secret}`, {
   615→        method: 'POST',
   616→        body: JSON.stringify({ text: t, conversationId: convRef.current, visitorId: visitorRef.current || undefined }),
   617→      });
   618→      convRef.current = d.conversationId;
   619→      try {
   620→        window.localStorage.setItem(storageKey, d.conversationId);
   621→      } catch {
   622→        /* ignore */
   623→      }
   624→      setMessages(d.messages);
   625→      scrollToBottom();
   626→    } catch (e) {
   627→      setMessages((prev) => [
   628→        ...prev,
   629→        { id: `err-${Date.now()}`, role: 'bot', text: e instanceof Error ? e.message : 'Ошибка' },
   630→      ]);
   631→    } finally {
   632→      setSending(false);
   633→    }
   634→  };
   635→
   636→  return (
   637→    <div className="flex flex-col">
   638→      <div className="flex items-center gap-3 rounded-t-xl bg-gradient-to-r from-emerald-600 to-teal-600 p-4 text-white">
   639→        <div className="flex h-10 w-10 items-center justify-center rounded-full bg-white/20">
   640→          <Bot className="h-5 w-5" />
   641→        </div>
   642→        <div className="flex-1">
   643→          <div className="text-sm font-semibold leading-tight">Чат поддержки</div>
   644→          <div className="flex items-center gap-1 text-xs text-white/80">
   645→            <span className="h-1.5 w-1.5 rounded-full bg-emerald-300" /> онлайн
   646→          </div>
   647→        </div>
   648→        <Badge variant="secondary" className="text-[10px]">
   649→          демо сайта
   650→        </Badge>
   651→      </div>
   652→      <div ref={scrollRef} className="h-80 space-y-2.5 overflow-y-auto bg-muted/40 p-3">
   653→        {messages.length === 0 && (
   654→          <div className="flex h-full items-center justify-center text-center text-sm text-muted-foreground">
   655→            Напишите сообщение — бот ответит
   656→            <br />
   657→            как в настоящем виджете на сайте
   658→          </div>
   659→        )}
   660→        {messages.map((m) => (
   661→          <div key={m.id} className={cn('flex', m.role === 'user' ? 'justify-end' : 'justify-start')}>
   662→            <div
   663→              className={cn(
   664→                'max-w-[85%] whitespace-pre-wrap rounded-2xl px-3 py-2 text-sm leading-snug',
   665→                m.role === 'user'
   666→                  ? 'rounded-br-md bg-primary text-primary-foreground'
   667→                  : 'rounded-bl-md border bg-card'
   668→              )}
   669→            >
   670→              {m.text}
   671→            </div>
   672→          </div>
   673→        ))}
   674→        {(() => {
   675→          const lastButtons = [...messages].reverse().find((m) => m.role === 'bot' && m.buttons?.length)?.buttons;
   676→          if (!lastButtons?.length) return null;
   677→          return (
   678→            <div className="flex flex-wrap gap-2 pt-1">
   679→              {lastButtons.map((b) => (
   680→                <button
   681→                  key={b.id}
   682→                  onClick={() => send(b.text)}
   683→                  disabled={sending}
   684→                  className="rounded-full border border-primary/40 bg-primary/5 px-3 py-1.5 text-xs font-medium text-primary transition-colors hover:bg-primary hover:text-primary-foreground disabled:opacity-50"
   685→                >
   686→                  {b.text}
   687→                </button>
   688→              ))}
   689→            </div>
   690→          );
   691→        })()}
   692→      </div>
   693→      <form
   694→        className="flex items-center gap-2 border-t p-3"
   695→        onSubmit={(e) => {
   696→          e.preventDefault();
   697→          send(input);
   698→        }}
   699→      >
   700→        <Input placeholder="Сообщение…" value={input} onChange={(e) => setInput(e.target.value)} />
   701→        <Button type="submit" size="icon" className="h-10 w-10 shrink-0" disabled={sending || !input.trim()}>
   702→          {sending ? <Loader2 className="h-4 w-4 animate-spin" /> : <Send className="h-4 w-4" />}
   703→        </Button>
   704→      </form>
   705→    </div>
