   437→function DemoChat({ secret }: { secret: string }) {
   438→  const [messages, setMessages] = useState<
   439→    { id: string; role: string; text: string; buttons?: { id: string; text: string }[] }[]
   440→  >([]);
   441→  const [input, setInput] = useState('');
   442→  const [sending, setSending] = useState(false);
   443→  const convRef = useRef<string | null>(null);
   444→  const visitorRef = useRef<string>('');
   445→  const scrollRef = useRef<HTMLDivElement>(null);
   446→  const storageKey = `botstudio_demo_conv_${secret}`;
   447→  const visitorKey = 'botstudio_visitor';
   448→
   449→  // Стабильный id посетителя: один гость = одно обращение,
   450→  // даже если conversationId был потерян
   451→  useEffect(() => {
   452→    try {
   453→      let v = window.localStorage.getItem(visitorKey);
   454→      if (!v) {
   455→        v =
   456→          typeof crypto !== 'undefined' && 'randomUUID' in crypto
   457→            ? crypto.randomUUID().replace(/-/g, '')
   458→            : Math.random().toString(36).slice(2) + Date.now().toString(36);
   459→        window.localStorage.setItem(visitorKey, v);
   460→      }
   461→      visitorRef.current = v;
   462→    } catch {
   463→      visitorRef.current = Math.random().toString(36).slice(2) + Date.now().toString(36);
   464→    }
   465→  }, [visitorKey]);
   466→
   467→  // Восстановить диалог посетителя (как реальный виджет сайта)
   468→  useEffect(() => {
   469→    try {
   470→      const saved = window.localStorage.getItem(storageKey);
   471→      if (saved) {
   472→        convRef.current = saved;
   473→        fetch(`/api/webhook/demo/${secret}?conversationId=${saved}`)
   474→          .then((r) => r.json())
   475→          .then((d) => {
   476→            if (d.messages?.length) setMessages(d.messages);
   477→          })
   478→          .catch(() => {});
   479→      }
   480→    } catch {
   481→      /* ignore */
   482→    }
   483→  }, [secret, storageKey]);
   484→
   485→  const scrollToBottom = () => {
   486→    requestAnimationFrame(() => {
   487→      scrollRef.current?.scrollTo({ top: scrollRef.current.scrollHeight });
   488→    });
   489→  };
   490→
   491→  useEffect(() => {
   492→    scrollToBottom();
   493→  }, [messages]);
   494→
   495→  const poll = useCallback(async () => {
   496→    if (!convRef.current) return;
   497→    try {
   498→      const d = await api<{
   499→        messages: { id: string; role: string; text: string; buttons?: { id: string; text: string }[] }[];
   500→      }>(`/api/webhook/demo/${secret}?conversationId=${convRef.current}`);
   501→      setMessages((prev) => (d.messages.length > prev.length ? d.messages : prev));
   502→    } catch {
   503→      /* ignore */
   504→    }
   505→  }, [secret]);
   506→
   507→  useEffect(() => {
   508→    const t = setInterval(poll, 4000);
   509→    return () => clearInterval(t);
   510→  }, [poll]);
   511→
   512→  const send = async (text: string) => {
   513→    const t = text.trim();
   514→    if (!t || sending) return;
   515→    setSending(true);
   516→    setInput('');
   517→    setMessages((prev) => [...prev, { id: `local-${Date.now()}`, role: 'user', text: t }]);
   518→    scrollToBottom();
   519→    try {
