    96→    96→function fmtAgo(ts: number, now: number): string {
    97→    97→  const s = Math.max(0, Math.round((now - ts) / 1000));
    98→    98→  if (s < 8) return 'только что';
    99→    99→  if (s < 60) return `${s} с назад`;
   100→   100→  const m = Math.floor(s / 60);
   101→   101→  if (m < 60) return `${m} мин назад`;
   102→   102→  return `${Math.floor(m / 60)} ч назад`;
   103→   103→}

   105→   105→/** Дедуп-слияние сообщений по id (append-only) + сортировка по createdAt.
   106→   106→ *  Если нового ничего нет — возвращает прежний массив (без ре-рендера).

   108→   108→function mergeMessages(prev: OrderMessageDto[], incoming: OrderMessageDto[]): OrderMessageDto[] {
   109→   109→  if (incoming.length === 0) return prev;
   110→   110→  const known = new Set(prev.map((m) => m.id));
   111→   111→  const fresh = incoming.filter((m) => !known.has(m.id));
   112→   112→  if (fresh.length === 0) return prev;
   113→   113→  return [...prev, ...fresh].sort(
   114→   114→    (a, b) => new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime()
   115→   115→  );
   116→   116→}

   146→   146→function OrdersMap({
   147→   147→  orders,
   148→   148→  showCompleted,
   149→   149→  mytkoEnabled,
   150→   150→  placementOrderId,
   151→   151→  selectedId,
   152→   152→  focusOrderId,
   153→   153→  focusTick,
   154→   154→  onSelect,
   155→   155→  onPlace,
   156→   156→  onHighlight,
   157→   157→}: {
   158→   158→  orders: OrderDto[];
   159→   159→  showCompleted: boolean;
   160→   160→  mytkoEnabled: boolean;
   161→   161→  placementOrderId: string | null;
   162→   162→  selectedId: string | null;
   163→   163→  focusOrderId: string | null;
   164→   164→  focusTick: number;
   165→   165→  onSelect: (id: string) => void;
   166→   166→  onPlace: (lat: number, lng: number) => void;
   167→   167→  /** Выделить заявку без открытия карточки (клик по списку точек) */
   168→   168→  onHighlight: (id: string) => void;
   169→   169→}) {
   170→   170→  const containerRef = useRef<HTMLDivElement>(null);
   171→   171→  const mapRef = useRef<MlMap | null>(null);
   172→   172→  const mlRef = useRef<typeof import('maplibre-gl') | null>(null);
   173→   173→  /** маркеры по id заявки — чтобы сдвигать их при «spiderfy» совпадающих точек */
   174→   174→  const markersRef = useRef<Map<string, MlMarker>>(new Map());
   175→   175→  /** исходные координаты заявки (без сдвига) — для пересчёта раскладки на каждом move */
   176→   176→  const baseCoordsRef = useRef<Map<string, { lat: number; lng: number }>>(new Map());
   177→   177→  /** id → заявка: эффекту выделения, чтобы перекрашивать маркеры без пересоздания */
   178→   178→  const ordersByIdRef = useRef<Map<string, OrderDto>>(new Map());
   179→   179→  /** зеркальная копия selectedId — доступ из эффектов маркеров без их пересоздания */
   180→   180→  const selectedIdRef = useRef<string | null>(selectedId);
   181→   181→  const [ready, setReady] = useState(false);
   182→   182→  const [failed, setFailed] = useState(false);
   183→   183→  const [listOpen, setListOpen] = useState(false);
   184→   184→
   185→   185→  // Инициализация карты (динамический импорт — компонент клиентский)
   186→   186→  useEffect(() => {
   187→   187→    let cancelled = false;
   188→   188→    (async () => {
   189→   189→      try {
   190→   190→        const ml = await import('maplibre-gl');
   191→   191→        if (cancelled || !containerRef.current) return;
   192→   192→        // Воркер MapLibre не собирается Turbopack'ом — берём локальную копию из public/
   193→   193→        ml.setWorkerUrl('/maplibre-gl-worker.mjs');
   194→   194→        mlRef.current = ml;
   195→   195→        const map = new ml.Map({
   196→   196→          container: containerRef.current,
   197→   197→          style: 'https://tiles.openfreemap.org/styles/liberty',
   198→   198→          center: [32.3, 58.35], // Новгородская область (Великий Новгород — Боровичи — Валдай)
   199→   199→          zoom: 7,
   200→   200→        });
   201→   201→        map.addControl(new ml.NavigationControl({ showCompass: false }), 'top-right');
   202→   202→        map.on('load', () => {
   203→   203→          if (!cancelled) setReady(true);
   204→   204→        });
   205→   205→        map.on('error', (e) => {
   206→   206→          // Тайловая сеть недоступна — сообщаем, но карту не убиваем
   207→   207→          console.warn('[orders map]', e?.error?.message ?? e);
   208→   208→        });
   209→   209→        mapRef.current = map;
   210→   210→      } catch (err) {
   211→   211→        console.error('[orders map] init failed', err);
   212→   212→        if (!cancelled) setFailed(true);
   213→   213→      }
   214→   214→    })();
   215→   215→    return () => {
   216→   216→      cancelled = true;
   217→   217→      mapRef.current?.remove();
   218→   218→      mapRef.current = null;
   219→   219→      setReady(false);
   220→   220→    };
   221→   221→  }, []);
   222→   222→
   223→   223→  // Маркеры заявок
   224→   224→  const visible = orders.filter(
   225→   225→    (o) => o.lat != null && o.lng != null && (showCompleted || ACTIVE_STATUSES.includes(o.status))
   226→   226→  );
   227→   227→
   228→   228→  /**
   229→   229→   * «Spiderfy» совпадающих точек: несколько заявок с одним адресом геокодятся
   230→   230→   * в идентичные координаты и ложатся маркерами друг на друга — видна только
   231→   231→   * одна. Группируем их по округлённым координатам и раскладываем кружком
   232→   232→   * с ПОСТОЯННЫМ пиксельным радиусом (пересчёт на каждом move/zoom), чтобы
   233→   233→   * все точки были видны и кликабельны на любом масштабе карты.
   234→   234→   */
   235→   235→  const spreadMarkers = useCallback(() => {
   236→   236→    const map = mapRef.current;
   237→   237→    if (!map) return;
   238→   238→
   239→   239→    // группы совпадающих координат (6 знаков ≈ 0.1 м — попадают и «почти равные»)
   240→   240→    const groups = new Map<string, string[]>();
   241→   241→    for (const [id, c] of baseCoordsRef.current) {
   242→   242→      const key = `${c.lat.toFixed(6)}|${c.lng.toFixed(6)}`;
   243→   243→      const g = groups.get(key);
   244→   244→      if (g) g.push(id);
   245→   245→      else groups.set(key, [id]);
   246→   246→    }
   247→   247→
   248→   248→    const zoom = map.getZoom();
   249→   249→    // пикселей на градус долготы в Web Mercator (тайл MapLibre = 512px!)
   250→   250→    const pxPerDegLng = 512 * Math.pow(2, zoom) / 360;
   251→   251→    const SPREAD_PX = 44; // диаметр раскладки вокруг истинной точки
   252→   252→
   253→   253→    for (const ids of groups.values()) {
   254→   254→      const n = ids.length;
   255→   255→      ids.forEach((id, i) => {
   256→   256→        const marker = markersRef.current.get(id);
   257→   257→        const base = baseCoordsRef.current.get(id);
   258→   258→        if (!marker || !base) return;
   259→   259→        if (n === 1) {
   260→   260→          marker.setLngLat([base.lng, base.lat]);
   261→   261→          return;
   262→   262→        }
   263→   263→        // равномерно по окружности; при 2 точках — слева и справа от истинной
   264→   264→        const angle = (2 * Math.PI * i) / n + (n === 2 ? Math.PI / 2 : Math.PI / 6);
   265→   265→        const dxPx = Math.cos(angle) * (SPREAD_PX / 2);
   266→   266→        const dyPx = Math.sin(angle) * (SPREAD_PX / 2);
   267→   267→        const dLng = dxPx / pxPerDegLng;
   268→   268→        const dLat = (dyPx / pxPerDegLng) * Math.cos((base.lat * Math.PI) / 180);
   269→   269→        marker.setLngLat([base.lng + dLng, base.lat + dLat]);
   270→   270→      });
   271→   271→    }
   272→   272→  }, []);
   273→   273→
   274→   274→  // ДИФФ-обновление маркеров (IMP-F10): удаляем исчезнувшие, обновляем координаты
   275→   275→  // и цвет существующих, добавляем только новые — поллинг больше не пересоздаёт
   276→   276→  // все маркеры и карта не «мигает». Выделение (selectedId) — отдельным эффектом ниже.
   277→   277→  useEffect(() => {
   278→   278→    const map = mapRef.current;
   279→   279→    const ml = mlRef.current;
   280→   280→    if (!map || !ml || !ready) return;
   281→   281→
   282→   282→    // (a) входящий набор видимых заявок
   283→   283→    const incoming = new Map<string, OrderDto>();
   284→   284→    for (const o of visible) incoming.set(o.id, o);
   285→   285→    ordersByIdRef.current = incoming;
   286→   286→
   287→   287→    // (b) удаляем маркеры, которых больше нет
   288→   288→    for (const [id, m] of markersRef.current) {
   289→   289→      if (!incoming.has(id)) {
   290→   290→        m.remove();
   291→   291→        markersRef.current.delete(id);
   292→   292→        baseCoordsRef.current.delete(id);
   293→   293→      }
   294→   294→    }
   295→   295→
   296→   296→    // (c)+(d) обновляем существующие / добавляем только новые
   297→   297→    for (const o of visible) {
   298→   298→      const lat = o.lat as number;
   299→   299→      const lng = o.lng as number;
   300→   300→      baseCoordsRef.current.set(o.id, { lat, lng });
   301→   301→      const sig = `${o.status}|${o.type}`;
   302→   302→      const existing = markersRef.current.get(o.id);
   303→   303→      if (existing) {
   304→   304→        const el = existing.getElement();
   305→   305→        if (el.dataset.sig !== sig) {
   306→   306→          // статус/тип изменились — перекрашиваем без пересоздания
   307→   307→          el.dataset.sig = sig;
   308→   308→          el.className = markerElClassName(o, o.id === selectedIdRef.current);
   309→   309→        }
   310→   310→      } else {
   311→   311→        const el = document.createElement('button');
   312→   312→        el.type = 'button';
   313→   313→        el.dataset.sig = sig;
   314→   314→        el.title = `Заявка №${o.number} — ${ORDER_TYPE_LABELS[o.type] ?? ''}`;
   315→   315→        // Без hover:scale — масштабирование сдвигает маркер и «прыгает»;
   316→   316→        // подсвечиваем тенью и рамкой, размер элемента не меняется
   317→   317→        el.className = markerElClassName(o, o.id === selectedIdRef.current);
   318→   318→        el.textContent = String(o.number);
   319→   319→        el.addEventListener('click', (e) => {
   320→   320→          e.stopPropagation();
   321→   321→          onSelect(o.id);
   322→   322→        });
   323→   323→        const marker = new ml.Marker({ element: el })
   324→   324→          .setLngLat([lng, lat])
   325→   325→          .addTo(map);
   326→   326→        markersRef.current.set(o.id, marker);
   327→   327→      }
   328→   328→    }
   329→   329→
   330→   330→    // (e) раскладываем совпадающие точки (spiderfy, 44px) и пересчитываем на каждом move
   331→   331→    spreadMarkers();
   332→   332→    map.on('move', spreadMarkers);
   333→   333→    return () => {
   334→   334→      map.off('move', spreadMarkers);
   335→   335→    };
   336→   336→    // зависимость — сериализованный список видимых маркеров (БЕЗ selectedId)
   337→   337→  }, [ready, visible.map((o) => `${o.id}:${o.lat}:${o.lng}:${o.status}`).join('|'), onSelect, spreadMarkers]);
   338→   338→
   339→   339→  // Выделение выбранной заявки — ОТДЕЛЬНЫЙ эффект: только переключаем класс
   340→   340→  // DOM-элемента существующих маркеров (getElement), ничего не пересоздаём
   341→   341→  useEffect(() => {
   342→   342→    selectedIdRef.current = selectedId;
   343→   343→    for (const [id, m] of markersRef.current) {
   344→   344→      const o = ordersByIdRef.current.get(id);
   345→   345→      if (o) m.getElement().className = markerElClassName(o, id === selectedId);
   346→   346→    }
   347→   347→  }, [selectedId]);
   348→   348→
   349→   349→  // Наведение на заявку из списка/карточки: перелетаем к точке (focusTick —
   350→   350→  // чтобы повторный клик по той же заявке тоже срабатывал)
   351→   351→  useEffect(() => {
   352→   352→    const map = mapRef.current;
   353→   353→    if (!map || !ready || !focusOrderId) return;
   354→   354→    const o = visible.find((x) => x.id === focusOrderId);
   355→   355→    if (!o || o.lat == null || o.lng == null) return;
   356→   356→    map.flyTo({ center: [o.lng, o.lat], zoom: Math.max(map.getZoom(), 15), duration: 800 });
   357→   357→  }, [ready, focusOrderId, focusTick]);
   358→   358→
   359→   359→  // Во время режима указания точки список скрываем (баннер занимает то же место)
   360→   360→  useEffect(() => {
   361→   361→    if (placementOrderId) setListOpen(false);
   362→   362→  }, [placementOrderId]);
   363→   363→
   364→   364→  // Переход к точке из списка: перелёт + выделение, без открытия карточки
   365→   365→  const jumpToListPoint = useCallback(
   366→   366→    (o: OrderDto) => {
   367→   367→      const map = mapRef.current;
   368→   368→      if (!map || o.lat == null || o.lng == null) return;
   369→   369→      onHighlight(o.id);
   370→   370→      map.flyTo({ center: [o.lng, o.lat], zoom: Math.max(map.getZoom(), 15), duration: 800 });
   371→   371→      setListOpen(false);
   372→   372→    },
   373→   373→    [onHighlight]
   374→   374→  );
   375→   375→
   376→   376→  // Режим указания точки: клик по карте → координаты заявки
   377→   377→  const handleMapClick = useCallback(
   378→   378→    (lat: number, lng: number) => onPlace(lat, lng),
   379→   379→    [onPlace]
   380→   380→  );
   381→   381→  useEffect(() => {
   382→   382→    const map = mapRef.current;
   383→   383→    if (!map || !ready || !placementOrderId) return;
   384→   384→    const onClick = (e: { lngLat: { lat: number; lng: number } }) =>
   385→   385→      handleMapClick(e.lngLat.lat, e.lngLat.lng);
   386→   386→    map.on('click', onClick);
   387→   387→    map.getCanvas().style.cursor = 'crosshair';
   388→   388→    return () => {
   389→   389→      map.off('click', onClick);
   390→   390→      map.getCanvas().style.cursor = '';
   391→   391→    };
   392→   392→  }, [ready, placementOrderId, handleMapClick]);
   393→   393→
   394→   394→  if (failed) {
   395→   395→    return (
   396→   396→      <div className="flex h-full min-h-[420px] flex-col items-center justify-center gap-3 rounded-xl border bg-muted/30 text-center text-muted-foreground">
   397→   397→        <MapPin className="h-10 w-10 opacity-30" />
   398→   398→        <p className="max-w-xs text-sm">
   399→   399→          Не удалось загрузить карту OpenFreeMap. Проверьте доступ к tiles.openfreemap.org и
   400→   400→          обновите страницу.
   401→   401→        </p>
   402→   402→      </div>
   403→   403→    );
   404→   404→  }
   405→   405→
   406→   406→  return (
   407→   407→    <div className="relative h-full min-h-[380px] overflow-hidden rounded-xl border bg-muted/30">
   408→   408→      <div ref={containerRef} className="h-full w-full" aria-label="Карта заявок" />
   409→   409→      {!ready && (
   410→   410→        <div className="absolute inset-0 flex items-center justify-center gap-2 text-sm text-muted-foreground">
   411→   411→          <Loader2 className="h-4 w-4 animate-spin" /> Загружаю карту OpenFreeMap…
   412→   412→        </div>
   413→   413→      )}
   414→   414→      {/* Легенда (компактная, не перекрывает атрибуцию и контролы) */}
   415→   415→      <div className="absolute bottom-2.5 left-2.5 z-10 flex flex-wrap items-center gap-x-2.5 gap-y-0.5 rounded-lg border bg-background/95 px-2.5 py-1.5 text-[10px] leading-tight shadow-sm backdrop-blur sm:bottom-3 sm:left-3 sm:text-[11px]">
   416→   416→        <span className="flex items-center gap-1.5">
   417→   417→          <span className="h-2.5 w-2.5 rounded-full bg-emerald-500" /> отходы
   418→   418→        </span>
   419→   419→        <span className="flex items-center gap-1.5">
   420→   420→          <span className="h-2.5 w-2.5 rounded-full bg-amber-500" /> КГМ
   421→   421→        </span>
   422→   422→        <span className="flex items-center gap-1.5">
   423→   423→          <span className="h-2.5 w-2.5 rounded-full bg-slate-400" /> выполненные
   424→   424→        </span>
   425→   425→      </div>
   426→   426→      {/* Счётчик маркеров + кнопка разворачиваемого списка точек */}
   427→   427→      <div className="absolute left-2.5 top-2.5 z-10 flex items-start gap-2 sm:left-3 sm:top-3">
   428→   428→        <div
   429→   429→          role="status"
   430→   430→          aria-label={`Точек на карте: ${visible.length}`}
   431→   431→          className="rounded-lg border bg-background/95 px-2.5 py-1.5 text-xs shadow-sm backdrop-blur"
   432→   432→        >
   433→   433→          <MapPin className="mr-1 inline h-3 w-3 text-primary" aria-hidden />

   522→   522→/** Optimistic-пузырь: пока sending — «отправляется…», при ошибке — «Повторить» (текст не теряется) */

   524→   524→
   525→   525→function OrderChat({
   526→   526→  botId,
   527→   527→  orderId,
   528→   528→  hasConversation,
   529→   529→}: {
   530→   530→  botId: string;
   531→   531→  orderId: string;
   532→   532→  hasConversation: boolean;
   533→   533→}) {
   534→   534→  const { toast } = useToast();
   535→   535→  const [messages, setMessages] = useState<OrderMessageDto[]>([]);
   536→   536→  const [pending, setPending] = useState<PendingMessage[]>([]);
   537→   537→  const [text, setText] = useState('');
   538→   538→  const [sending, setSending] = useState(false);
   539→   539→  const scrollRef = useRef<HTMLDivElement>(null);
   540→   540→  /** Пользователь у дна переписки? Скролл при новых сообщениях — только тогда */
   541→   541→  const isNearBottomRef = useRef(true);
   542→   542→  /** Ключ активной заявки — защита от гонки: ответ старого запроса не должен
   543→   543→   *  попасть в чат новой заявки при быстром переключении карточек */
   544→   544→  const activeKeyRef = useRef('');
   545→   545→

   556→   556→        setMessages((prev) => mergeMessages(prev, d.messages));
   557→   557→      } catch {
   558→   558→        /* polling errors ignored */
   559→   559→      }

   563→   563→
