    51→/**
    52→ * Валидация телефона РФ: 10–11 цифр (можно с +7/8, скобками, дефисами).
    53→ * Возвращает true, если телефон похож на настоящий.
    54→ */
    55→export function isValidPhone(input: string): boolean {
    56→  const digits = input.replace(/\D/g, '');
    57→  if (digits.length === 11 && (digits[0] === '7' || digits[0] === '8')) return true;
    58→  if (digits.length === 10) return true;
    59→  return false;
    60→}

    62→/**
    63→ * Проверка ответа на вопрос по data.validate.
    64→ * Возвращает текст ошибки для переспрашивания или null, если ответ корректен.
    65→ * «пропустить» обрабатывается вызывающим кодом.
    66→ */
    67→export async function validateAnswer(
    68→  data: FlowNode['data'],
    69→  input: string,
    70→  vars: Record<string, string>
    71→): Promise<string | null> {
    72→  if (data.validate === 'phone') {
    73→    if (isValidPhone(input)) return null;
    74→    return (
    75→      'Телефон выглядит некорректно 🙏 Введите номер в формате +7 900 123-45-67 (10–11 цифр) ' +
    76→      'или напишите «пропустить».'
    77→    );
    78→  }
    79→
    80→  if (data.validate === 'address') {
    81→    const q = input.trim();
    82→    if (q.length < 3) {
    83→      return 'Уточните адрес подробнее: улица и номер дома (например: улица Гоголя, 5).';
    84→    }
    85→    // Город шагом раньше (выбор города обслуживания)
    86→    const city = ['city', 'kgm_city']
    87→      .map((k) => (vars[k] ?? '').trim())
    88→      .find((c) => c.length > 0);
    89→
    90→    // Строгий режим: только РФ + при известном городе проверяем, что найденный
    91→    // объект действительно в этом городе (мусорные совпадения не проходят)
    92→    const ok = city
    93→      ? q.toLowerCase().includes(city.toLowerCase())
    94→        ? await geocodeVerify(q, city)
    95→        : await geocodeVerify(`${city}, ${q}`, city)
    96→      : await geocodeVerify(q);
    97→
    98→    if (ok) return null;
    99→    return (
   100→      'Не нашёл такой адрес на карте 🤔 Проверьте улицу и номер дома ' +
   101→      '(можно с корпусом: улица Гоголя, 5к2). Либо напишите «пропустить», чтобы продолжить без проверки.'
   102→    );
   103→  }
   104→
   105→  return null;
   106→}

   142→/** Подстановка переменных {{name}} в текст */
   143→export function interpolate(
   144→  text: string | undefined,
   145→  vars: Record<string, string>,
   146→  input?: string
   147→): string {
   148→  if (!text) return '';
   149→  let out = text.replace(/\{\{\s*([\w.]+)\s*\}\}/g, (_m, key: string) => {
   150→    if (key === 'input' && input !== undefined) return input;
   151→    const v = vars[key];
   152→    return v !== undefined ? v : '';
   153→  });
   154→  // Приглаживаем пустые подстановки: «Завтра, » → «Завтра», «Боровичи,» → «Боровичи»
   155→  out = out.replace(/,\s*,+/g, ', ');
   156→  out = out.replace(/,\s*$/gm, '');
   157→  return out;
   158→}

   160→function findNode(flow: Flow, id: string): FlowNode | null {
   161→  return flow.nodes.find((n) => n.id === id) ?? null;
   162→}
   163→
   164→function findStart(flow: Flow): FlowNode | null {
   165→  return flow.nodes.find((n) => n.type === 'start') ?? flow.nodes[0] ?? null;
   166→}
   167→
   168→function nextNode(flow: Flow, nodeId: string, handle?: string | null): FlowNode | null {
   169→  const edge = flow.edges.find(
   170→    (e) => e.source === nodeId && (handle ? (e.sourceHandle ?? '') === handle : true)
   171→  );
   172→  return edge ? findNode(flow, edge.target) : null;
   173→}

   175→function findMenuNode(flow: Flow, lastMenuId?: string): FlowNode | null {
   176→  if (lastMenuId) {
   177→    const n = findNode(flow, lastMenuId);
   178→    if (n?.type === 'buttons') return n;
   179→  }
   180→  return flow.nodes.find((n) => n.type === 'buttons') ?? null;
   181→}

   183→/**
   184→ * Главное меню сценария: первый узел «Кнопки» на пути от старта
   185→ * (BFS), иначе — первый узел «Кнопки» в сценарии. Используется,
   186→ * чтобы вернуть пользователя в меню (например, после закрытия обращения).
   187→ */
   188→export function findMainMenu(flow: Flow): FlowNode | null {
   189→  const start = findStart(flow);
   190→  if (start) {
   191→    const visited = new Set<string>([start.id]);
   192→    const queue: string[] = [start.id];
   193→    while (queue.length) {
   194→      const id = queue.shift() as string;
   195→      const node = findNode(flow, id);
   196→      if (!node) continue;
   197→      if (node.type === 'buttons') return node;
   198→      for (const e of flow.edges) {
   199→        if (e.source === id && !visited.has(e.target)) {
   200→          visited.add(e.target);
   201→          queue.push(e.target);
   202→        }
   203→      }
   204→    }
   205→  }
   206→  return flow.nodes.find((n) => n.type === 'buttons') ?? null;
   207→}

   209→/**
   210→ * Команда старта сценария (/start, «старт», «начать») — детерминированный вход
   211→ * в сценарий, минуя ИИ-ассистента (мессенджеры шлют /start при первом открытии бота).
   212→ */
   213→function isStartCommand(text: string): boolean {

   217→function evalCondition(
   218→  cond: FlowNode['data']['condition'],
   219→  vars: Record<string, string>
   220→): boolean {
   221→  if (!cond) return false;
   222→  const leftRaw = (cond.left ?? '').trim();
   223→  const left = leftRaw.includes('{{')
   224→    ? interpolate(leftRaw, vars)
   225→    : (vars[leftRaw] ?? leftRaw);
   226→  const rightRaw = cond.right ?? '';
   227→  const right = rightRaw.includes('{{') ? interpolate(rightRaw, vars) : rightRaw;
   228→  const op: ConditionOp = cond.op ?? 'eq';
   229→  switch (op) {
   230→    case 'eq':
   231→      return left.trim().toLowerCase() === right.trim().toLowerCase();
   232→    case 'neq':
   233→      return left.trim().toLowerCase() !== right.trim().toLowerCase();
   234→    case 'contains':
   235→      return left.toLowerCase().includes(right.toLowerCase());
   236→    case 'not_contains':
   237→      return !left.toLowerCase().includes(right.toLowerCase());
   238→    case 'gt':
   239→      return parseFloat(left) > parseFloat(right);
   240→    case 'lt':
   241→      return parseFloat(left) < parseFloat(right);
   242→    case 'empty':
   243→      return left.trim() === '';
   244→    case 'not_empty':
   245→      return left.trim() !== '';
   246→    default:
   247→      return false;
   248→  }
   249→}

   259→async function runAI(
   260→  data: AiAskData,
   261→  state: EngineState,
   262→  lastInput: string,
   263→  extraSystem?: string
   264→): Promise<string> {
   265→  const zai = await getZai();
   266→
   267→  const sysParts: string[] = [];
   268→  sysParts.push(
   269→    data.prompt?.trim() ||
   270→      'Ты — вежливый бот техподдержки. Отвечай кратко, по делу и на языке пользователя.'
   271→  );
   272→  if (extraSystem?.trim()) {
   273→    sysParts.push(extraSystem.trim());
   274→  }
   275→  if (data.knowledge?.trim()) {
   276→    sysParts.push(`База знаний компании:\n${data.knowledge.trim()}`);
   277→  }
   278→  const varLines = Object.entries(state.vars)
   279→    .filter(([k]) => !k.startsWith('__'))
   280→    .map(([k, v]) => `${k}: ${v}`);
   281→  if (varLines.length) {
   282→    sysParts.push(`Известные данные о пользователе:\n${varLines.join('\n')}`);
   283→  }
   284→
   285→  const messages: { role: 'assistant' | 'user'; content: string }[] = [
   286→    { role: 'assistant', content: sysParts.join('\n\n') },
   287→  ];
   288→
   289→  if (data.useMemory !== false) {
   290→    for (const h of state.history.slice(-MEMORY_TAIL)) {
   291→      messages.push({
   292→        role: h.role === 'user' ? 'user' : 'assistant',
   293→        content: h.text,
   294→      });
   295→    }
   296→  }
   297→  if (lastInput) {
   298→    messages.push({ role: 'user', content: lastInput });
   299→  } else if (messages.length === 1) {
   300→    messages.push({ role: 'user', content: 'Привет' });
   301→  }
   302→
   303→  // IMP-BE21-02: SDK не принимает signal — таймаут через Promise.race (45 с).
   304→  // На таймауте/ошибке выбрасываем исключение — существующие catch показывают
   305→  // fallback-текст (AI-узел сценария, свободный чат, assistantFallback).
   306→  let aiTimer: ReturnType<typeof setTimeout> | undefined;
   307→  let completion: Awaited<ReturnType<typeof zai.chat.completions.create>>;
   308→  try {
   309→    completion = await Promise.race([
   310→      zai.chat.completions.create({
   311→        messages,
   312→        thinking: { type: 'disabled' },
   313→      }),
   314→      new Promise<never>((_, reject) => {
   315→        aiTimer = setTimeout(() => reject(new Error('AI timeout: ответ ИИ не получен за 45 с')), 45_000);
   316→        // таймер не должен удерживать процесс при завершении
   317→        if (typeof aiTimer.unref === 'function') aiTimer.unref();
   318→      }),
   319→    ]);
   320→  } finally {
   321→    // очищаем таймер, чтобы не течь (иначе гонка «ответ пришёл последним» держит handle)
   322→    if (aiTimer) clearTimeout(aiTimer);
   323→  }
   324→  const content = completion.choices[0]?.message?.content;
   325→  return content && content.trim().length > 0
   326→    ? content
   327→    : 'Извините, я не смог сформулировать ответ. Попробуйте перефразировать вопрос.';
   328→}

   330→/** Ответ ИИ-ассистента на свободный вопрос (используется и режимом «Свободный чат») */
   331→export async function askAssistant(
   332→  assistant: AiAssistantConfig,
   333→  state: EngineState,
   334→  userText: string
   335→): Promise<string | null> {
   336→  try {
   337→    const raw = await runAI(
   338→      { prompt: assistant.prompt, knowledge: assistant.knowledge, useMemory: true },
   339→      state,
   340→      userText,
   341→      ASSISTANT_SYSTEM_RULES
   342→    );
   343→    return raw && raw.trim().length > 0 ? raw : null;
   344→  } catch (err) {
   345→    console.error('[flow-engine] askAssistant error:', err);
   346→    return null;
   347→  }
   348→}

   350→// ─── Оркестратор: главный цикл выполнения сценария ───────────────────────────
   351→
   352→export async function runEngine(
   353→  flow: Flow,
   354→  input: string | null,
   355→  prevState: EngineState | null,
   356→  assistant?: AiAssistantConfig,
   357→  ctx?: EngineContext
   358→): Promise<EngineResult> {
   359→  const state: EngineState = prevState
   360→    ? {
   361→        currentNodeId: prevState.currentNodeId ?? null,
   362→        waiting: prevState.waiting ?? 'none',
   363→        vars: { ...prevState.vars },
   364→        history: [...(prevState.history ?? [])].slice(-MEMORY_TAIL * 2),
   365→      }
   366→    : emptyState();
   367→
   368→  const messages: EngineMessage[] = [];
   369→  let needsOperator = state.vars['__operator'] === 'true';
   370→  let lastInput = input ?? '';
   371→  let current: FlowNode | null = null;
   372→
   373→  const assistantEnabled = assistant?.enabled === true;
   374→
   375→  /**
   376→   * Ответ ИИ-ассистента на свободный вопрос (вне сценария).
   377→   * После ответа возвращает пользователя в меню, чтобы навигация
   378→   * по сценарию продолжала работать.
   379→   */
   380→  const assistantFallback = async (userText: string, menuNode?: FlowNode): Promise<void> => {
   381→    let answer: string | null = null;
   382→    try {
   383→      const raw = await runAI(
   384→        { prompt: assistant?.prompt, knowledge: assistant?.knowledge, useMemory: true },
   385→        state,
   386→        userText,
   387→        ASSISTANT_SYSTEM_RULES
   388→      );
   389→      answer = raw && raw.trim().length > 0 ? raw : null;
   390→    } catch (err) {
   391→      console.error('[flow-engine] AI assistant error:', err);
   392→    }
   393→
   394→    // ИИ недоступен — вежливо сообщаем, пользователь может повторить вопрос
   395→    if (!answer) {
   396→      const errText =
   397→        'Извините, сервис ИИ временно недоступен. Попробуйте задать вопрос ещё раз через минуту.';
   398→      messages.push({ text: errText, nodeId: 'assistant' });
   399→      state.history.push({ role: 'bot', text: errText });
   400→      state.waiting = 'none';
   401→      state.currentNodeId = null;
   402→      return;
   403→    }
   404→
   405→    // Пользователь просит живого оператора — передаём диалог
   406→    if (answer.toUpperCase().includes('OPERATOR_REQUEST')) {
   407→      state.vars['__operator'] = 'true';
   408→      needsOperator = true;
   409→      const text = 'Соединяю вас с живым оператором, оставайтесь на линии 🙌';
   410→      messages.push({ text, nodeId: 'assistant' });
   411→      state.history.push({ role: 'bot', text });
   412→      state.waiting = 'none';
   413→      state.currentNodeId = null;
   414→      return;
   415→    }
   416→
   417→    messages.push({ text: answer, nodeId: 'assistant' });
   418→    state.history.push({ role: 'bot', text: answer });
   419→
   420→    // Возвращаем меню, чтобы можно было продолжить навигацию по сценарию
   421→    const menu = menuNode ?? findMenuNode(flow, state.vars['__last_menu']);
   422→    if (menu) {
   423→      if (assistant?.reaskMenu !== false) {
   424→        const menuText = interpolate(menu.data.text, state.vars) || 'Выберите вариант:';
   425→        messages.push({
   426→          text: menuText,
   427→          nodeId: menu.id,
   428→          buttons: menu.data.buttons ?? [],
   429→        });
   430→        state.history.push({ role: 'bot', text: menuText });
   431→      }
   432→      state.waiting = 'buttons';
   433→      state.currentNodeId = menu.id;
   434→    } else {
   435→      state.waiting = 'none';
   436→      state.currentNodeId = null;
   437→    }
   438→  };
   439→
   440→  // 1. Возобновление: если движок ждал ввода/кнопки
   441→  if (input && state.waiting !== 'none' && state.currentNodeId) {
   442→    const node = findNode(flow, state.currentNodeId);
   443→    if (node?.type === 'question') {
   444→      // Постоянная кнопка «🏠 Главное меню» — работает и вместо ответа на вопрос
   445→      if (isMainMenuCommand(input)) {
   446→        state.history.push({ role: 'user', text: input });
   447→        state.waiting = 'none';
   448→        lastInput = '';
   449→        current = findStart(flow);
   450→        if (current && current.type === 'start') {
   451→          current = nextNode(flow, current.id);
   452→        }
   453→      } else if (isSkipWord(input) && node.data.validate) {
   454→        // «пропустить» — принимаем ответ без проверки (пустая переменная)
   455→        const v = node.data.variable?.trim() || 'answer';
   456→        if (!state.vars[v]) state.vars[v] = '';
   457→        state.history.push({ role: 'user', text: input });
   458→        state.waiting = 'none';
   459→        lastInput = '';
   460→        current = nextNode(flow, node.id);
   461→      } else {
   462→        // Автопроверка ответа (телефон/адрес): при ошибке переспрашиваем
   463→        const err = await validateAnswer(node.data, input, state.vars);
   464→        if (err) {
   465→          state.history.push({ role: 'user', text: input });
   466→          state.history.push({ role: 'bot', text: err });
   467→          messages.push({ text: err, nodeId: node.id });
   468→          state.waiting = 'input';
   469→          state.currentNodeId = node.id;
   470→          return { messages, state, needsOperator };
   471→        }
   472→        const v = node.data.variable?.trim() || 'answer';
   473→        state.vars[v] = input;
   474→        state.history.push({ role: 'user', text: input });
   475→        state.waiting = 'none';
   476→        lastInput = '';
   477→        current = nextNode(flow, node.id);
   478→      }
   479→    } else if (node?.type === 'buttons') {
   480→      state.history.push({ role: 'user', text: input });
   481→      const btn = (node.data.buttons ?? []).find(
   482→        (b) => b.text.trim().toLowerCase() === input.trim().toLowerCase()
   483→      );
   484→      state.waiting = 'none';
   485→      lastInput = '';
   486→      if (btn) {
   487→        // Выбор из меню можно сохранить в переменную (напр. объём контейнера)
   488→        if (node.data.saveSelection) {
   489→          state.vars[node.data.saveSelection] = btn.text;
   490→        }
   491→        current = nextNode(flow, node.id, btn.id);
   492→      } else if (isMainMenuCommand(input) || isStartCommand(input)) {
   493→        // «🏠 Главное меню» или /start в любом месте диалога — начать заново
   494→        current = findStart(flow);
   495→        if (current && current.type === 'start') {
   496→          current = nextNode(flow, current.id);
   497→        }
   498→      } else if (assistantEnabled) {
   499→        // Свободный текст вместо кнопки — отвечает ИИ-ассистент,
   500→        // затем пользователю снова показывается это же меню
   501→        await assistantFallback(input, node);
   502→        return { messages, state, needsOperator };
   503→      } else {
   504→        // Нет совпадения — переспрашиваем
   505→        const text = interpolate(node.data.text, state.vars) || 'Выберите вариант:';
   506→        messages.push({ text, nodeId: node.id, buttons: node.data.buttons ?? [] });
   507→        state.waiting = 'buttons';
   508→        state.currentNodeId = node.id;
   509→        state.history.push({ role: 'bot', text });
   510→        return { messages, state, needsOperator };
   511→      }
   512→    } else {
   513→      state.waiting = 'none';
   514→      current = findStart(flow);
   515→    }
   516→  }
   517→
   518→  // 2. Старт нового диалога (или продолжение после ожидания)
   519→  if (!current) {
   520→    if (state.waiting !== 'none') {
   521→      // Ждём ввода, но его не пришло — ничего не делаем
   522→      return { messages, state, needsOperator };
   523→    }
   524→    if (input) {
   525→      state.history.push({ role: 'user', text: input });
   526→    }
   527→    // Диалог уже передан оператору — бот молчит, пишет только человек-оператор
   528→    if (needsOperator) {
   529→      return { messages, state, needsOperator };
   530→    }
   531→    // ИИ-ассистент: свободный вопрос пользователя (в т.ч. первое сообщение);
   532→    // /start всегда запускает сценарий с приветствия — детерминированный вход
   533→    if (assistantEnabled && input && !isStartCommand(input)) {
   534→      await assistantFallback(input);
   535→      return { messages, state, needsOperator };
   536→    }
   537→    current = findStart(flow);
   538→    if (current && current.type === 'start') {
   539→      current = nextNode(flow, current.id);
   540→    }
   541→  }
   542→
   543→  // 3. Главный цикл traversal'а
   544→  let steps = 0;
   545→  while (current && steps < MAX_STEPS) {
   546→    steps++;
   547→    const node = current;
   548→    switch (node.type) {
   549→      case 'start': {
   550→        current = nextNode(flow, node.id);
   551→        break;
   552→      }
   553→      case 'message': {
   554→        // Узел с флагом «Создать заявку»: номер попадает в {{order.number}}
   555→        if (node.data.createOrder && ctx?.botId) {
   556→          try {
   557→            const cfg = node.data.createOrder;
   558→            // Значение переменной; «{{city}}, {{street}}» — подстановка шаблона
   559→            const pick = (v?: string) => {
   560→              if (!v || !v.trim()) return '';
   561→              if (v.includes('{{')) return interpolate(v, state.vars).trim();
   562→              return (state.vars[v.trim()] ?? '').trim();
   563→            };
   564→            const order = await createOrder({
   565→              botId: ctx.botId,
   566→              conversationId: ctx.conversationId,
   567→              externalUserId: ctx.externalUserId ?? null,
   568→              type: cfg.type ?? 'waste',
   569→              clientName: pick(cfg.nameVar) || null,
   570→              phone: pick(cfg.phoneVar) || null,
   571→              city: pick(cfg.cityVar) || null,
   572→              address: pick(cfg.addressVar) || null,
   573→              size: pick(cfg.sizeVar) || null,
   574→              wishDate: pick(cfg.dateVar) || null,
   575→              comment: cfg.comment ? interpolate(cfg.comment, state.vars) : null,
   576→            });
   577→            state.vars['order.number'] = String(order.number);
   578→            state.vars['order.id'] = order.id;
   579→            state.vars['order.type'] = order.type;
   580→            console.log(`[flow-engine] заявка №${order.number} создана (${order.type})`);
   581→          } catch (err) {
   582→            console.error('[flow-engine] createOrder error:', err);
   583→            state.vars['order.number'] = '—';
   584→          }
   585→        }
   586→        const text = interpolate(node.data.text, state.vars, lastInput || undefined);
   587→        if (text.trim()) {
   588→          messages.push({ text, nodeId: node.id });
   589→          state.history.push({ role: 'bot', text });
   590→        }
   591→        lastInput = '';
   592→        current = nextNode(flow, node.id);
   593→        break;
   594→      }
   595→      case 'question': {
   596→        const text = interpolate(node.data.text, state.vars, lastInput || undefined);
   597→        if (text.trim()) {
   598→          messages.push({ text, nodeId: node.id });
   599→          state.history.push({ role: 'bot', text });
   600→        }
   601→        state.waiting = 'input';
   602→        state.currentNodeId = node.id;
   603→        current = null;
   604→        break;
   605→      }
   606→      case 'buttons': {
   607→        const text = interpolate(node.data.text, state.vars, lastInput || undefined);
   608→        if (text.trim()) {
   609→          messages.push({
   610→            text,
   611→            nodeId: node.id,
   612→            buttons: node.data.buttons ?? [],
   613→          });
   614→          state.history.push({ role: 'bot', text });
   615→        }
   616→        // Запоминаем последнее меню — ассистент вернёт к нему после ответа
   617→        state.vars['__last_menu'] = node.id;
   618→        state.waiting = 'buttons';
   619→        state.currentNodeId = node.id;
   620→        current = null;
   621→        break;
   622→      }
   623→      case 'condition': {
   624→        const ok = evalCondition(node.data.condition, state.vars);
   625→        current = nextNode(flow, node.id, ok ? 'yes' : 'no');
   626→        break;
   627→      }
   628→      case 'ai': {
   629→        try {
   630→          // AI-узел дополняется общей базой знаний ассистента бота
   631→          const d = { ...node.data };
   632→          if (assistantEnabled && d.useBotKnowledge !== false && assistant?.knowledge?.trim()) {
   633→            d.knowledge = d.knowledge?.trim()
   634→              ? `${d.knowledge.trim()}\n\n${assistant.knowledge.trim()}`
   635→              : assistant.knowledge;
   636→          }
   637→          const raw = await runAI(d, state, lastInput);
   638→          const text = interpolate(raw, state.vars);
   639→          messages.push({ text, nodeId: node.id });
   640→          state.history.push({ role: 'bot', text });
   641→        } catch (err) {
   642→          console.error('[flow-engine] AI node error:', err);
   643→          const fallback =
   644→            'Извините, сервис ИИ временно недоступен. Попробуйте ещё раз через минуту.';
   645→          messages.push({ text: fallback, nodeId: node.id });
   646→          state.history.push({ role: 'bot', text: fallback });
   647→        }
   648→        lastInput = '';
   649→        current = nextNode(flow, node.id);
   650→        break;
   651→      }
   652→      case 'http': {
   653→        const url = interpolate(node.data.url, state.vars, lastInput || undefined);
   654→        if (url.trim()) {
   655→          try {
   656→            const method = (node.data.method || 'GET').toUpperCase();
   657→            const init: RequestInit = { method };
   658→            if (method !== 'GET' && node.data.body) {
   659→              init.body = interpolate(node.data.body, state.vars, lastInput || undefined);
   660→              init.headers = { 'Content-Type': 'application/json' };
   661→            }
   662→            const res = await fetch(url, { ...init, signal: AbortSignal.timeout(10000) });
   663→            state.vars['__http_status'] = String(res.status);
   664→          } catch {
   665→            state.vars['__http_status'] = 'error';
   666→          }
   667→        }
   668→        lastInput = '';
   669→        current = nextNode(flow, node.id);
   670→        break;
   671→      }
   672→      case 'delay': {
   673→        const sec = Math.min(Math.max(node.data.seconds ?? 1, 0), 3);
   674→        if (sec > 0) await new Promise((r) => setTimeout(r, sec * 1000));
   675→        current = nextNode(flow, node.id);
   676→        break;
   677→      }
   678→      case 'handoff': {
   679→        state.vars['__operator'] = 'true';
   680→        needsOperator = true;
   681→        const text =
   682→          interpolate(node.data.text, state.vars, lastInput || undefined) ||
   683→          'Соединяю вас с живым оператором, оставайтесь на линии 🙌';
   684→        if (text.trim()) {
   685→          messages.push({ text, nodeId: node.id });
   686→          state.history.push({ role: 'bot', text });
   687→        }
   688→        lastInput = '';
   689→        current = nextNode(flow, node.id);
   690→        break;
   691→      }
   692→      case 'end':
   693→      default: {
   694→        state.waiting = 'none';
   695→        state.currentNodeId = null;
   696→        current = null;
   697→        break;
   698→      }
   699→    }
   700→  }
   701→
   702→  if (steps >= MAX_STEPS) {
   703→    state.waiting = 'none';
   704→    state.currentNodeId = null;
   705→  }
   706→
   707→  return { messages, state, needsOperator };
   708→}
