    64→function markerColor(order: OrderDto): string {

    67→    case 'waste':
    68→      return 'bg-emerald-500';
    69→    case 'kgm':
    70→      return 'bg-amber-500';
    71→    default:
    72→      return 'bg-violet-500';
    73→  }

    87→function OrdersMap({
    88→  orders,
    89→  showCompleted,
    90→  placementOrderId,
    91→  selectedId,
    92→  focusOrderId,
    93→  focusTick,
    94→  onSelect,
    95→  onPlace,
    96→}: {
    97→  orders: OrderDto[];
    98→  showCompleted: boolean;
    99→  placementOrderId: string | null;
   100→  selectedId: string | null;
   101→  focusOrderId: string | null;
   102→  focusTick: number;
   103→  onSelect: (id: string) => void;
   104→  onPlace: (lat: number, lng: number) => void;
   105→}) {
   106→  const containerRef = useRef<HTMLDivElement>(null);
   107→  const mapRef = useRef<MlMap | null>(null);
   108→  const mlRef = useRef<typeof import('maplibre-gl') | null>(null);
   109→  const markersRef = useRef<MlMarker[]>([]);
   110→  const [ready, setReady] = useState(false);
   111→  const [failed, setFailed] = useState(false);
   112→
   113→  // Инициализация карты (динамический импорт — компонент клиентский)
   114→  useEffect(() => {
   115→    let cancelled = false;
   116→    (async () => {
   117→      try {
   118→        const ml = await import('maplibre-gl');
   119→        if (cancelled || !containerRef.current) return;
   120→        // Воркер MapLibre не собирается Turbopack'ом — берём локальную копию из public/
   121→        ml.setWorkerUrl('/maplibre-gl-worker.mjs');
   122→        mlRef.current = ml;
   123→        const map = new ml.Map({
   124→          container: containerRef.current,
   125→          style: 'https://tiles.openfreemap.org/styles/liberty',
   126→          center: [37.6176, 55.7558], // Москва — стартовый вид
   127→          zoom: 8.5,
   128→        });
   129→        map.addControl(new ml.NavigationControl({ showCompass: false }), 'top-right');
   130→        map.on('load', () => {
   131→          if (!cancelled) setReady(true);
   132→        });
   133→        map.on('error', (e) => {
   134→          // Тайловая сеть недоступна — сообщаем, но карту не убиваем
   135→          console.warn('[orders map]', e?.error?.message ?? e);
   136→        });
   137→        mapRef.current = map;
   138→      } catch (err) {
   139→        console.error('[orders map] init failed', err);
   140→        if (!cancelled) setFailed(true);
   141→      }
   142→    })();
   143→    return () => {
   144→      cancelled = true;
   145→      mapRef.current?.remove();
   146→      mapRef.current = null;
   147→      setReady(false);
   148→    };
   149→  }, []);
   150→
   151→  // Маркеры заявок
   152→  const visible = orders.filter(
   153→    (o) => o.lat != null && o.lng != null && (showCompleted || ACTIVE_STATUSES.includes(o.status))
   154→  );
   155→
   156→  useEffect(() => {
   157→    const map = mapRef.current;
   158→    const ml = mlRef.current;
   159→    if (!map || !ml || !ready) return;
   160→    for (const m of markersRef.current) m.remove();
   161→    markersRef.current = [];
   162→    for (const o of visible) {
   163→      const el = document.createElement('button');
   164→      el.type = 'button';
   165→      el.title = `Заявка №${o.number} — ${ORDER_TYPE_LABELS[o.type] ?? ''}`;
   166→      // Без hover:scale — масштабирование сдвигает маркер и «прыгает»;
   167→      // подсвечиваем тенью и рамкой, размер элемента не меняется
   168→      el.className = cn(
   169→        'flex h-8 w-8 cursor-pointer items-center justify-center rounded-full border-2 text-[11px] font-bold text-white shadow-md transition-[box-shadow,filter,border-color] hover:shadow-xl hover:brightness-110',
   170→        markerColor(o),
   171→        o.id === selectedId ? 'border-primary ring-2 ring-primary/50' : 'border-white'
   172→      );
   173→      el.textContent = String(o.number);
   174→      el.addEventListener('click', (e) => {
   175→        e.stopPropagation();
   176→        onSelect(o.id);
   177→      });
   178→      const marker = new ml.Marker({ element: el })
   179→        .setLngLat([o.lng as number, o.lat as number])
   180→        .addTo(map);
   181→      markersRef.current.push(marker);
   182→    }
   183→    // зависимость — сериализованный список видимых маркеров
   184→  }, [ready, visible.map((o) => `${o.id}:${o.lat}:${o.lng}:${o.status}`).join('|'), selectedId]);
   185→
   186→  // Наведение на заявку из списка/карточки: перелетаем к точке (focusTick —
   187→  // чтобы повторный клик по той же заявке тоже срабатывал)
   188→  useEffect(() => {
   189→    const map = mapRef.current;
   190→    if (!map || !ready || !focusOrderId) return;
   191→    const o = visible.find((x) => x.id === focusOrderId);
   192→    if (!o || o.lat == null || o.lng == null) return;
   193→    map.flyTo({ center: [o.lng, o.lat], zoom: Math.max(map.getZoom(), 15), duration: 800 });
   194→  }, [ready, focusOrderId, focusTick]);
   195→
   196→  // Режим указания точки: клик по карте → координаты заявки
   197→  const handleMapClick = useCallback(
   198→    (lat: number, lng: number) => onPlace(lat, lng),
   199→    [onPlace]
   200→  );
   201→  useEffect(() => {
   202→    const map = mapRef.current;
   203→    if (!map || !ready || !placementOrderId) return;
   204→    const onClick = (e: { lngLat: { lat: number; lng: number } }) =>
   205→      handleMapClick(e.lngLat.lat, e.lngLat.lng);
   206→    map.on('click', onClick);
   207→    map.getCanvas().style.cursor = 'crosshair';
   208→    return () => {
   209→      map.off('click', onClick);
   210→      map.getCanvas().style.cursor = '';
   211→    };
   212→  }, [ready, placementOrderId, handleMapClick]);
   213→
   214→  if (failed) {
   215→    return (
   216→      <div className="flex h-full min-h-[420px] flex-col items-center justify-center gap-3 rounded-xl border bg-muted/30 text-center text-muted-foreground">
   217→        <MapPin className="h-10 w-10 opacity-30" />
   218→        <p className="max-w-xs text-sm">
   219→          Не удалось загрузить карту OpenFreeMap. Проверьте доступ к tiles.openfreemap.org и
   220→          обновите страницу.
   221→        </p>
   222→      </div>
   223→    );
   224→  }
   225→
   226→  return (
   227→    <div className="relative h-full min-h-[380px] overflow-hidden rounded-xl border bg-muted/30">
   228→      <div ref={containerRef} className="h-full w-full" aria-label="Карта заявок" />
   229→      {!ready && (
   230→        <div className="absolute inset-0 flex items-center justify-center gap-2 text-sm text-muted-foreground">
   231→          <Loader2 className="h-4 w-4 animate-spin" /> Загружаю карту OpenFreeMap…
   232→        </div>
   233→      )}
   234→      {/* Легенда (компактная, не перекрывает атрибуцию и контролы) */}
   235→      <div className="absolute bottom-2.5 left-2.5 z-10 flex flex-wrap items-center gap-x-2.5 gap-y-0.5 rounded-lg border bg-background/95 px-2.5 py-1.5 text-[10px] leading-tight shadow-sm backdrop-blur sm:bottom-3 sm:left-3 sm:text-[11px]">
   236→        <span className="flex items-center gap-1.5">
   237→          <span className="h-2.5 w-2.5 rounded-full bg-emerald-500" /> отходы
   238→        </span>
   239→        <span className="flex items-center gap-1.5">
   240→          <span className="h-2.5 w-2.5 rounded-full bg-amber-500" /> КГМ
   241→        </span>
   242→        <span className="flex items-center gap-1.5">
   243→          <span className="h-2.5 w-2.5 rounded-full bg-slate-400" /> выполненные
   244→        </span>
   245→      </div>
   246→      {/* Счётчик маркеров */}
   247→      <div className="absolute left-2.5 top-2.5 z-10 rounded-lg border bg-background/95 px-2.5 py-1.5 text-xs shadow-sm backdrop-blur sm:left-3 sm:top-3">
   248→        <MapPin className="mr-1 inline h-3 w-3 text-primary" />
   249→        На карте: <b>{visible.length}</b>
   250→      </div>
   251→      {placementOrderId && (
   252→        <div className="absolute inset-x-2.5 top-12 z-20 flex items-center justify-between gap-2 rounded-lg border border-primary/40 bg-primary px-3 py-2 text-xs font-medium text-primary-foreground shadow-md sm:inset-x-auto sm:right-auto sm:left-3 sm:top-12">
   253→          <span className="flex items-center gap-1.5">
   254→            <Crosshair className="h-3.5 w-3.5 shrink-0" /> Кликните по карте — точка заявки №
   255→            {placementOrderId}
   256→          </span>
   257→        </div>
   258→      )}
   259→    </div>
   260→  );
   261→}

   265→function OrderChat({
   266→  botId,
   267→  orderId,
   268→  hasConversation,
   269→}: {
   270→  botId: string;
   271→  orderId: string;
   272→  hasConversation: boolean;
   273→}) {
   274→  const { toast } = useToast();
   275→  const [messages, setMessages] = useState<OrderMessageDto[]>([]);
   276→  const [text, setText] = useState('');
   277→  const [sending, setSending] = useState(false);
   278→  const scrollRef = useRef<HTMLDivElement>(null);
   279→
   280→  const load = useCallback(async () => {
   281→    try {
   282→      const d = await api<{ messages: OrderMessageDto[] }>(
   283→        `/api/bots/${botId}/orders/${orderId}/messages`
   284→      );
   285→      setMessages(d.messages);
   286→    } catch {
   287→      /* polling errors ignored */
   288→    }
   289→  }, [botId, orderId]);
   290→
   291→  useEffect(() => {
   292→    setMessages([]);
   293→    load();
   294→    const t = setInterval(load, 3000);
   295→    return () => clearInterval(t);
   296→  }, [load]);
   297→
   298→  useEffect(() => {
   299→    requestAnimationFrame(() => {
   300→      scrollRef.current?.scrollTo({ top: scrollRef.current.scrollHeight });
   301→    });
   302→  }, [messages.length]);
   303→
   304→  const send = async () => {
   305→    const t = text.trim();
   306→    if (!t) return;
   307→    setSending(true);
   308→    setText('');
   309→    try {
   310→      const d = await api<{ message: OrderMessageDto; delivered: boolean; deliveryError?: string }>(
   311→        `/api/bots/${botId}/orders/${orderId}/messages`,
   312→        { method: 'POST', body: JSON.stringify({ text: t }) }
   313→      );
   314→      setMessages((prev) => [...prev, d.message]);
   315→      if (!d.delivered && d.deliveryError) {
   316→        toast({ title: 'Не доставлено в мессенджер', description: d.deliveryError, variant: 'destructive' });
   317→      }
   318→    } catch {
   319→      toast({ title: 'Не удалось отправить', variant: 'destructive' });
   320→    } finally {
   321→      setSending(false);
   322→    }
   323→  };
   324→
   325→  return (
   326→    <div className="flex min-h-0 flex-1 flex-col">
   327→      <div className="flex items-center gap-2 border-b px-3 py-2 text-xs text-muted-foreground">
   328→        <MessageSquare className="h-3.5 w-3.5" />
   329→        Чат с клиентом по заявке — сообщения доставляются в его мессенджер
   330→      </div>
   331→      <div ref={scrollRef} className="min-h-0 flex-1 space-y-2 overflow-y-auto bg-muted/30 p-3">
   332→        {messages.length === 0 && (
   333→          <div className="flex h-full items-center justify-center text-center text-xs text-muted-foreground">
   334→            Сообщений по заявке пока нет.
   335→            <br />
   336→            Клиент может написать в боте, открыв карточку своей заявки.
   337→          </div>
   338→        )}
   339→        {messages.map((m) => {
   340→          const fromClient = m.role === 'user';
   341→          return (
   342→            <div key={m.id} className={cn('flex', fromClient ? 'justify-start' : 'justify-end')}>
   343→              <div
   344→                className={cn(
   345→                  'max-w-[85%] space-y-0.5 rounded-2xl px-3 py-1.5 text-sm leading-snug',
   346→                  fromClient
   347→                    ? 'rounded-bl-md border bg-card'
   348→                    : 'rounded-br-md bg-primary text-primary-foreground'
   349→                )}
   350→              >
   351→                {m.nodeId === '__operator' && (
   352→                  <div className="flex items-center gap-1 text-[10px] opacity-70">
   353→                    <Headset className="h-3 w-3" /> оператор
   354→                  </div>
   355→                )}
   356→                <div className="whitespace-pre-wrap">{m.text}</div>
   357→                <div className="text-right text-[9px] opacity-60">
   358→                  {new Date(m.createdAt).toLocaleTimeString('ru-RU', {
   359→                    hour: '2-digit',
   360→                    minute: '2-digit',
   361→                  })}
   362→                </div>
   363→              </div>
   364→            </div>
   365→          );
   366→        })}
   367→      </div>
   368→      <form
   369→        className="flex items-center gap-2 border-t p-2.5 pb-[max(0.625rem,env(safe-area-inset-bottom))]"
   370→        onSubmit={(e) => {
   371→          e.preventDefault();
   372→          send();
   373→        }}
   374→      >
   375→        <Input
   376→          placeholder={
   377→            hasConversation
   378→              ? 'Ответить клиенту…'
   379→              : 'Диалог с клиентом не привязан — чат только для чтения'
   380→          }
   381→          value={text}
   382→          disabled={!hasConversation}
   383→          onChange={(e) => setText(e.target.value)}
   384→        />
   385→        <Button
   386→          type="submit"
   387→          size="icon"
   388→          className="h-10 w-10 shrink-0"
   389→          disabled={sending || !hasConversation || !text.trim()}
   390→          aria-label="Отправить"
   391→        >
   392→          {sending ? <Loader2 className="h-4 w-4 animate-spin" /> : <Send className="h-4 w-4" />}
   393→        </Button>
   394→      </form>
   395→    </div>
   396→  );
   397→}

   401→function OrderDetailDialog({
   402→  botId,
   403→  orderId,
   404→  open,
   405→  onClose,
   406→  onSwitchOrder,
   407→  onPlaceRequest,
   408→  onShowOnMap,
   409→  onDataChanged,
   410→}: {
   411→  botId: string;
   412→  orderId: string | null;
   413→  open: boolean;
   414→  onClose: () => void;
   415→  onSwitchOrder: (id: string) => void;
   416→  onPlaceRequest: (orderId: string) => void;
   417→  onShowOnMap: (orderId: string) => void;
   418→  onDataChanged: () => void;
   419→}) {
   420→  const { toast } = useToast();
   421→  const [order, setOrder] = useState<OrderDto | null>(null);
   422→  const [clientOrders, setClientOrders] = useState<OrderDto[]>([]);
   423→  const [loading, setLoading] = useState(false);
   424→  const [assignee, setAssignee] = useState('');
   425→  const [address, setAddress] = useState('');
   426→  const [wishDate, setWishDate] = useState('');
   427→  const [comment, setComment] = useState('');
   428→  const [saving, setSaving] = useState(false);
   429→  const [geoLoading, setGeoLoading] = useState(false);
   430→  const [confirmGeo, setConfirmGeo] = useState(false);
   431→  const [confirmDelete, setConfirmDelete] = useState(false);
   432→
   433→  const load = useCallback(async () => {
   434→    if (!orderId) return;
   435→    setLoading(true);
   436→    try {
   437→      const d = await api<{ order: OrderDto; clientOrders: OrderDto[] }>(
   438→        `/api/bots/${botId}/orders/${orderId}`
   439→      );
   440→      setOrder(d.order);
   441→      setClientOrders(d.clientOrders);
   442→      setAssignee(d.order.assignee ?? '');
   443→      setAddress(d.order.address ?? '');
   444→      setWishDate(d.order.wishDate ?? '');
   445→      setComment(d.order.comment ?? '');
   446→    } catch {
   447→      toast({ title: 'Не удалось загрузить заявку', variant: 'destructive' });
   448→    } finally {
   449→      setLoading(false);
   450→    }
   451→  }, [botId, orderId, toast]);
   452→
   453→  useEffect(() => {
   454→    if (open && orderId) load();
   455→    if (!open) {
   456→      setOrder(null);
   457→      setClientOrders([]);
   458→      setConfirmDelete(false);
   459→      setConfirmGeo(false);
   460→    }
   461→  }, [open, orderId, load]);
   462→
   463→  const patch = async (data: Record<string, unknown>, successMsg?: string) => {
   464→    if (!orderId) return;
   465→    setSaving(true);
   466→    try {
   467→      const d = await api<{ order: OrderDto }>(`/api/bots/${botId}/orders/${orderId}`, {
   468→        method: 'PATCH',
   469→        body: JSON.stringify(data),
   470→      });
   471→      setOrder(d.order);
   472→      onDataChanged();
   473→      if (successMsg) toast({ title: successMsg });
   474→    } catch {
   475→      toast({ title: 'Ошибка сохранения', variant: 'destructive' });
   476→    } finally {
   477→      setSaving(false);
   478→    }
   479→  };
   480→
   481→  /**
   482→   * Геокодирование адреса. Ручная точка защищена: если координаты ставили
   483→   * вручную — сначала подтверждение; если адрес не нашёлся — прежние
   484→   * координаты остаются (не «слетают»).
   485→   */
   486→  const runGeocode = async () => {
   487→    if (!orderId || !order) return;
   488→    setConfirmGeo(false);
   489→    setGeoLoading(true);
   490→    try {
   491→      const d = await api<{ order: OrderDto; geo?: { ok: boolean; lat?: number; lng?: number } }>(
   492→        `/api/bots/${botId}/orders/${orderId}`,
   493→        { method: 'PATCH', body: JSON.stringify({ geocode: true }) }
   494→      );
   495→      setOrder(d.order);
   496→      onDataChanged();
   497→      if (d.geo?.ok) {
   498→        toast({
   499→          title: `Координаты обновлены: ${d.geo.lat?.toFixed(5)}, ${d.geo.lng?.toFixed(5)}`,
   500→        });
   501→      } else if (d.order.lat != null) {
   502→        toast({
   503→          title: 'Адрес не найден на карте',
   504→          description: 'Прежние координаты сохранены без изменений',
   505→        });
   506→      } else {
   507→        toast({
   508→          title: 'Адрес не найден на карте',
   509→          description: 'Уточните адрес или укажите точку кнопкой «Указать на карте»',
   510→          variant: 'destructive',
   511→        });
   512→      }
   513→    } catch {

  1135→export default function OrdersView({

  1139→  bot: { id: string; name: string };
  1140→  onBack: () => void;

  1144→  const [loading, setLoading] = useState(true);
  1145→  const [tab, setTab] = useState<Tab>('map');
  1146→  const [selectedId, setSelectedId] = useState<string | null>(null);
  1147→  const [dialogOpen, setDialogOpen] = useState(false);
  1148→  const [placementId, setPlacementId] = useState<string | null>(null);
  1149→  const [focus, setFocus] = useState<{ id: string; tick: number } | null>(null);
  1150→  const [showCompleted, setShowCompleted] = useState(true);
