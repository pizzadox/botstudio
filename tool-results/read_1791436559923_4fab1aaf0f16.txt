   492→    { id: string; role: string; text: string; buttons?: { id: string; text: string }[] }[]
   493→  >([]);
   494→  const [input, setInput] = useState('');
   495→  const [sending, setSending] = useState(false);
   496→  const convRef = useRef<string | null>(null);
   497→  const visitorRef = useRef<string>('');
   498→  const scrollRef = useRef<HTMLDivElement>(null);
   499→  const storageKey = `botstudio_demo_conv_${secret}`;
   500→  const visitorKey = 'botstudio_visitor';
   501→
   502→  // Стабильный id посетителя: один гость = одно обращение,
   503→  // даже если conversationId был потерян
   504→  useEffect(() => {
   505→    try {
   506→      let v = window.localStorage.getItem(visitorKey);
   507→      if (!v) {
   508→        v =
   509→          typeof crypto !== 'undefined' && 'randomUUID' in crypto
   510→            ? crypto.randomUUID().replace(/-/g, '')
   511→            : Math.random().toString(36).slice(2) + Date.now().toString(36);
   512→        window.localStorage.setItem(visitorKey, v);
   513→      }
   514→      visitorRef.current = v;
   515→    } catch {
   516→      visitorRef.current = Math.random().toString(36).slice(2) + Date.now().toString(36);
   517→    }
   518→  }, [visitorKey]);
   519→
   520→  // Восстановить диалог посетителя (как реальный виджет сайта)
   521→  useEffect(() => {
   522→    try {
   523→      const saved = window.localStorage.getItem(storageKey);
   524→      if (saved) {
   525→        convRef.current = saved;
   526→        fetch(`/api/webhook/demo/${secret}?conversationId=${saved}`)
   527→          .then((r) => r.json())
   528→          .then((d) => {
   529→            if (d.messages?.length) setMessages(d.messages);
   530→          })
   531→          .catch(() => {});
   532→      }
   533→    } catch {
   534→      /* ignore */
   535→    }
   536→  }, [secret, storageKey]);
   537→
   538→  const scrollToBottom = () => {
   539→    requestAnimationFrame(() => {
   540→      scrollRef.current?.scrollTo({ top: scrollRef.current.scrollHeight });
   541→    });
   542→  };
   543→
   544→  useEffect(() => {
   545→    scrollToBottom();
   546→  }, [messages]);
   547→
   548→  const poll = useCallback(async () => {
   549→    if (!convRef.current) return;
   550→    try {
   551→      const d = await api<{
   552→        messages: { id: string; role: string; text: string; buttons?: { id: string; text: string }[] }[];
   553→      }>(`/api/webhook/demo/${secret}?conversationId=${convRef.current}`);
   554→      setMessages((prev) => (d.messages.length > prev.length ? d.messages : prev));
   555→    } catch {
   556→      /* ignore */
   557→    }
   558→  }, [secret]);
   559→
   560→  useEffect(() => {
   561→    const t = setInterval(poll, 4000);
   562→    return () => clearInterval(t);
   563→  }, [poll]);
   564→
   565→  const send = async (text: string) => {
   566→    const t = text.trim();
   567→    if (!t || sending) return;
   568→    setSending(true);
   569→    setInput('');
   570→    setMessages((prev) => [...prev, { id: `local-${Date.now()}`, role: 'user', text: t }]);
   571→    scrollToBottom();
   572→    try {
   573→      const d = await api<{
   574→        conversationId: string;
   575→        messages: { id: string; role: string; text: string; buttons?: { id: string; text: string }[] }[];
   576→      }>(`/api/webhook/demo/${secret}`, {
   577→        method: 'POST',
   578→        body: JSON.stringify({ text: t, conversationId: convRef.current, visitorId: visitorRef.current || undefined }),
   579→      });
   580→      convRef.current = d.conversationId;
   581→      try {
   582→        window.localStorage.setItem(storageKey, d.conversationId);
   583→      } catch {
   584→        /* ignore */
   585→      }
   586→      setMessages(d.messages);
   587→      scrollToBottom();
   588→    } catch (e) {
   589→      setMessages((prev) => [
   590→        ...prev,
   591→        { id: `err-${Date.now()}`, role: 'bot', text: e instanceof Error ? e.message : 'Ошибка' },
   592→      ]);
   593→    } finally {
   594→      setSending(false);
   595→    }
   596→  };
   597→
   598→  return (
   599→    <div className="flex flex-col">
   600→      <div className="flex items-center gap-3 rounded-t-xl bg-gradient-to-r from-emerald-600 to-teal-600 p-4 text-white">
   601→        <div className="flex h-10 w-10 items-center justify-center rounded-full bg-white/20">
   602→          <Bot className="h-5 w-5" />
   603→        </div>
   604→        <div className="flex-1">
   605→          <div className="text-sm font-semibold leading-tight">Чат поддержки</div>
   606→          <div className="flex items-center gap-1 text-xs text-white/80">
   607→            <span className="h-1.5 w-1.5 rounded-full bg-emerald-300" /> онлайн
   608→          </div>
   609→        </div>
   610→        <Badge variant="secondary" className="text-[10px]">
   611→          демо сайта
   612→        </Badge>
   613→      </div>
   614→      <div ref={scrollRef} className="h-80 space-y-2.5 overflow-y-auto bg-muted/40 p-3">
   615→        {messages.length === 0 && (
   616→          <div className="flex h-full items-center justify-center text-center text-sm text-muted-foreground">
   617→            Напишите сообщение — бот ответит
   618→            <br />
   619→            как в настоящем виджете на сайте
   620→          </div>
   621→        )}
   622→        {messages.map((m) => (
   623→          <div key={m.id} className={cn('flex', m.role === 'user' ? 'justify-end' : 'justify-start')}>
   624→            <div
   625→              className={cn(
   626→                'max-w-[85%] whitespace-pre-wrap rounded-2xl px-3 py-2 text-sm leading-snug',
   627→                m.role === 'user'
   628→                  ? 'rounded-br-md bg-primary text-primary-foreground'
   629→                  : 'rounded-bl-md border bg-card'
   630→              )}
   631→            >
   632→              {m.text}
   633→            </div>
   634→          </div>
   635→        ))}
   636→        {(() => {
   637→          const lastButtons = [...messages].reverse().find((m) => m.role === 'bot' && m.buttons?.length)?.buttons;
   638→          if (!lastButtons?.length) return null;
   639→          return (
   640→            <div className="flex flex-wrap gap-2 pt-1">
   641→              {lastButtons.map((b) => (
   642→                <button
   643→                  key={b.id}
   644→                  onClick={() => send(b.text)}
   645→                  disabled={sending}
   646→                  className="rounded-full border border-primary/40 bg-primary/5 px-3 py-1.5 text-xs font-medium text-primary transition-colors hover:bg-primary hover:text-primary-foreground disabled:opacity-50"
   647→                >
   648→                  {b.text}
   649→                </button>
   650→              ))}
   651→            </div>
   652→          );
   653→        })()}
   654→      </div>
   655→      <form
   656→        className="flex items-center gap-2 border-t p-3"
   657→        onSubmit={(e) => {
   658→          e.preventDefault();
   659→          send(input);
   660→        }}
   661→      >
   662→        <Input placeholder="Сообщение…" value={input} onChange={(e) => setInput(e.target.value)} />
   663→        <Button type="submit" size="icon" className="h-10 w-10 shrink-0" disabled={sending || !input.trim()}>
   664→          {sending ? <Loader2 className="h-4 w-4 animate-spin" /> : <Send className="h-4 w-4" />}
   665→        </Button>
   666→      </form>
   667→    </div>
   668→  );
   669→}
