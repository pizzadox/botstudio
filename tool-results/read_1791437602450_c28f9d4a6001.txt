   105→   105→function timeAgo(iso: string): string {
   106→   106→  const diff = Date.now() - new Date(iso).getTime();
   107→   107→  const m = Math.floor(diff / 60000);
   108→   108→  if (m < 1) return 'только что';
   109→   109→  if (m < 60) return `${m} мин назад`;
   110→   110→  const h = Math.floor(m / 60);
   111→   111→  if (h < 24) return `${h} ч назад`;
   112→   112→  return `${Math.floor(h / 24)} дн назад`;
   113→   113→}

   163→   163→function ConversationOrdersPanel({
   164→   164→  conversationId,

   168→   168→  conversationId: string;

   174→   174→  const [orders, setOrders] = useState<ConvOrder[]>([]);
   175→   175→  const [mytkoEnabled, setMytkoEnabled] = useState(false);
   176→   176→  const [loaded, setLoaded] = useState(false);
   177→   177→  const [savingId, setSavingId] = useState<string | null>(null);
   178→   178→
   179→   179→  const load = useCallback(async () => {
   180→   180→    try {
   181→   181→      const d = await api<{ orders: ConvOrder[]; mytkoEnabled?: boolean }>(
   182→   182→        `/api/conversations/${conversationId}/orders`

   189→   189→      setLoaded(true);

   193→   193→  useEffect(() => {

   197→   197→  const patch = async (orderId: string, data: Record<string, unknown>) => {

   201→   201→        `/api/bots/${botId}/orders/${orderId}`,
   202→   202→        { method: 'PATCH', body: JSON.stringify(data) }
   203→   203→      );
   204→   204→      setOrders((prev) =>
   205→   205→        prev.map((o) =>
   206→   206→          o.id === orderId ? { ...o, status: d.order.status, assignee: d.order.assignee } : o
   207→   207→        )
   208→   208→      );
   209→   209→      toast({ title: 'Заявка обновлена' });
   210→   210→    } catch {
   211→   211→      toast({ title: 'Не удалось обновить заявку', variant: 'destructive' });
   212→   212→    } finally {
   213→   213→      setSavingId(null);
   214→   214→    }
   215→   215→  };
   216→   216→
   217→   217→  return (
   218→   218→    <div className="border-b bg-background px-4 py-2">
   219→   219→      <button
   220→   220→        className="flex w-full items-center gap-1.5 rounded text-left text-xs font-medium text-muted-foreground transition-colors hover:text-foreground focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring focus-visible:ring-offset-2"

   223→   223→        aria-controls="conversation-orders-list"

   225→   225→        {open ? <ChevronDown className="h-3.5 w-3.5" /> : <ChevronRight className="h-3.5 w-3.5" />}
   226→   226→        <Truck className="h-3.5 w-3.5" />

   228→   228→      </button>
   229→   229→      <div id="conversation-orders-list">
   230→   230→        {open && loaded && orders.length > 0 && (
   231→   231→          <div className="mt-2 max-h-56 space-y-1.5 overflow-y-auto pr-1">
   232→   232→            {orders.map((o) => (
   233→   233→              <div key={o.id} className="rounded-lg border p-2">
   234→   234→                <div className="flex flex-wrap items-center gap-x-2 gap-y-1">
   235→   235→                  <span className="text-xs font-semibold tabular-nums">№{o.number}</span>
   236→   236→                  <Badge
   237→   237→                    variant="outline"
   238→   238→                    className={cn('h-4 px-1 text-[9px]', ORDER_STATUS_BADGES[o.status])}

   243→   243→                  <span className="min-w-0 flex-1 truncate text-xs text-muted-foreground">
   244→   244→                    {ORDER_TYPE_LABELS[o.type] ?? o.type}
   245→   245→                    {o.address ? ` · ${o.address}` : ''}
   246→   246→                  </span>
   247→   247→                </div>
   248→   248→                <div className="mt-1.5 flex flex-wrap items-center gap-1.5">
   249→   249→                  <Select
   250→   250→                    value={o.status}

   262→   262→                      ))}
   263→   263→                    </SelectContent>
   264→   264→                  </Select>

   267→   267→                    defaultValue={o.assignee ?? ''}
   268→   268→                    placeholder="Исполнитель…"
   269→   269→                    aria-label="Исполнитель"
   270→   270→                    onBlur={(e) => {
   271→   271→                      if (e.target.value !== (o.assignee ?? '')) patch(o.id, { assignee: e.target.value });
   272→   272→                    }}
   273→   273→                  />
   274→   274→                  {o.wishDate && (
   275→   275→                    <span className="text-[11px] text-muted-foreground">🗓 {o.wishDate}</span>
   276→   276→                  )}
   277→   277→                </div>
   278→   278→              </div>

   280→   280→          </div>

   282→   282→        {open && loaded && orders.length === 0 && (
   283→   283→          <p className="mt-1.5 text-[11px] text-muted-foreground">У клиента пока нет заявок.</p>

   289→   289→          </div>
   290→   290→        )}
   291→   291→      </div>
