    87→    87→function OrdersMap({
    88→    88→  orders,
    89→    89→  showCompleted,
    90→    90→  placementOrderId,
    91→    91→  selectedId,
    92→    92→  focusOrderId,
    93→    93→  focusTick,
    94→    94→  onSelect,
    95→    95→  onPlace,
    96→    96→}: {
    97→    97→  orders: OrderDto[];
    98→    98→  showCompleted: boolean;
    99→    99→  placementOrderId: string | null;
   100→   100→  selectedId: string | null;
   101→   101→  focusOrderId: string | null;
   102→   102→  focusTick: number;
   103→   103→  onSelect: (id: string) => void;
   104→   104→  onPlace: (lat: number, lng: number) => void;
   105→   105→}) {
   106→   106→  const containerRef = useRef<HTMLDivElement>(null);
   107→   107→  const mapRef = useRef<MlMap | null>(null);
   108→   108→  const mlRef = useRef<typeof import('maplibre-gl') | null>(null);
   109→   109→  const markersRef = useRef<MlMarker[]>([]);
   110→   110→  const [ready, setReady] = useState(false);
   111→   111→  const [failed, setFailed] = useState(false);
   112→   112→
   113→   113→  // Инициализация карты (динамический импорт — компонент клиентский)
   114→   114→  useEffect(() => {
   115→   115→    let cancelled = false;
   116→   116→    (async () => {
   117→   117→      try {
   118→   118→        const ml = await import('maplibre-gl');
   119→   119→        if (cancelled || !containerRef.current) return;
   120→   120→        // Воркер MapLibre не собирается Turbopack'ом — берём локальную копию из public/
   121→   121→        ml.setWorkerUrl('/maplibre-gl-worker.mjs');
   122→   122→        mlRef.current = ml;
   123→   123→        const map = new ml.Map({
   124→   124→          container: containerRef.current,
   125→   125→          style: 'https://tiles.openfreemap.org/styles/liberty',
   126→   126→          center: [37.6176, 55.7558], // Москва — стартовый вид
   127→   127→          zoom: 8.5,
   128→   128→        });
   129→   129→        map.addControl(new ml.NavigationControl({ showCompass: false }), 'top-right');
   130→   130→        map.on('load', () => {
   131→   131→          if (!cancelled) setReady(true);
   132→   132→        });
   133→   133→        map.on('error', (e) => {
   134→   134→          // Тайловая сеть недоступна — сообщаем, но карту не убиваем
   135→   135→          console.warn('[orders map]', e?.error?.message ?? e);
   136→   136→        });
   137→   137→        mapRef.current = map;
   138→   138→      } catch (err) {
   139→   139→        console.error('[orders map] init failed', err);
   140→   140→        if (!cancelled) setFailed(true);
   141→   141→      }
   142→   142→    })();
   143→   143→    return () => {
   144→   144→      cancelled = true;
   145→   145→      mapRef.current?.remove();
   146→   146→      mapRef.current = null;
   147→   147→      setReady(false);
   148→   148→    };
   149→   149→  }, []);
   150→   150→
   151→   151→  // Маркеры заявок
   152→   152→  const visible = orders.filter(
   153→   153→    (o) => o.lat != null && o.lng != null && (showCompleted || ACTIVE_STATUSES.includes(o.status))
   154→   154→  );
   155→   155→
   156→   156→  useEffect(() => {
   157→   157→    const map = mapRef.current;
   158→   158→    const ml = mlRef.current;
   159→   159→    if (!map || !ml || !ready) return;
   160→   160→    for (const m of markersRef.current) m.remove();
   161→   161→    markersRef.current = [];
   162→   162→    for (const o of visible) {
   163→   163→      const el = document.createElement('button');
   164→   164→      el.type = 'button';
   165→   165→      el.title = `Заявка №${o.number} — ${ORDER_TYPE_LABELS[o.type] ?? ''}`;
   166→   166→      // Без hover:scale — масштабирование сдвигает маркер и «прыгает»;
   167→   167→      // подсвечиваем тенью и рамкой, размер элемента не меняется
   168→   168→      el.className = cn(
   169→   169→        'flex h-8 w-8 cursor-pointer items-center justify-center rounded-full border-2 text-[11px] font-bold text-white shadow-md transition-[box-shadow,filter,border-color] hover:shadow-xl hover:brightness-110',
   170→   170→        markerColor(o),
   171→   171→        o.id === selectedId ? 'border-primary ring-2 ring-primary/50' : 'border-white'
   172→   172→      );
   173→   173→      el.textContent = String(o.number);
   174→   174→      el.addEventListener('click', (e) => {
   175→   175→        e.stopPropagation();
   176→   176→        onSelect(o.id);
   177→   177→      });
   178→   178→      const marker = new ml.Marker({ element: el })
   179→   179→        .setLngLat([o.lng as number, o.lat as number])
   180→   180→        .addTo(map);
   181→   181→      markersRef.current.push(marker);
   182→   182→    }
   183→   183→    // зависимость — сериализованный список видимых маркеров
   184→   184→  }, [ready, visible.map((o) => `${o.id}:${o.lat}:${o.lng}:${o.status}`).join('|'), selectedId]);
   185→   185→
   186→   186→  // Наведение на заявку из списка/карточки: перелетаем к точке (focusTick —
   187→   187→  // чтобы повторный клик по той же заявке тоже срабатывал)
   188→   188→  useEffect(() => {
   189→   189→    const map = mapRef.current;
   190→   190→    if (!map || !ready || !focusOrderId) return;
   191→   191→    const o = visible.find((x) => x.id === focusOrderId);
   192→   192→    if (!o || o.lat == null || o.lng == null) return;
   193→   193→    map.flyTo({ center: [o.lng, o.lat], zoom: Math.max(map.getZoom(), 15), duration: 800 });
   194→   194→  }, [ready, focusOrderId, focusTick]);
   195→   195→
   196→   196→  // Режим указания точки: клик по карте → координаты заявки
   197→   197→  const handleMapClick = useCallback(
   198→   198→    (lat: number, lng: number) => onPlace(lat, lng),
   199→   199→    [onPlace]
   200→   200→  );
   201→   201→  useEffect(() => {
   202→   202→    const map = mapRef.current;
   203→   203→    if (!map || !ready || !placementOrderId) return;
   204→   204→    const onClick = (e: { lngLat: { lat: number; lng: number } }) =>
   205→   205→      handleMapClick(e.lngLat.lat, e.lngLat.lng);
   206→   206→    map.on('click', onClick);
   207→   207→    map.getCanvas().style.cursor = 'crosshair';
   208→   208→    return () => {
   209→   209→      map.off('click', onClick);
   210→   210→      map.getCanvas().style.cursor = '';
   211→   211→    };
   212→   212→  }, [ready, placementOrderId, handleMapClick]);
   213→   213→
   214→   214→  if (failed) {
   215→   215→    return (
   216→   216→      <div className="flex h-full min-h-[420px] flex-col items-center justify-center gap-3 rounded-xl border bg-muted/30 text-center text-muted-foreground">
   217→   217→        <MapPin className="h-10 w-10 opacity-30" />
   218→   218→        <p className="max-w-xs text-sm">
   219→   219→          Не удалось загрузить карту OpenFreeMap. Проверьте доступ к tiles.openfreemap.org и
   220→   220→          обновите страницу.
   221→   221→        </p>
   222→   222→      </div>
   223→   223→    );
   224→   224→  }
   225→   225→
   226→   226→  return (
   227→   227→    <div className="relative h-full min-h-[380px] overflow-hidden rounded-xl border bg-muted/30">
   228→   228→      <div ref={containerRef} className="h-full w-full" aria-label="Карта заявок" />
   229→   229→      {!ready && (
   230→   230→        <div className="absolute inset-0 flex items-center justify-center gap-2 text-sm text-muted-foreground">
   231→   231→          <Loader2 className="h-4 w-4 animate-spin" /> Загружаю карту OpenFreeMap…
   232→   232→        </div>
   233→   233→      )}
   234→   234→      {/* Легенда (компактная, не перекрывает атрибуцию и контролы) */}
   235→   235→      <div className="absolute bottom-2.5 left-2.5 z-10 flex flex-wrap items-center gap-x-2.5 gap-y-0.5 rounded-lg border bg-background/95 px-2.5 py-1.5 text-[10px] leading-tight shadow-sm backdrop-blur sm:bottom-3 sm:left-3 sm:text-[11px]">
   236→   236→        <span className="flex items-center gap-1.5">
   237→   237→          <span className="h-2.5 w-2.5 rounded-full bg-emerald-500" /> отходы
   238→   238→        </span>
   239→   239→        <span className="flex items-center gap-1.5">
   240→   240→          <span className="h-2.5 w-2.5 rounded-full bg-amber-500" /> КГМ
   241→   241→        </span>
   242→   242→        <span className="flex items-center gap-1.5">
   243→   243→          <span className="h-2.5 w-2.5 rounded-full bg-slate-400" /> выполненные
   244→   244→        </span>
   245→   245→      </div>
   246→   246→      {/* Счётчик маркеров */}
   247→   247→      <div className="absolute left-2.5 top-2.5 z-10 rounded-lg border bg-background/95 px-2.5 py-1.5 text-xs shadow-sm backdrop-blur sm:left-3 sm:top-3">
   248→   248→        <MapPin className="mr-1 inline h-3 w-3 text-primary" />
   249→   249→        На карте: <b>{visible.length}</b>
   250→   250→      </div>
   251→   251→      {placementOrderId && (
   252→   252→        <div className="absolute inset-x-2.5 top-12 z-20 flex items-center justify-between gap-2 rounded-lg border border-primary/40 bg-primary px-3 py-2 text-xs font-medium text-primary-foreground shadow-md sm:inset-x-auto sm:right-auto sm:left-3 sm:top-12">
   253→   253→          <span className="flex items-center gap-1.5">
   254→   254→            <Crosshair className="h-3.5 w-3.5 shrink-0" /> Кликните по карте — точка заявки №
   255→   255→            {placementOrderId}
   256→   256→          </span>
   257→   257→        </div>
   258→   258→      )}
   259→   259→    </div>
   260→   260→  );
   261→   261→}

   265→   265→function OrderChat({
   266→   266→  botId,
   267→   267→  orderId,
   268→   268→  hasConversation,
   269→   269→}: {
   270→   270→  botId: string;
   271→   271→  orderId: string;
   272→   272→  hasConversation: boolean;
   273→   273→}) {
   274→   274→  const { toast } = useToast();
   275→   275→  const [messages, setMessages] = useState<OrderMessageDto[]>([]);
   276→   276→  const [text, setText] = useState('');
   277→   277→  const [sending, setSending] = useState(false);
   278→   278→  const scrollRef = useRef<HTMLDivElement>(null);
   279→   279→
   280→   280→  const load = useCallback(async () => {
   281→   281→    try {
   282→   282→      const d = await api<{ messages: OrderMessageDto[] }>(
   283→   283→        `/api/bots/${botId}/orders/${orderId}/messages`
   284→   284→      );
   285→   285→      setMessages(d.messages);
   286→   286→    } catch {
   287→   287→      /* polling errors ignored */
   288→   288→    }
   289→   289→  }, [botId, orderId]);
   290→   290→
   291→   291→  useEffect(() => {
   292→   292→    setMessages([]);
   293→   293→    load();
   294→   294→    const t = setInterval(load, 3000);
   295→   295→    return () => clearInterval(t);
   296→   296→  }, [load]);
   297→   297→
   298→   298→  useEffect(() => {
   299→   299→    requestAnimationFrame(() => {
   300→   300→      scrollRef.current?.scrollTo({ top: scrollRef.current.scrollHeight });
   301→   301→    });
   302→   302→  }, [messages.length]);
   303→   303→
   304→   304→  const send = async () => {
   305→   305→    const t = text.trim();
   306→   306→    if (!t) return;
   307→   307→    setSending(true);
   308→   308→    setText('');
   309→   309→    try {
   310→   310→      const d = await api<{ message: OrderMessageDto; delivered: boolean; deliveryError?: string }>(
   311→   311→        `/api/bots/${botId}/orders/${orderId}/messages`,
   312→   312→        { method: 'POST', body: JSON.stringify({ text: t }) }
   313→   313→      );
   314→   314→      setMessages((prev) => [...prev, d.message]);
   315→   315→      if (!d.delivered && d.deliveryError) {
   316→   316→        toast({ title: 'Не доставлено в мессенджер', description: d.deliveryError, variant: 'destructive' });
   317→   317→      }
   318→   318→    } catch {
   319→   319→      toast({ title: 'Не удалось отправить', variant: 'destructive' });
   320→   320→    } finally {
   321→   321→      setSending(false);
   322→   322→    }
   323→   323→  };
   324→   324→
   325→   325→  return (
   326→   326→    <div className="flex min-h-0 flex-1 flex-col">
   327→   327→      <div className="flex items-center gap-2 border-b px-3 py-2 text-xs text-muted-foreground">
   328→   328→        <MessageSquare className="h-3.5 w-3.5" />
   329→   329→        Чат с клиентом по заявке — сообщения доставляются в его мессенджер
   330→   330→      </div>
   331→   331→      <div ref={scrollRef} className="min-h-0 flex-1 space-y-2 overflow-y-auto bg-muted/30 p-3">
   332→   332→        {messages.length === 0 && (
   333→   333→          <div className="flex h-full items-center justify-center text-center text-xs text-muted-foreground">
   334→   334→            Сообщений по заявке пока нет.
   335→   335→            <br />
   336→   336→            Клиент может написать в боте, открыв карточку своей заявки.
   337→   337→          </div>
   338→   338→        )}
   339→   339→        {messages.map((m) => {
   340→   340→          const fromClient = m.role === 'user';
   341→   341→          return (
   342→   342→            <div key={m.id} className={cn('flex', fromClient ? 'justify-start' : 'justify-end')}>
   343→   343→              <div
   344→   344→                className={cn(
   345→   345→                  'max-w-[85%] space-y-0.5 rounded-2xl px-3 py-1.5 text-sm leading-snug',
   346→   346→                  fromClient
   347→   347→                    ? 'rounded-bl-md border bg-card'
   348→   348→                    : 'rounded-br-md bg-primary text-primary-foreground'
   349→   349→                )}
   350→   350→              >
   351→   351→                {m.nodeId === '__operator' && (
   352→   352→                  <div className="flex items-center gap-1 text-[10px] opacity-70">
   353→   353→                    <Headset className="h-3 w-3" /> оператор
   354→   354→                  </div>
   355→   355→                )}
   356→   356→                <div className="whitespace-pre-wrap">{m.text}</div>
   357→   357→                <div className="text-right text-[9px] opacity-60">
   358→   358→                  {new Date(m.createdAt).toLocaleTimeString('ru-RU', {
   359→   359→                    hour: '2-digit',
   360→   360→                    minute: '2-digit',
   361→   361→                  })}
   362→   362→                </div>
   363→   363→              </div>
   364→   364→            </div>
   365→   365→          );
   366→   366→        })}
   367→   367→      </div>
   368→   368→      <form
   369→   369→        className="flex items-center gap-2 border-t p-2.5 pb-[max(0.625rem,env(safe-area-inset-bottom))]"
   370→   370→        onSubmit={(e) => {
   371→   371→          e.preventDefault();
   372→   372→          send();
   373→   373→        }}
   374→   374→      >
   375→   375→        <Input
   376→   376→          placeholder={
   377→   377→            hasConversation
   378→   378→              ? 'Ответить клиенту…'
   379→   379→              : 'Диалог с клиентом не привязан — чат только для чтения'
   380→   380→          }
   381→   381→          value={text}
   382→   382→          disabled={!hasConversation}
   383→   383→          onChange={(e) => setText(e.target.value)}
   384→   384→        />
   385→   385→        <Button
   386→   386→          type="submit"
   387→   387→          size="icon"
   388→   388→          className="h-10 w-10 shrink-0"
   389→   389→          disabled={sending || !hasConversation || !text.trim()}
   390→   390→          aria-label="Отправить"
   391→   391→        >
   392→   392→          {sending ? <Loader2 className="h-4 w-4 animate-spin" /> : <Send className="h-4 w-4" />}
   393→   393→        </Button>
   394→   394→      </form>
   395→   395→    </div>
   396→   396→  );
   397→   397→}
