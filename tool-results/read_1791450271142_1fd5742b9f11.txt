    72→/** Цвет маркера на карте */
    73→function markerColor(order: OrderDto): string {
    74→  if (ARCHIVE_STATUSES.includes(order.status)) return 'bg-slate-400';
    75→  switch (order.type) {
    76→    case 'waste':
    77→      return 'bg-emerald-500';
    78→    case 'kgm':
    79→      return 'bg-amber-500';
    80→    default:
    81→      return 'bg-violet-500';
    82→  }
    83→}

    95→/** «Обновлено N с назад» для чипа свежести данных */
    96→function fmtAgo(ts: number, now: number): string {
    97→  const s = Math.max(0, Math.round((now - ts) / 1000));
    98→  if (s < 8) return 'только что';
    99→  if (s < 60) return `${s} с назад`;
   100→  const m = Math.floor(s / 60);
   101→  if (m < 60) return `${m} мин назад`;
   102→  return `${Math.floor(m / 60)} ч назад`;
   103→}

   105→/** Дедуп-слияние сообщений по id (append-only) + сортировка по createdAt.
   106→ *  Если нового ничего нет — возвращает прежний массив (без ре-рендера).
   107→ *  Работает в обоих режимах бэкенда: и с ?take=N, и без него. */
   108→function mergeMessages(prev: OrderMessageDto[], incoming: OrderMessageDto[]): OrderMessageDto[] {
   109→  if (incoming.length === 0) return prev;
   110→  const known = new Set(prev.map((m) => m.id));
   111→  const fresh = incoming.filter((m) => !known.has(m.id));
   112→  if (fresh.length === 0) return prev;
   113→  return [...prev, ...fresh].sort(
   114→    (a, b) => new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime()
   115→  );
   116→}

   118→/** localStorage с защитой от недоступности (SSR / приватный режим) */
   119→function readStored(key: string): string | null {
   120→  try {
   121→    return window.localStorage.getItem(key);
   122→  } catch {
   123→    return null;
   124→  }
   125→}
   126→
   127→function writeStored(key: string, value: string): void {
   128→  try {
   129→    window.localStorage.setItem(key, value);
   130→  } catch {
   131→    /* ignore */
   132→  }
   133→}

   144→// ─── Карта OpenFreeMap (MapLibre GL, бесплатные векторные тайлы без ключа) ───
   145→
   146→function OrdersMap({
   147→  orders,
   148→  showCompleted,
   149→  mytkoEnabled,
   150→  placementOrderId,
   151→  selectedId,
   152→  focusOrderId,
   153→  focusTick,
   154→  onSelect,
   155→  onPlace,
   156→  onHighlight,
   157→}: {
   158→  orders: OrderDto[];
   159→  showCompleted: boolean;
   160→  mytkoEnabled: boolean;
   161→  placementOrderId: string | null;
   162→  selectedId: string | null;
   163→  focusOrderId: string | null;
   164→  focusTick: number;
   165→  onSelect: (id: string) => void;
   166→  onPlace: (lat: number, lng: number) => void;
   167→  /** Выделить заявку без открытия карточки (клик по списку точек) */
   168→  onHighlight: (id: string) => void;
   169→}) {
   170→  const containerRef = useRef<HTMLDivElement>(null);
   171→  const mapRef = useRef<MlMap | null>(null);
   172→  const mlRef = useRef<typeof import('maplibre-gl') | null>(null);
   173→  /** маркеры по id заявки — чтобы сдвигать их при «spiderfy» совпадающих точек */
   174→  const markersRef = useRef<Map<string, MlMarker>>(new Map());
   175→  /** исходные координаты заявки (без сдвига) — для пересчёта раскладки на каждом move */
   176→  const baseCoordsRef = useRef<Map<string, { lat: number; lng: number }>>(new Map());
   177→  /** id → заявка: эффекту выделения, чтобы перекрашивать маркеры без пересоздания */
   178→  const ordersByIdRef = useRef<Map<string, OrderDto>>(new Map());
   179→  /** зеркальная копия selectedId — доступ из эффектов маркеров без их пересоздания */
   180→  const selectedIdRef = useRef<string | null>(selectedId);
   181→  const [ready, setReady] = useState(false);
   182→  const [failed, setFailed] = useState(false);
   183→  const [listOpen, setListOpen] = useState(false);
   184→
   185→  // Инициализация карты (динамический импорт — компонент клиентский)
   186→  useEffect(() => {
   187→    let cancelled = false;
   188→    (async () => {
   189→      try {
   190→        const ml = await import('maplibre-gl');
   191→        if (cancelled || !containerRef.current) return;
   192→        // Воркер MapLibre не собирается Turbopack'ом — берём локальную копию из public/
   193→        ml.setWorkerUrl('/maplibre-gl-worker.mjs');
   194→        mlRef.current = ml;
   195→        const map = new ml.Map({
   196→          container: containerRef.current,
   197→          style: 'https://tiles.openfreemap.org/styles/liberty',
   198→          center: [32.3, 58.35], // Новгородская область (Великий Новгород — Боровичи — Валдай)
   199→          zoom: 7,
   200→        });
   201→        map.addControl(new ml.NavigationControl({ showCompass: false }), 'top-right');
   202→        map.on('load', () => {
   203→          if (!cancelled) setReady(true);
   204→        });
   205→        map.on('error', (e) => {
   206→          // Тайловая сеть недоступна — сообщаем, но карту не убиваем
   207→          console.warn('[orders map]', e?.error?.message ?? e);
   208→        });
   209→        mapRef.current = map;
   210→      } catch (err) {
   211→        console.error('[orders map] init failed', err);
   212→        if (!cancelled) setFailed(true);
   213→      }
   214→    })();
   215→    return () => {
   216→      cancelled = true;
   217→      mapRef.current?.remove();
   218→      mapRef.current = null;
   219→      setReady(false);
   220→    };
   221→  }, []);
   222→
   223→  // Маркеры заявок
   224→  const visible = orders.filter(
   225→    (o) => o.lat != null && o.lng != null && (showCompleted || ACTIVE_STATUSES.includes(o.status))
   226→  );
   227→
   228→  /**
   229→   * «Spiderfy» совпадающих точек: несколько заявок с одним адресом геокодятся
   230→   * в идентичные координаты и ложатся маркерами друг на друга — видна только
   231→   * одна. Группируем их по округлённым координатам и раскладываем кружком
   232→   * с ПОСТОЯННЫМ пиксельным радиусом (пересчёт на каждом move/zoom), чтобы
   233→   * все точки были видны и кликабельны на любом масштабе карты.
   234→   */
   235→  const spreadMarkers = useCallback(() => {
   236→    const map = mapRef.current;
   237→    if (!map) return;
   238→
   239→    // группы совпадающих координат (6 знаков ≈ 0.1 м — попадают и «почти равные»)
   240→    const groups = new Map<string, string[]>();
   241→    for (const [id, c] of baseCoordsRef.current) {
   242→      const key = `${c.lat.toFixed(6)}|${c.lng.toFixed(6)}`;
   243→      const g = groups.get(key);
   244→      if (g) g.push(id);
   245→      else groups.set(key, [id]);
   246→    }
   247→
   248→    const zoom = map.getZoom();
   249→    // пикселей на градус долготы в Web Mercator (тайл MapLibre = 512px!)
   250→    const pxPerDegLng = 512 * Math.pow(2, zoom) / 360;
   251→    const SPREAD_PX = 44; // диаметр раскладки вокруг истинной точки
   252→
   253→    for (const ids of groups.values()) {
   254→      const n = ids.length;
   255→      ids.forEach((id, i) => {
   256→        const marker = markersRef.current.get(id);
   257→        const base = baseCoordsRef.current.get(id);
   258→        if (!marker || !base) return;
   259→        if (n === 1) {
   260→          marker.setLngLat([base.lng, base.lat]);
   261→          return;
   262→        }
   263→        // равномерно по окружности; при 2 точках — слева и справа от истинной
   264→        const angle = (2 * Math.PI * i) / n + (n === 2 ? Math.PI / 2 : Math.PI / 6);
   265→        const dxPx = Math.cos(angle) * (SPREAD_PX / 2);
   266→        const dyPx = Math.sin(angle) * (SPREAD_PX / 2);
   267→        const dLng = dxPx / pxPerDegLng;
   268→        const dLat = (dyPx / pxPerDegLng) * Math.cos((base.lat * Math.PI) / 180);
   269→        marker.setLngLat([base.lng + dLng, base.lat + dLat]);
   270→      });
   271→    }
   272→  }, []);
   273→
   274→  // ДИФФ-обновление маркеров (IMP-F10): удаляем исчезнувшие, обновляем координаты
   275→  // и цвет существующих, добавляем только новые — поллинг больше не пересоздаёт
   276→  // все маркеры и карта не «мигает». Выделение (selectedId) — отдельным эффектом ниже.
   277→  useEffect(() => {
   278→    const map = mapRef.current;
   279→    const ml = mlRef.current;
   280→    if (!map || !ml || !ready) return;
   281→
   282→    // (a) входящий набор видимых заявок
   283→    const incoming = new Map<string, OrderDto>();
   284→    for (const o of visible) incoming.set(o.id, o);
   285→    ordersByIdRef.current = incoming;
   286→
   287→    // (b) удаляем маркеры, которых больше нет
   288→    for (const [id, m] of markersRef.current) {
   289→      if (!incoming.has(id)) {
   290→        m.remove();
   291→        markersRef.current.delete(id);
   292→        baseCoordsRef.current.delete(id);
   293→      }
   294→    }
   295→
   296→    // (c)+(d) обновляем существующие / добавляем только новые
   297→    for (const o of visible) {
   298→      const lat = o.lat as number;
   299→      const lng = o.lng as number;
   300→      baseCoordsRef.current.set(o.id, { lat, lng });
   301→      const sig = `${o.status}|${o.type}`;
   302→      const existing = markersRef.current.get(o.id);
   303→      if (existing) {
   304→        const el = existing.getElement();
   305→        if (el.dataset.sig !== sig) {
   306→          // статус/тип изменились — перекрашиваем без пересоздания
   307→          el.dataset.sig = sig;
   308→          el.className = markerElClassName(o, o.id === selectedIdRef.current);
   309→        }
   310→      } else {
   311→        const el = document.createElement('button');
   312→        el.type = 'button';
   313→        el.dataset.sig = sig;
   314→        el.title = `Заявка №${o.number} — ${ORDER_TYPE_LABELS[o.type] ?? ''}`;
   315→        // Без hover:scale — масштабирование сдвигает маркер и «прыгает»;
   316→        // подсвечиваем тенью и рамкой, размер элемента не меняется
   317→        el.className = markerElClassName(o, o.id === selectedIdRef.current);
   318→        el.textContent = String(o.number);
   319→        el.addEventListener('click', (e) => {
   320→          e.stopPropagation();
   321→          onSelect(o.id);
   322→        });
   323→        const marker = new ml.Marker({ element: el })
   324→          .setLngLat([lng, lat])
   325→          .addTo(map);
   326→        markersRef.current.set(o.id, marker);
   327→      }
   328→    }
   329→
   330→    // (e) раскладываем совпадающие точки (spiderfy, 44px) и пересчитываем на каждом move
   331→    spreadMarkers();
   332→    map.on('move', spreadMarkers);
   333→    return () => {
   334→      map.off('move', spreadMarkers);
   335→    };
   336→    // зависимость — сериализованный список видимых маркеров (БЕЗ selectedId)
   337→  }, [ready, visible.map((o) => `${o.id}:${o.lat}:${o.lng}:${o.status}`).join('|'), onSelect, spreadMarkers]);
   338→
   339→  // Выделение выбранной заявки — ОТДЕЛЬНЫЙ эффект: только переключаем класс
   340→  // DOM-элемента существующих маркеров (getElement), ничего не пересоздаём
   341→  useEffect(() => {
   342→    selectedIdRef.current = selectedId;
   343→    for (const [id, m] of markersRef.current) {
   344→      const o = ordersByIdRef.current.get(id);
   345→      if (o) m.getElement().className = markerElClassName(o, id === selectedId);
   346→    }
   347→  }, [selectedId]);
   348→
   349→  // Наведение на заявку из списка/карточки: перелетаем к точке (focusTick —
   350→  // чтобы повторный клик по той же заявке тоже срабатывал)
   351→  useEffect(() => {
   352→    const map = mapRef.current;
   353→    if (!map || !ready || !focusOrderId) return;
   354→    const o = visible.find((x) => x.id === focusOrderId);
   355→    if (!o || o.lat == null || o.lng == null) return;
   356→    map.flyTo({ center: [o.lng, o.lat], zoom: Math.max(map.getZoom(), 15), duration: 800 });
   357→  }, [ready, focusOrderId, focusTick]);
   358→
   359→  // Во время режима указания точки список скрываем (баннер занимает то же место)
   360→  useEffect(() => {
   361→    if (placementOrderId) setListOpen(false);
   362→  }, [placementOrderId]);
   363→
   364→  // Переход к точке из списка: перелёт + выделение, без открытия карточки
   365→  const jumpToListPoint = useCallback(
   366→    (o: OrderDto) => {
   367→      const map = mapRef.current;
   368→      if (!map || o.lat == null || o.lng == null) return;
   369→      onHighlight(o.id);
   370→      map.flyTo({ center: [o.lng, o.lat], zoom: Math.max(map.getZoom(), 15), duration: 800 });
   371→      setListOpen(false);
   372→    },
   373→    [onHighlight]
   374→  );
   375→
   376→  // Режим указания точки: клик по карте → координаты заявки
   377→  const handleMapClick = useCallback(
   378→    (lat: number, lng: number) => onPlace(lat, lng),
   379→    [onPlace]
   380→  );
   381→  useEffect(() => {
   382→    const map = mapRef.current;
   383→    if (!map || !ready || !placementOrderId) return;
   384→    const onClick = (e: { lngLat: { lat: number; lng: number } }) =>
   385→      handleMapClick(e.lngLat.lat, e.lngLat.lng);
   386→    map.on('click', onClick);
   387→    map.getCanvas().style.cursor = 'crosshair';
   388→    return () => {
   389→      map.off('click', onClick);
   390→      map.getCanvas().style.cursor = '';
   391→    };
   392→  }, [ready, placementOrderId, handleMapClick]);
   393→
   394→  if (failed) {
   395→    return (
   396→      <div className="flex h-full min-h-[420px] flex-col items-center justify-center gap-3 rounded-xl border bg-muted/30 text-center text-muted-foreground">
   397→        <MapPin className="h-10 w-10 opacity-30" />
   398→        <p className="max-w-xs text-sm">
   399→          Не удалось загрузить карту OpenFreeMap. Проверьте доступ к tiles.openfreemap.org и
   400→          обновите страницу.
   401→        </p>
   402→      </div>
   403→    );
   404→  }
   405→
   406→  return (
   407→    <div className="relative h-full min-h-[380px] overflow-hidden rounded-xl border bg-muted/30">
   408→      <div ref={containerRef} className="h-full w-full" aria-label="Карта заявок" />
   409→      {!ready && (
   410→        <div className="absolute inset-0 flex items-center justify-center gap-2 text-sm text-muted-foreground">
   411→          <Loader2 className="h-4 w-4 animate-spin" /> Загружаю карту OpenFreeMap…
   412→        </div>
   413→      )}
   414→      {/* Легенда (компактная, не перекрывает атрибуцию и контролы) */}
   415→      <div className="absolute bottom-2.5 left-2.5 z-10 flex flex-wrap items-center gap-x-2.5 gap-y-0.5 rounded-lg border bg-background/95 px-2.5 py-1.5 text-[10px] leading-tight shadow-sm backdrop-blur sm:bottom-3 sm:left-3 sm:text-[11px]">
   416→        <span className="flex items-center gap-1.5">
   417→          <span className="h-2.5 w-2.5 rounded-full bg-emerald-500" /> отходы
   418→        </span>
   419→        <span className="flex items-center gap-1.5">
   420→          <span className="h-2.5 w-2.5 rounded-full bg-amber-500" /> КГМ
   421→        </span>
   422→        <span className="flex items-center gap-1.5">
   423→          <span className="h-2.5 w-2.5 rounded-full bg-slate-400" /> выполненные
   424→        </span>
   425→      </div>
   426→      {/* Счётчик маркеров + кнопка разворачиваемого списка точек */}
   427→      <div className="absolute left-2.5 top-2.5 z-10 flex items-start gap-2 sm:left-3 sm:top-3">
   428→        <div
   429→          role="status"
   430→          aria-label={`Точек на карте: ${visible.length}`}
   431→          className="rounded-lg border bg-background/95 px-2.5 py-1.5 text-xs shadow-sm backdrop-blur"
   432→        >
   433→          <MapPin className="mr-1 inline h-3 w-3 text-primary" aria-hidden />
   434→          На карте: <b className="tabular-nums">{visible.length}</b>
   435→        </div>
   436→        {!placementOrderId && visible.length > 0 && (
   437→          <button
   438→            type="button"
   439→            onClick={() => setListOpen((v) => !v)}
   440→            aria-expanded={listOpen}
   441→            className={cn(
   442→              'flex items-center gap-1 rounded-lg border bg-background/95 px-2.5 py-1.5 text-xs shadow-sm backdrop-blur transition-colors hover:bg-muted',
   443→              listOpen && 'bg-muted'
   444→            )}
   445→          >
   446→            <List className="h-3 w-3 text-primary" />
   447→            Список
   448→            <ChevronDown
   449→              className={cn('h-3 w-3 transition-transform duration-200', listOpen && 'rotate-180')}
   450→            />
   451→          </button>
   452→        )}
   453→      </div>

   522→/** Optimistic-пузырь: пока sending — «отправляется…», при ошибке — «Повторить» (текст не теряется) */
   523→type PendingMessage = { tmpId: string; text: string; status: 'sending' | 'error' };
   524→
   525→function OrderChat({
   526→  botId,
   527→  orderId,
   528→  hasConversation,
   529→}: {
   530→  botId: string;
   531→  orderId: string;
   532→  hasConversation: boolean;
   533→}) {
   534→  const { toast } = useToast();
   535→  const [messages, setMessages] = useState<OrderMessageDto[]>([]);
   536→  const [pending, setPending] = useState<PendingMessage[]>([]);
   537→  const [text, setText] = useState('');
   538→  const [sending, setSending] = useState(false);
   539→  const scrollRef = useRef<HTMLDivElement>(null);
   540→  /** Пользователь у дна переписки? Скролл при новых сообщениях — только тогда */
   541→  const isNearBottomRef = useRef(true);
   542→  /** Ключ активной заявки — защита от гонки: ответ старого запроса не должен
   543→   *  попасть в чат новой заявки при быстром переключении карточек */
   544→  const activeKeyRef = useRef('');
   545→
   546→  // Поллинг с дедуп-слиянием по id (IMP-F11): первая загрузка — полная история,
   547→  // далее ?take=50. Слияние защитает и бэкенд без поддержки ?take, и гонки ответов.
   548→  const fetchMessages = useCallback(
   549→    async (take: number | null) => {
   550→      const key = `${botId}/${orderId}`;
   551→      try {
   552→        const d = await api<{ messages: OrderMessageDto[] }>(
   553→          `/api/bots/${botId}/orders/${orderId}/messages${take ? `?take=${take}` : ''}`
   554→        );
   555→        if (activeKeyRef.current !== key) return; // ответ устарел — заявка уже другая
   556→        setMessages((prev) => mergeMessages(prev, d.messages));
   557→      } catch {
   558→        /* polling errors ignored */
   559→      }
   560→    },
   561→    [botId, orderId]
   562→  );
   563→
   564→  useEffect(() => {
   565→    activeKeyRef.current = `${botId}/${orderId}`;
   566→    setMessages([]);
   567→    setPending([]);
   568→    isNearBottomRef.current = true;
   569→    fetchMessages(null); // полная первая загрузка
   570→    let timer: ReturnType<typeof setInterval> | null = setInterval(() => fetchMessages(50), 3000);
   571→    // пауза поллинга на скрытой вкладке + немедленный тик при возврате
   572→    const onVisibility = () => {
   573→      if (document.hidden) {
   574→        if (timer) {
   575→          clearInterval(timer);
   576→          timer = null;
   577→        }
   578→      } else if (!timer) {
   579→        fetchMessages(50);
   580→        timer = setInterval(() => fetchMessages(50), 3000);
   581→      }
   582→    };
   583→    document.addEventListener('visibilitychange', onVisibility);
   584→    return () => {
   585→      if (timer) clearInterval(timer);
   586→      document.removeEventListener('visibilitychange', onVisibility);
   587→    };
   588→  }, [botId, orderId, fetchMessages]);
   589→
   590→  // Умный автоскролл (IMP-F11): тянем к дну только если пользователь и так у дна (< 120px)
   591→  useEffect(() => {
   592→    requestAnimationFrame(() => {
   593→      const el = scrollRef.current;
   594→      if (!el || !isNearBottomRef.current) return;
   595→      el.scrollTo({ top: el.scrollHeight });
   596→    });
   597→  }, [messages.length, pending]);
   598→
   599→  const handleScroll = useCallback(() => {
   600→    const el = scrollRef.current;
   601→    if (!el) return;
   602→    isNearBottomRef.current = el.scrollHeight - el.scrollTop - el.clientHeight < 120;
   603→  }, []);
   604→
   605→  const sendMessage = useCallback(
   606→    async (body: string, tmpId: string) => {
   607→      try {
   608→        const d = await api<{ message: OrderMessageDto; delivered: boolean; deliveryError?: string }>(
   609→          `/api/bots/${botId}/orders/${orderId}/messages`,
   610→          { method: 'POST', body: JSON.stringify({ text: body }) }
   611→        );
   612→        setMessages((prev) => mergeMessages(prev, [d.message]));
   613→        setPending((p) => p.filter((x) => x.tmpId !== tmpId));
   614→        if (!d.delivered && d.deliveryError) {
   615→          toast({ title: 'Не доставлено в мессенджер', description: d.deliveryError, variant: 'destructive' });
   616→        }
   617→      } catch {
   618→        // текст не теряем: пузырь становится ошибочным, доступна кнопка «Повторить»
   619→        setPending((p) => p.map((x) => (x.tmpId === tmpId ? { ...x, status: 'error' } : x)));
   620→        toast({ title: 'Не удалось отправить', variant: 'destructive' });
   621→      } finally {
   622→        setSending(false);
   623→      }
   624→    },
   625→    [botId, orderId, toast]
   626→  );
   627→
   628→  const send = () => {
   629→    const t = text.trim();
   630→    if (!t) return;
   631→    const tmpId = `tmp-${Date.now()}-${Math.random().toString(36).slice(2, 7)}`;
   632→    setSending(true);
   633→    setText('');
   634→    setPending((p) => [...p, { tmpId, text: t, status: 'sending' }]);
   635→    void sendMessage(t, tmpId);
   636→  };
   637→
   638→  const retry = (m: PendingMessage) => {
   639→    setPending((p) => p.map((x) => (x.tmpId === m.tmpId ? { ...x, status: 'sending' } : x)));
   640→    setSending(true);
   641→    void sendMessage(m.text, m.tmpId);
   642→  };
   643→
   644→  return (
   645→    <div className="flex min-h-0 flex-1 flex-col">
   646→      <div className="flex items-center gap-2 border-b px-3 py-2 text-xs text-muted-foreground">
   647→        <MessageSquare className="h-3.5 w-3.5" />
   648→        Чат с клиентом по заявке — сообщения доставляются в его мессенджер
   649→      </div>
   650→      {/* Чат с клиентом по заявке */}
   651→      <div
   652→        ref={scrollRef}
   653→        aria-live="polite"
   654→        onScroll={handleScroll}
   655→        className="min-h-0 flex-1 space-y-2 overflow-y-auto bg-muted/30 p-3"
   656→      >
   657→        {messages.length === 0 && pending.length === 0 && (
   658→          <div className="flex h-full flex-col items-center justify-center gap-2 px-4 text-center text-muted-foreground">
   659→            <div className="flex h-10 w-10 items-center justify-center rounded-xl bg-primary/10 text-primary" aria-hidden>
   660→              <MessageSquare className="h-5 w-5" />
   661→            </div>
   662→            <p className="text-xs">
   663→              Сообщений по заявке пока нет.
   664→              <br />
   665→              Клиент может написать в боте, открыв карточку своей заявки.
   666→            </p>
   667→          </div>
   668→        )}
   669→        {messages.map((m) => {
   670→          const fromClient = m.role === 'user';
   671→          return (
   672→            <div key={m.id} className={cn('flex', fromClient ? 'justify-start' : 'justify-end')}>
   673→              <div
   674→                className={cn(
   675→                  'max-w-[85%] space-y-0.5 rounded-2xl px-3 py-1.5 text-sm leading-snug',
   676→                  fromClient
   677→                    ? 'rounded-bl-md border bg-card'
   678→                    : 'rounded-br-md bg-primary text-primary-foreground'
   679→                )}
   680→              >
   681→                {m.nodeId === '__operator' && (
   682→                  <div className="flex items-center gap-1 text-[10px] opacity-70">
   683→                    <Headset className="h-3 w-3" /> оператор
   684→                  </div>
   685→                )}
   686→                <div className="whitespace-pre-wrap">{m.text}</div>
   687→                <div className="text-right text-[10px] opacity-60">
   688→                  {new Date(m.createdAt).toLocaleTimeString('ru-RU', {
   689→                    hour: '2-digit',
   690→                    minute: '2-digit',
   691→                  })}
   692→                </div>
   693→              </div>
   694→            </div>
   695→          );
   696→        })}
   697→        {/* Optimistic-пузыри (IMP-F11): полупрозрачный с часами — отправляется;
   698→            при ошибке — error-пузырь с кнопкой «Повторить», введённый текст сохранён */}
   699→        {pending.map((p) => (
   700→          <div key={p.tmpId} className="flex justify-end">
   701→            <div
   702→              className={cn(
   703→                'max-w-[85%] space-y-0.5 rounded-2xl rounded-br-md px-3 py-1.5 text-sm leading-snug',
   704→                p.status === 'error'
   705→                  ? 'border border-destructive/40 bg-destructive/10 text-destructive'
   706→                  : 'bg-primary text-primary-foreground opacity-70'
   707→              )}
   708→            >
   709→              <div className="whitespace-pre-wrap">{p.text}</div>
   710→              <div className="flex items-center justify-end gap-1 text-[10px] opacity-80">
   711→                {p.status === 'sending' ? (
   712→                  <>
   713→                    <Clock className="h-3 w-3" aria-hidden /> отправляется…
   714→                  </>
   715→                ) : (
   716→                  <button
   717→                    type="button"
   718→                    onClick={() => retry(p)}
   719→                    aria-label="Повторить отправку сообщения"
   720→                    className="rounded font-semibold underline underline-offset-2 transition-colors hover:opacity-100 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
   721→                  >
   722→                    Повторить
   723→                  </button>
   724→                )}
   725→              </div>
   726→            </div>
   727→          </div>
   728→        ))}
   729→      </div>
   730→      <form
   731→        className="flex items-center gap-2 border-t p-2.5 pb-[max(0.625rem,env(safe-area-inset-bottom))]"
   732→        onSubmit={(e) => {
   733→          e.preventDefault();
   734→          send();
   735→        }}
   736→      >
   737→        <Input
   738→          aria-label="Ответ клиенту"
   739→          placeholder={
   740→            hasConversation
   741→              ? 'Ответить клиенту…'
   742→              : 'Диалог с клиентом не привязан — чат только для чтения'
   743→          }
   744→          value={text}
   745→          disabled={!hasConversation}
   746→          onChange={(e) => setText(e.target.value)}
   747→        />
   748→        <Button
   749→          type="submit"
   750→          size="icon"
   751→          className="h-10 w-10 shrink-0"
   752→          disabled={sending || !hasConversation || !text.trim()}
   753→          aria-label="Отправить"
   754→        >
   755→          {sending ? <Loader2 className="h-4 w-4 animate-spin" /> : <Send className="h-4 w-4" />}
   756→        </Button>
   757→      </form>
   758→    </div>
   759→  );
   760→}

   764→function CopyButton({
   765→  value,
   766→  label,
   767→  className,
   768→}: {
   769→  value: string;
   770→  label: string;
   771→  className?: string;
   772→}) {
   773→  const { toast } = useToast();
   774→  const [copied, setCopied] = useState(false);
   775→  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   776→
   777→  useEffect(
   778→    () => () => {
   779→      if (timerRef.current) clearTimeout(timerRef.current);
   780→    },
   781→    []
   782→  );
   783→
   784→  const copy = async () => {
   785→    try {
   786→      await navigator.clipboard.writeText(value);
   787→      setCopied(true);
   788→      if (timerRef.current) clearTimeout(timerRef.current);
   789→      timerRef.current = setTimeout(() => setCopied(false), 1500);
   790→      toast({ title: `Скопировано: ${label}` });
   791→    } catch {
   792→      toast({ title: 'Не удалось скопировать', variant: 'destructive' });
   793→    }
   794→  };
   795→
   796→  return (
   797→    <button
   798→      type="button"
   799→      onClick={copy}
   800→      aria-label={`Копировать: ${label}`}
   801→      title={`Копировать: ${label}`}
   802→      className={cn(
   803→        'flex h-8 w-8 shrink-0 items-center justify-center rounded-lg text-muted-foreground transition-colors hover:bg-muted hover:text-foreground focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring',
   804→        className
   805→      )}
   806→    >
   807→      <Copy
   808→        className={cn('h-3.5 w-3.5', copied && 'text-emerald-600 dark:text-emerald-400')}
   809→        aria-hidden
   810→      />
   811→    </button>
   812→  );
   813→}

   855→function OrderDetailDialog({
   856→  botId,
   857→  orderId,
   858→  open,
   859→  onClose,
   860→  onSwitchOrder,
   861→  onPlaceRequest,
   862→  onShowOnMap,
   863→  onDataChanged,
   864→  mytkoEnabled,
   865→}: {
   866→  botId: string;
   867→  orderId: string | null;
   868→  open: boolean;
   869→  onClose: () => void;
   870→  onSwitchOrder: (id: string) => void;
   871→  onPlaceRequest: (orderId: string) => void;
   872→  onShowOnMap: (orderId: string) => void;
   873→  onDataChanged: () => void;
   874→  mytkoEnabled: boolean;
   875→}) {
   876→  const { toast } = useToast();
   877→  const [order, setOrder] = useState<OrderDto | null>(null);
   878→  const [clientOrders, setClientOrders] = useState<OrderDto[]>([]);
   879→  const [loading, setLoading] = useState(false);
   880→  const [assignee, setAssignee] = useState('');
   881→  const [address, setAddress] = useState('');
   882→  const [wishDate, setWishDate] = useState('');
   883→  const [comment, setComment] = useState('');
   884→  const [saving, setSaving] = useState(false);
   885→  const [geoLoading, setGeoLoading] = useState(false);
   886→  const [confirmGeo, setConfirmGeo] = useState(false);
   887→  const [confirmDelete, setConfirmDelete] = useState(false);
   888→  const [mytkoBusy, setMytkoBusy] = useState(false);
   889→
   890→  const load = useCallback(async () => {
   891→    if (!orderId) return;
   892→    setLoading(true);
   893→    try {
   894→      const d = await api<{ order: OrderDto; clientOrders: OrderDto[] }>(
   895→        `/api/bots/${botId}/orders/${orderId}`
   896→      );
   897→      setOrder(d.order);
   898→      setClientOrders(d.clientOrders);
   899→      setAssignee(d.order.assignee ?? '');
   900→      setAddress(d.order.address ?? '');
   901→      setWishDate(d.order.wishDate ?? '');
   902→      setComment(d.order.comment ?? '');
   903→    } catch {
   904→      toast({ title: 'Не удалось загрузить заявку', variant: 'destructive' });
   905→    } finally {
   906→      setLoading(false);
   907→    }
   908→  }, [botId, orderId, toast]);
   909→
   910→  useEffect(() => {
   911→    if (open && orderId) load();
   912→    if (!open) {
   913→      setOrder(null);
   914→      setClientOrders([]);
   915→      setConfirmDelete(false);
   916→      setConfirmGeo(false);
   917→    }
   918→  }, [open, orderId, load]);
   919→
   920→  const patch = async (data: Record<string, unknown>, successMsg?: string) => {
   921→    if (!orderId) return;
   922→    setSaving(true);
   923→    try {
   924→      const d = await api<{ order: OrderDto }>(`/api/bots/${botId}/orders/${orderId}`, {
   925→        method: 'PATCH',
   926→        body: JSON.stringify(data),
   927→      });
   928→      setOrder(d.order);
   929→      onDataChanged();
   930→      if (successMsg) toast({ title: successMsg });
   931→    } catch {
   932→      toast({ title: 'Ошибка сохранения', variant: 'destructive' });
   933→    } finally {
   934→      setSaving(false);
   935→    }
   936→  };
   937→
   938→  /**
   939→   * Геокодирование адреса. Ручная точка защищена: если координаты ставили
   940→   * вручную — сначала подтверждение; если адрес не нашёлся — прежние
   941→   * координаты остаются (не «слетают»).
