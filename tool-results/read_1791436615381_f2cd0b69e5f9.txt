    36→/**
    37→ * Валидация телефона РФ: 10–11 цифр (можно с +7/8, скобками, дефисами).
    38→ * Возвращает true, если телефон похож на настоящий.
    39→ */
    40→export function isValidPhone(input: string): boolean {
    41→  const digits = input.replace(/\D/g, '');
    42→  if (digits.length === 11 && (digits[0] === '7' || digits[0] === '8')) return true;
    43→  if (digits.length === 10) return true;
    44→  return false;
    45→}

    47→/**
    48→ * Проверка ответа на вопрос по data.validate.
    49→ * Возвращает текст ошибки для переспрашивания или null, если ответ корректен.
    50→ * «пропустить» обрабатывается вызывающим кодом.
    51→ */
    52→export async function validateAnswer(
    53→  data: FlowNode['data'],
    54→  input: string,
    55→  vars: Record<string, string>
    56→): Promise<string | null> {
    57→  if (data.validate === 'phone') {
    58→    if (isValidPhone(input)) return null;
    59→    return (
    60→      'Телефон выглядит некорректно 🙏 Введите номер в формате +7 900 123-45-67 (10–11 цифр) ' +
    61→      'или напишите «пропустить».'
    62→    );
    63→  }
    64→
    65→  if (data.validate === 'address') {
    66→    const q = input.trim();
    67→    if (q.length < 3) {
    68→      return 'Уточните адрес подробнее: улица и номер дома (например: улица Гоголя, 5).';
    69→    }
    70→    // Город шагом раньше (выбор города обслуживания)
    71→    const city = ['city', 'kgm_city']
    72→      .map((k) => (vars[k] ?? '').trim())
    73→      .find((c) => c.length > 0);
    74→
    75→    // Строгий режим: только РФ + при известном городе проверяем, что найденный
    76→    // объект действительно в этом городе (мусорные совпадения не проходят)
    77→    const ok = city
    78→      ? q.toLowerCase().includes(city.toLowerCase())
    79→        ? await geocodeVerify(q, city)
    80→        : await geocodeVerify(`${city}, ${q}`, city)
    81→      : await geocodeVerify(q);
    82→
    83→    if (ok) return null;
    84→    return (
    85→      'Не нашёл такой адрес на карте 🤔 Проверьте улицу и номер дома ' +
    86→      '(можно с корпусом: улица Гоголя, 5к2). Либо напишите «пропустить», чтобы продолжить без проверки.'
    87→    );
    88→  }
    89→
    90→  return null;
    91→}

   127→/** Подстановка переменных {{name}} в текст */
   128→export function interpolate(
   129→  text: string | undefined,
   130→  vars: Record<string, string>,
   131→  input?: string
   132→): string {
   133→  if (!text) return '';
   134→  let out = text.replace(/\{\{\s*([\w.]+)\s*\}\}/g, (_m, key: string) => {
   135→    if (key === 'input' && input !== undefined) return input;
   136→    const v = vars[key];
   137→    return v !== undefined ? v : '';
   138→  });
   139→  // Приглаживаем пустые подстановки: «Завтра, » → «Завтра», «Боровичи,» → «Боровичи»
   140→  out = out.replace(/,\s*,+/g, ', ');
   141→  out = out.replace(/,\s*$/gm, '');
   142→  return out;
   143→}

   145→function findNode(flow: Flow, id: string): FlowNode | null {
   146→  return flow.nodes.find((n) => n.id === id) ?? null;
   147→}
   148→
   149→function findStart(flow: Flow): FlowNode | null {
   150→  return flow.nodes.find((n) => n.type === 'start') ?? flow.nodes[0] ?? null;
   151→}
   152→
   153→function nextNode(flow: Flow, nodeId: string, handle?: string | null): FlowNode | null {
   154→  const edge = flow.edges.find(
   155→    (e) => e.source === nodeId && (handle ? (e.sourceHandle ?? '') === handle : true)
   156→  );
   157→  return edge ? findNode(flow, edge.target) : null;
   158→}
   159→
   160→function findMenuNode(flow: Flow, lastMenuId?: string): FlowNode | null {
   161→  if (lastMenuId) {
   162→    const n = findNode(flow, lastMenuId);
   163→    if (n?.type === 'buttons') return n;
   164→  }
   165→  return flow.nodes.find((n) => n.type === 'buttons') ?? null;
   166→}
   167→
   168→/**
   169→ * Главное меню сценария: первый узел «Кнопки» на пути от старта
   170→ * (BFS), иначе — первый узел «Кнопки» в сценарии. Используется,
   171→ * чтобы вернуть пользователя в меню (например, после закрытия обращения).
   172→ */
   173→export function findMainMenu(flow: Flow): FlowNode | null {
   174→  const start = findStart(flow);
   175→  if (start) {
   176→    const visited = new Set<string>([start.id]);
   177→    const queue: string[] = [start.id];
   178→    while (queue.length) {
   179→      const id = queue.shift() as string;
   180→      const node = findNode(flow, id);
   181→      if (!node) continue;
   182→      if (node.type === 'buttons') return node;
   183→      for (const e of flow.edges) {
   184→        if (e.source === id && !visited.has(e.target)) {
   185→          visited.add(e.target);
   186→          queue.push(e.target);
   187→        }
   188→      }
   189→    }
   190→  }
   191→  return flow.nodes.find((n) => n.type === 'buttons') ?? null;
   192→}
   193→
   194→/**
   195→ * Команда старта сценария (/start, «старт», «начать») — детерминированный вход
   196→ * в сценарий, минуя ИИ-ассистента (мессенджеры шлют /start при первом открытии бота).
   197→ */
   198→function isStartCommand(text: string): boolean {
   199→  return /^\/(start|старт|начать)$|^(старт|начать)\s*$/i.test(text.trim());
   200→}

   202→function evalCondition(
   203→  cond: FlowNode['data']['condition'],
   204→  vars: Record<string, string>
   205→): boolean {
   206→  if (!cond) return false;
   207→  const leftRaw = (cond.left ?? '').trim();
   208→  const left = leftRaw.includes('{{')
   209→    ? interpolate(leftRaw, vars)
   210→    : (vars[leftRaw] ?? leftRaw);
   211→  const rightRaw = cond.right ?? '';
   212→  const right = rightRaw.includes('{{') ? interpolate(rightRaw, vars) : rightRaw;
   213→  const op: ConditionOp = cond.op ?? 'eq';
   214→  switch (op) {
   215→    case 'eq':
   216→      return left.trim().toLowerCase() === right.trim().toLowerCase();
   217→    case 'neq':
   218→      return left.trim().toLowerCase() !== right.trim().toLowerCase();
   219→    case 'contains':
   220→      return left.toLowerCase().includes(right.toLowerCase());
   221→    case 'not_contains':
   222→      return !left.toLowerCase().includes(right.toLowerCase());
   223→    case 'gt':
   224→      return parseFloat(left) > parseFloat(right);
   225→    case 'lt':
   226→      return parseFloat(left) < parseFloat(right);
   227→    case 'empty':
   228→      return left.trim() === '';
   229→    case 'not_empty':
   230→      return left.trim() !== '';
   231→    default:
   232→      return false;
   233→  }
   234→}

   238→interface AiAskData {
   239→  prompt?: string;
   240→  knowledge?: string;
   241→  useMemory?: boolean;

   244→async function runAI(
   245→  data: AiAskData,
   246→  state: EngineState,
   247→  lastInput: string,
   248→  extraSystem?: string
   249→): Promise<string> {  const zai = await ZAI.create();
   250→
   251→  const sysParts: string[] = [];
   252→  sysParts.push(
   253→    data.prompt?.trim() ||
   254→      'Ты — вежливый бот техподдержки. Отвечай кратко, по делу и на языке пользователя.'
   255→  );
   256→  if (extraSystem?.trim()) {
   257→    sysParts.push(extraSystem.trim());
   258→  }
   259→  if (data.knowledge?.trim()) {
   260→    sysParts.push(`База знаний компании:\n${data.knowledge.trim()}`);
   261→  }
   262→  const varLines = Object.entries(state.vars)
   263→    .filter(([k]) => !k.startsWith('__'))
   264→    .map(([k, v]) => `${k}: ${v}`);
   265→  if (varLines.length) {
   266→    sysParts.push(`Известные данные о пользователе:\n${varLines.join('\n')}`);
   267→  }
   268→
   269→  const messages: { role: 'assistant' | 'user'; content: string }[] = [
   270→    { role: 'assistant', content: sysParts.join('\n\n') },
   271→  ];
   272→
   273→  if (data.useMemory !== false) {
   274→    for (const h of state.history.slice(-MEMORY_TAIL)) {
   275→      messages.push({
   276→        role: h.role === 'user' ? 'user' : 'assistant',
   277→        content: h.text,
   278→      });
   279→    }
   280→  }
   281→  if (lastInput) {
   282→    messages.push({ role: 'user', content: lastInput });
   283→  } else if (messages.length === 1) {
   284→    messages.push({ role: 'user', content: 'Привет' });
   285→  }
   286→
   287→  const completion = await zai.chat.completions.create({
   288→    messages,
   289→    thinking: { type: 'disabled' },
   290→  });
   291→  const content = completion.choices[0]?.message?.content;
   292→  return content && content.trim().length > 0
   293→    ? content
   294→    : 'Извините, я не смог сформулировать ответ. Попробуйте перефразировать вопрос.';
   295→}

   297→/** Ответ ИИ-ассистента на свободный вопрос (используется и режимом «Свободный чат») */
   298→export async function askAssistant(
   299→  assistant: AiAssistantConfig,
   300→  state: EngineState,
   301→  userText: string
   302→): Promise<string | null> {
   303→  try {
   304→    const raw = await runAI(
   305→      { prompt: assistant.prompt, knowledge: assistant.knowledge, useMemory: true },
   306→      state,
   307→      userText,
   308→      ASSISTANT_SYSTEM_RULES
   309→    );
   310→    return raw && raw.trim().length > 0 ? raw : null;
   311→  } catch (err) {
   312→    console.error('[flow-engine] askAssistant error:', err);
   313→    return null;
   314→  }
   315→}

   317→// ─── Оркестратор: главный цикл выполнения сценария ───────────────────────────
   318→
   319→export async function runEngine(
   320→  flow: Flow,
   321→  input: string | null,
   322→  prevState: EngineState | null,
   323→  assistant?: AiAssistantConfig,
   324→  ctx?: EngineContext
   325→): Promise<EngineResult> {
   326→  const state: EngineState = prevState
   327→    ? {
   328→        currentNodeId: prevState.currentNodeId ?? null,
   329→        waiting: prevState.waiting ?? 'none',
   330→        vars: { ...prevState.vars },
   331→        history: [...(prevState.history ?? [])].slice(-MEMORY_TAIL * 2),
   332→      }
   333→    : emptyState();
   334→
   335→  const messages: EngineMessage[] = [];
   336→  let needsOperator = state.vars['__operator'] === 'true';
   337→  let lastInput = input ?? '';
   338→  let current: FlowNode | null = null;
   339→
   340→  const assistantEnabled = assistant?.enabled === true;
   341→
   342→  /**
   343→   * Ответ ИИ-ассистента на свободный вопрос (вне сценария).
   344→   * После ответа возвращает пользователя в меню, чтобы навигация
   345→   * по сценарию продолжала работать.
   346→   */
   347→  const assistantFallback = async (userText: string, menuNode?: FlowNode): Promise<void> => {
   348→    let answer: string | null = null;
   349→    try {
   350→      const raw = await runAI(
   351→        { prompt: assistant?.prompt, knowledge: assistant?.knowledge, useMemory: true },
   352→        state,
   353→        userText,
   354→        ASSISTANT_SYSTEM_RULES
   355→      );
   356→      answer = raw && raw.trim().length > 0 ? raw : null;
   357→    } catch (err) {
   358→      console.error('[flow-engine] AI assistant error:', err);
   359→    }
   360→
   361→    // ИИ недоступен — вежливо сообщаем, пользователь может повторить вопрос
   362→    if (!answer) {
   363→      const errText =
   364→        'Извините, сервис ИИ временно недоступен. Попробуйте задать вопрос ещё раз через минуту.';
   365→      messages.push({ text: errText, nodeId: 'assistant' });
   366→      state.history.push({ role: 'bot', text: errText });
   367→      state.waiting = 'none';
   368→      state.currentNodeId = null;
   369→      return;
   370→    }
   371→
   372→    // Пользователь просит живого оператора — передаём диалог
   373→    if (answer.toUpperCase().includes('OPERATOR_REQUEST')) {
   374→      state.vars['__operator'] = 'true';
   375→      needsOperator = true;
   376→      const text = 'Соединяю вас с живым оператором, оставайтесь на линии 🙌';
   377→      messages.push({ text, nodeId: 'assistant' });
   378→      state.history.push({ role: 'bot', text });
   379→      state.waiting = 'none';
   380→      state.currentNodeId = null;
   381→      return;
   382→    }
   383→
   384→    messages.push({ text: answer, nodeId: 'assistant' });
   385→    state.history.push({ role: 'bot', text: answer });
   386→
   387→    // Возвращаем меню, чтобы можно было продолжить навигацию по сценарию
   388→    const menu = menuNode ?? findMenuNode(flow, state.vars['__last_menu']);
   389→    if (menu) {
   390→      if (assistant?.reaskMenu !== false) {
   391→        const menuText = interpolate(menu.data.text, state.vars) || 'Выберите вариант:';
   392→        messages.push({
   393→          text: menuText,
   394→          nodeId: menu.id,
   395→          buttons: menu.data.buttons ?? [],
   396→        });
   397→        state.history.push({ role: 'bot', text: menuText });
   398→      }
   399→      state.waiting = 'buttons';
   400→      state.currentNodeId = menu.id;
   401→    } else {
   402→      state.waiting = 'none';
   403→      state.currentNodeId = null;
   404→    }
   405→  };
   406→
   407→  // 1. Возобновление: если движок ждал ввода/кнопки
   408→  if (input && state.waiting !== 'none' && state.currentNodeId) {
   409→    const node = findNode(flow, state.currentNodeId);
   410→    if (node?.type === 'question') {
   411→      // Постоянная кнопка «🏠 Главное меню» — работает и вместо ответа на вопрос
   412→      if (isMainMenuCommand(input)) {
   413→        state.history.push({ role: 'user', text: input });
   414→        state.waiting = 'none';
   415→        lastInput = '';
   416→        current = findStart(flow);
   417→        if (current && current.type === 'start') {
   418→          current = nextNode(flow, current.id);
   419→        }
   420→      } else if (isSkipWord(input) && node.data.validate) {
   421→        // «пропустить» — принимаем ответ без проверки (пустая переменная)
   422→        const v = node.data.variable?.trim() || 'answer';
   423→        if (!state.vars[v]) state.vars[v] = '';
   424→        state.history.push({ role: 'user', text: input });
   425→        state.waiting = 'none';
   426→        lastInput = '';
   427→        current = nextNode(flow, node.id);
   428→      } else {
   429→        // Автопроверка ответа (телефон/адрес): при ошибке переспрашиваем
   430→        const err = await validateAnswer(node.data, input, state.vars);
   431→        if (err) {
   432→          state.history.push({ role: 'user', text: input });
   433→          state.history.push({ role: 'bot', text: err });
   434→          messages.push({ text: err, nodeId: node.id });
   435→          state.waiting = 'input';
   436→          state.currentNodeId = node.id;
   437→          return { messages, state, needsOperator };
   438→        }
   439→        const v = node.data.variable?.trim() || 'answer';
   440→        state.vars[v] = input;
   441→        state.history.push({ role: 'user', text: input });
   442→        state.waiting = 'none';
   443→        lastInput = '';
   444→        current = nextNode(flow, node.id);
   445→      }
   446→    } else if (node?.type === 'buttons') {
   447→      state.history.push({ role: 'user', text: input });
   448→      const btn = (node.data.buttons ?? []).find(
   449→        (b) => b.text.trim().toLowerCase() === input.trim().toLowerCase()
   450→      );
   451→      state.waiting = 'none';
   452→      lastInput = '';
   453→      if (btn) {
   454→        // Выбор из меню можно сохранить в переменную (напр. объём контейнера)
   455→        if (node.data.saveSelection) {
   456→          state.vars[node.data.saveSelection] = btn.text;
   457→        }
   458→        current = nextNode(flow, node.id, btn.id);
   459→      } else if (isMainMenuCommand(input) || isStartCommand(input)) {
   460→        // «🏠 Главное меню» или /start в любом месте диалога — начать заново
   461→        current = findStart(flow);
   462→        if (current && current.type === 'start') {
   463→          current = nextNode(flow, current.id);
   464→        }
   465→      } else if (assistantEnabled) {
   466→        // Свободный текст вместо кнопки — отвечает ИИ-ассистент,
   467→        // затем пользователю снова показывается это же меню
   468→        await assistantFallback(input, node);
   469→        return { messages, state, needsOperator };
   470→      } else {
   471→        // Нет совпадения — переспрашиваем
   472→        const text = interpolate(node.data.text, state.vars) || 'Выберите вариант:';
   473→        messages.push({ text, nodeId: node.id, buttons: node.data.buttons ?? [] });
   474→        state.waiting = 'buttons';
   475→        state.currentNodeId = node.id;
   476→        state.history.push({ role: 'bot', text });
   477→        return { messages, state, needsOperator };
   478→      }
   479→    } else {
   480→      state.waiting = 'none';
   481→      current = findStart(flow);
   482→    }
   483→  }
   484→
   485→  // 2. Старт нового диалога (или продолжение после ожидания)
   486→  if (!current) {
   487→    if (state.waiting !== 'none') {
   488→      // Ждём ввода, но его не пришло — ничего не делаем
   489→      return { messages, state, needsOperator };
   490→    }
   491→    if (input) {
   492→      state.history.push({ role: 'user', text: input });
   493→    }
   494→    // Диалог уже передан оператору — бот молчит, пишет только человек-оператор
   495→    if (needsOperator) {
   496→      return { messages, state, needsOperator };
   497→    }
   498→    // ИИ-ассистент: свободный вопрос пользователя (в т.ч. первое сообщение);
   499→    // /start всегда запускает сценарий с приветствия — детерминированный вход
   500→    if (assistantEnabled && input && !isStartCommand(input)) {
   501→      await assistantFallback(input);
   502→      return { messages, state, needsOperator };
   503→    }
   504→    current = findStart(flow);
   505→    if (current && current.type === 'start') {
   506→      current = nextNode(flow, current.id);
   507→    }
   508→  }
   509→
   510→  // 3. Главный цикл traversal'а
   511→  let steps = 0;
   512→  while (current && steps < MAX_STEPS) {
   513→    steps++;
   514→    const node = current;
   515→    switch (node.type) {
   516→      case 'start': {
   517→        current = nextNode(flow, node.id);
   518→        break;
   519→      }
   520→      case 'message': {
   521→        // Узел с флагом «Создать заявку»: номер попадает в {{order.number}}
   522→        if (node.data.createOrder && ctx?.botId) {
   523→          try {
   524→            const cfg = node.data.createOrder;
   525→            // Значение переменной; «{{city}}, {{street}}» — подстановка шаблона
   526→            const pick = (v?: string) => {
   527→              if (!v || !v.trim()) return '';
   528→              if (v.includes('{{')) return interpolate(v, state.vars).trim();
   529→              return (state.vars[v.trim()] ?? '').trim();
   530→            };
   531→            const order = await createOrder({
   532→              botId: ctx.botId,
   533→              conversationId: ctx.conversationId,
   534→              externalUserId: ctx.externalUserId ?? null,
   535→              type: cfg.type ?? 'waste',
   536→              clientName: pick(cfg.nameVar) || null,
   537→              phone: pick(cfg.phoneVar) || null,
   538→              city: pick(cfg.cityVar) || null,
   539→              address: pick(cfg.addressVar) || null,
   540→              size: pick(cfg.sizeVar) || null,
   541→              wishDate: pick(cfg.dateVar) || null,
   542→              comment: cfg.comment ? interpolate(cfg.comment, state.vars) : null,
   543→            });
   544→            state.vars['order.number'] = String(order.number);
   545→            state.vars['order.id'] = order.id;
   546→            state.vars['order.type'] = order.type;
   547→            console.log(`[flow-engine] заявка №${order.number} создана (${order.type})`);
   548→          } catch (err) {
   549→            console.error('[flow-engine] createOrder error:', err);
   550→            state.vars['order.number'] = '—';
   551→          }
   552→        }
   553→        const text = interpolate(node.data.text, state.vars, lastInput || undefined);
   554→        if (text.trim()) {
   555→          messages.push({ text, nodeId: node.id });
   556→          state.history.push({ role: 'bot', text });
   557→        }
   558→        lastInput = '';
   559→        current = nextNode(flow, node.id);
   560→        break;
   561→      }
   562→      case 'question': {
   563→        const text = interpolate(node.data.text, state.vars, lastInput || undefined);
   564→        if (text.trim()) {
   565→          messages.push({ text, nodeId: node.id });
   566→          state.history.push({ role: 'bot', text });
   567→        }
   568→        state.waiting = 'input';
   569→        state.currentNodeId = node.id;
   570→        current = null;
   571→        break;
   572→      }
   573→      case 'buttons': {
   574→        const text = interpolate(node.data.text, state.vars, lastInput || undefined);
   575→        if (text.trim()) {
   576→          messages.push({
   577→            text,
   578→            nodeId: node.id,
   579→            buttons: node.data.buttons ?? [],
   580→          });
   581→          state.history.push({ role: 'bot', text });
   582→        }
   583→        // Запоминаем последнее меню — ассистент вернёт к нему после ответа
   584→        state.vars['__last_menu'] = node.id;
   585→        state.waiting = 'buttons';
   586→        state.currentNodeId = node.id;
   587→        current = null;
   588→        break;
   589→      }
   590→      case 'condition': {
   591→        const ok = evalCondition(node.data.condition, state.vars);
   592→        current = nextNode(flow, node.id, ok ? 'yes' : 'no');
   593→        break;
   594→      }
   595→      case 'ai': {
   596→        try {
   597→          // AI-узел дополняется общей базой знаний ассистента бота
   598→          const d = { ...node.data };
   599→          if (assistantEnabled && d.useBotKnowledge !== false && assistant?.knowledge?.trim()) {
   600→            d.knowledge = d.knowledge?.trim()
   601→              ? `${d.knowledge.trim()}\n\n${assistant.knowledge.trim()}`
   602→              : assistant.knowledge;
   603→          }
   604→          const raw = await runAI(d, state, lastInput);
   605→          const text = interpolate(raw, state.vars);
   606→          messages.push({ text, nodeId: node.id });
   607→          state.history.push({ role: 'bot', text });
   608→        } catch (err) {
   609→          console.error('[flow-engine] AI node error:', err);
   610→          const fallback =
   611→            'Извините, сервис ИИ временно недоступен. Попробуйте ещё раз через минуту.';
   612→          messages.push({ text: fallback, nodeId: node.id });
   613→          state.history.push({ role: 'bot', text: fallback });
   614→        }
   615→        lastInput = '';
   616→        current = nextNode(flow, node.id);
   617→        break;
   618→      }
   619→      case 'http': {
   620→        const url = interpolate(node.data.url, state.vars, lastInput || undefined);
   621→        if (url.trim()) {
   622→          try {
   623→            const method = (node.data.method || 'GET').toUpperCase();
   624→            const init: RequestInit = { method };
   625→            if (method !== 'GET' && node.data.body) {
   626→              init.body = interpolate(node.data.body, state.vars, lastInput || undefined);
   627→              init.headers = { 'Content-Type': 'application/json' };
   628→            }
   629→            const res = await fetch(url, { ...init, signal: AbortSignal.timeout(10000) });
   630→            state.vars['__http_status'] = String(res.status);
   631→          } catch {
   632→            state.vars['__http_status'] = 'error';
   633→          }
   634→        }
   635→        lastInput = '';
   636→        current = nextNode(flow, node.id);
   637→        break;
   638→      }
   639→      case 'delay': {
   640→        const sec = Math.min(Math.max(node.data.seconds ?? 1, 0), 3);
   641→        if (sec > 0) await new Promise((r) => setTimeout(r, sec * 1000));
   642→        current = nextNode(flow, node.id);
   643→        break;
   644→      }
   645→      case 'handoff': {
   646→        state.vars['__operator'] = 'true';
   647→        needsOperator = true;
   648→        const text =
   649→          interpolate(node.data.text, state.vars, lastInput || undefined) ||
   650→          'Соединяю вас с живым оператором, оставайтесь на линии 🙌';
   651→        if (text.trim()) {
   652→          messages.push({ text, nodeId: node.id });
   653→          state.history.push({ role: 'bot', text });
   654→        }
   655→        lastInput = '';
   656→        current = nextNode(flow, node.id);
   657→        break;
   658→      }
   659→      case 'end':
   660→      default: {
   661→        state.waiting = 'none';
   662→        state.currentNodeId = null;
   663→        current = null;
   664→        break;
   665→      }
   666→    }
   667→  }
   668→
   669→  if (steps >= MAX_STEPS) {
   670→    state.waiting = 'none';
   671→    state.currentNodeId = null;
   672→  }
   673→
   674→  return { messages, state, needsOperator };
   675→}
