   142→   142→/** Подстановка переменных {{name}} в текст */
   143→   143→export function interpolate(
   144→   144→  text: string | undefined,
   145→   145→  vars: Record<string, string>,
   146→   146→  input?: string
   147→   147→): string {
   148→   148→  if (!text) return '';
   149→   149→  let out = text.replace(/\{\{\s*([\w.]+)\s*\}\}/g, (_m, key: string) => {
   150→   150→    if (key === 'input' && input !== undefined) return input;
   151→   151→    const v = vars[key];
   152→   152→    return v !== undefined ? v : '';
   153→   153→  });
   154→   154→  // Приглаживаем пустые подстановки: «Завтра, » → «Завтра», «Боровичи,» → «Боровичи»
   155→   155→  out = out.replace(/,\s*,+/g, ', ');
   156→   156→  out = out.replace(/,\s*$/gm, '');
   157→   157→  return out;
   158→   158→}

   188→   188→export function findMainMenu(flow: Flow): FlowNode | null {
   189→   189→  const start = findStart(flow);
   190→   190→  if (start) {
   191→   191→    const visited = new Set<string>([start.id]);
   192→   192→    const queue: string[] = [start.id];
   193→   193→    while (queue.length) {
   194→   194→      const id = queue.shift() as string;
   195→   195→      const node = findNode(flow, id);
   196→   196→      if (!node) continue;
   197→   197→      if (node.type === 'buttons') return node;
   198→   198→      for (const e of flow.edges) {
   199→   199→        if (e.source === id && !visited.has(e.target)) {
   200→   200→          visited.add(e.target);
   201→   201→          queue.push(e.target);
   202→   202→        }
   203→   203→      }
   204→   204→    }
   205→   205→  }
   206→   206→  return flow.nodes.find((n) => n.type === 'buttons') ?? null;
   207→   207→}

   217→   217→function evalCondition(
   218→   218→  cond: FlowNode['data']['condition'],
   219→   219→  vars: Record<string, string>
   220→   220→): boolean {
   221→   221→  if (!cond) return false;
   222→   222→  const leftRaw = (cond.left ?? '').trim();
   223→   223→  const left = leftRaw.includes('{{')
   224→   224→    ? interpolate(leftRaw, vars)
   225→   225→    : (vars[leftRaw] ?? leftRaw);
   226→   226→  const rightRaw = cond.right ?? '';
   227→   227→  const right = rightRaw.includes('{{') ? interpolate(rightRaw, vars) : rightRaw;
   228→   228→  const op: ConditionOp = cond.op ?? 'eq';
   229→   229→  switch (op) {
   230→   230→    case 'eq':
   231→   231→      return left.trim().toLowerCase() === right.trim().toLowerCase();
   232→   232→    case 'neq':
   233→   233→      return left.trim().toLowerCase() !== right.trim().toLowerCase();
   234→   234→    case 'contains':
   235→   235→      return left.toLowerCase().includes(right.toLowerCase());
   236→   236→    case 'not_contains':
   237→   237→      return !left.toLowerCase().includes(right.toLowerCase());
   238→   238→    case 'gt':
   239→   239→      return parseFloat(left) > parseFloat(right);
   240→   240→    case 'lt':
   241→   241→      return parseFloat(left) < parseFloat(right);
   242→   242→    case 'empty':
   243→   243→      return left.trim() === '';
   244→   244→    case 'not_empty':
   245→   245→      return left.trim() !== '';
   246→   246→    default:
   247→   247→      return false;
   248→   248→  }
   249→   249→}

   259→   259→async function runAI(
   260→   260→  data: AiAskData,
   261→   261→  state: EngineState,
   262→   262→  lastInput: string,
   263→   263→  extraSystem?: string
   264→   264→): Promise<string> {
   265→   265→  const zai = await getZai();
   266→   266→
   267→   267→  const sysParts: string[] = [];
   268→   268→  sysParts.push(
   269→   269→    data.prompt?.trim() ||
   270→   270→      'Ты — вежливый бот техподдержки. Отвечай кратко, по делу и на языке пользователя.'
   271→   271→  );
   272→   272→  if (extraSystem?.trim()) {
   273→   273→    sysParts.push(extraSystem.trim());
   274→   274→  }
   275→   275→  if (data.knowledge?.trim()) {
   276→   276→    sysParts.push(`База знаний компании:\n${data.knowledge.trim()}`);
   277→   277→  }
   278→   278→  const varLines = Object.entries(state.vars)
   279→   279→    .filter(([k]) => !k.startsWith('__'))
   280→   280→    .map(([k, v]) => `${k}: ${v}`);
   281→   281→  if (varLines.length) {
   282→   282→    sysParts.push(`Известные данные о пользователе:\n${varLines.join('\n')}`);
   283→   283→  }
   284→   284→
   285→   285→  const messages: { role: 'assistant' | 'user'; content: string }[] = [
   286→   286→    { role: 'assistant', content: sysParts.join('\n\n') },
   287→   287→  ];
   288→   288→
   289→   289→  if (data.useMemory !== false) {
   290→   290→    for (const h of state.history.slice(-MEMORY_TAIL)) {
   291→   291→      messages.push({
   292→   292→        role: h.role === 'user' ? 'user' : 'assistant',
   293→   293→        content: h.text,
   294→   294→      });
   295→   295→    }
   296→   296→  }
   297→   297→  if (lastInput) {
   298→   298→    messages.push({ role: 'user', content: lastInput });
   299→   299→  } else if (messages.length === 1) {
   300→   300→    messages.push({ role: 'user', content: 'Привет' });
   301→   301→  }
   302→   302→
   303→   303→  // IMP-BE21-02: SDK не принимает signal — таймаут через Promise.race (45 с).
   304→   304→  // На таймауте/ошибке выбрасываем исключение — существующие catch показывают
   305→   305→  // fallback-текст (AI-узел сценария, свободный чат, assistantFallback).
   306→   306→  let aiTimer: ReturnType<typeof setTimeout> | undefined;
   307→   307→  let completion: Awaited<ReturnType<typeof zai.chat.completions.create>>;
   308→   308→  try {
   309→   309→    completion = await Promise.race([
   310→   310→      zai.chat.completions.create({
   311→   311→        messages,
   312→   312→        thinking: { type: 'disabled' },
   313→   313→      }),
   314→   314→      new Promise<never>((_, reject) => {
   315→   315→        aiTimer = setTimeout(() => reject(new Error('AI timeout: ответ ИИ не получен за 45 с')), 45_000);
   316→   316→        // таймер не должен удерживать процесс при завершении
   317→   317→        if (typeof aiTimer.unref === 'function') aiTimer.unref();
   318→   318→      }),
   319→   319→    ]);
   320→   320→  } finally {
   321→   321→    // очищаем таймер, чтобы не течь (иначе гонка «ответ пришёл последним» держит handle)
   322→   322→    if (aiTimer) clearTimeout(aiTimer);
   323→   323→  }
   324→   324→  const content = completion.choices[0]?.message?.content;
   325→   325→  return content && content.trim().length > 0
   326→   326→    ? content
   327→   327→    : 'Извините, я не смог сформулировать ответ. Попробуйте перефразировать вопрос.';
   328→   328→}

   352→   352→export async function runEngine(
   353→   353→  flow: Flow,
   354→   354→  input: string | null,
   355→   355→  prevState: EngineState | null,
   356→   356→  assistant?: AiAssistantConfig,
   357→   357→  ctx?: EngineContext
   358→   358→): Promise<EngineResult> {
   359→   359→  const state: EngineState = prevState
   360→   360→    ? {
   361→   361→        currentNodeId: prevState.currentNodeId ?? null,
   362→   362→        waiting: prevState.waiting ?? 'none',
   363→   363→        vars: { ...prevState.vars },
   364→   364→        history: [...(prevState.history ?? [])].slice(-MEMORY_TAIL * 2),
   365→   365→      }
   366→   366→    : emptyState();
   367→   367→
   368→   368→  const messages: EngineMessage[] = [];
   369→   369→  let needsOperator = state.vars['__operator'] === 'true';
   370→   370→  let lastInput = input ?? '';
   371→   371→  let current: FlowNode | null = null;
   372→   372→
   373→   373→  const assistantEnabled = assistant?.enabled === true;
   374→   374→
   375→   375→  /**
   376→   376→   * Ответ ИИ-ассистента на свободный вопрос (вне сценария).
   377→   377→   * После ответа возвращает пользователя в меню, чтобы навигация
   378→   378→   * по сценарию продолжала работать.
   379→   379→   */
   380→   380→  const assistantFallback = async (userText: string, menuNode?: FlowNode): Promise<void> => {
   381→   381→    let answer: string | null = null;
   382→   382→    try {
   383→   383→      const raw = await runAI(
   384→   384→        { prompt: assistant?.prompt, knowledge: assistant?.knowledge, useMemory: true },
   385→   385→        state,
   386→   386→        userText,
   387→   387→        ASSISTANT_SYSTEM_RULES
   388→   388→      );
   389→   389→      answer = raw && raw.trim().length > 0 ? raw : null;
   390→   390→    } catch (err) {
   391→   391→      console.error('[flow-engine] AI assistant error:', err);
   392→   392→    }
   393→   393→
   394→   394→    // ИИ недоступен — вежливо сообщаем, пользователь может повторить вопрос
   395→   395→    if (!answer) {
   396→   396→      const errText =
   397→   397→        'Извините, сервис ИИ временно недоступен. Попробуйте задать вопрос ещё раз через минуту.';
   398→   398→      messages.push({ text: errText, nodeId: 'assistant' });
   399→   399→      state.history.push({ role: 'bot', text: errText });
   400→   400→      state.waiting = 'none';
   401→   401→      state.currentNodeId = null;
   402→   402→      return;
   403→   403→    }
   404→   404→
   405→   405→    // Пользователь просит живого оператора — передаём диалог
   406→   406→    if (answer.toUpperCase().includes('OPERATOR_REQUEST')) {
   407→   407→      state.vars['__operator'] = 'true';
   408→   408→      needsOperator = true;
   409→   409→      const text = 'Соединяю вас с живым оператором, оставайтесь на линии 🙌';
   410→   410→      messages.push({ text, nodeId: 'assistant' });
   411→   411→      state.history.push({ role: 'bot', text });
   412→   412→      state.waiting = 'none';
   413→   413→      state.currentNodeId = null;
   414→   414→      return;
   415→   415→    }
   416→   416→
   417→   417→    messages.push({ text: answer, nodeId: 'assistant' });
   418→   418→    state.history.push({ role: 'bot', text: answer });
   419→   419→
   420→   420→    // Возвращаем меню, чтобы можно было продолжить навигацию по сценарию
   421→   421→    const menu = menuNode ?? findMenuNode(flow, state.vars['__last_menu']);
   422→   422→    if (menu) {
   423→   423→      if (assistant?.reaskMenu !== false) {
   424→   424→        const menuText = interpolate(menu.data.text, state.vars) || 'Выберите вариант:';
   425→   425→        messages.push({
   426→   426→          text: menuText,
   427→   427→          nodeId: menu.id,
   428→   428→          buttons: menu.data.buttons ?? [],
   429→   429→        });
   430→   430→        state.history.push({ role: 'bot', text: menuText });
   431→   431→      }
   432→   432→      state.waiting = 'buttons';
   433→   433→      state.currentNodeId = menu.id;
   434→   434→    } else {
   435→   435→      state.waiting = 'none';
   436→   436→      state.currentNodeId = null;
   437→   437→    }
   438→   438→  };
   439→   439→
   440→   440→  // 1. Возобновление: если движок ждал ввода/кнопки
   441→   441→  if (input && state.waiting !== 'none' && state.currentNodeId) {
   442→   442→    const node = findNode(flow, state.currentNodeId);
   443→   443→    if (node?.type === 'question') {
   444→   444→      // Постоянная кнопка «🏠 Главное меню» — работает и вместо ответа на вопрос
   445→   445→      if (isMainMenuCommand(input)) {
   446→   446→        state.history.push({ role: 'user', text: input });
   447→   447→        state.waiting = 'none';
   448→   448→        lastInput = '';
   449→   449→        current = findStart(flow);
   450→   450→        if (current && current.type === 'start') {
   451→   451→          current = nextNode(flow, current.id);
   452→   452→        }
   453→   453→      } else if (isSkipWord(input) && node.data.validate) {
   454→   454→        // «пропустить» — принимаем ответ без проверки (пустая переменная)
   455→   455→        const v = node.data.variable?.trim() || 'answer';
   456→   456→        if (!state.vars[v]) state.vars[v] = '';
   457→   457→        state.history.push({ role: 'user', text: input });
   458→   458→        state.waiting = 'none';
   459→   459→        lastInput = '';
   460→   460→        current = nextNode(flow, node.id);
   461→   461→      } else {
   462→   462→        // Автопроверка ответа (телефон/адрес): при ошибке переспрашиваем
   463→   463→        const err = await validateAnswer(node.data, input, state.vars);
   464→   464→        if (err) {
   465→   465→          state.history.push({ role: 'user', text: input });
   466→   466→          state.history.push({ role: 'bot', text: err });
   467→   467→          messages.push({ text: err, nodeId: node.id });
   468→   468→          state.waiting = 'input';
   469→   469→          state.currentNodeId = node.id;
   470→   470→          return { messages, state, needsOperator };
   471→   471→        }
   472→   472→        const v = node.data.variable?.trim() || 'answer';
   473→   473→        state.vars[v] = input;
   474→   474→        state.history.push({ role: 'user', text: input });
   475→   475→        state.waiting = 'none';
   476→   476→        lastInput = '';
   477→   477→        current = nextNode(flow, node.id);
   478→   478→      }
   479→   479→    } else if (node?.type === 'buttons') {
   480→   480→      state.history.push({ role: 'user', text: input });
   481→   481→      const btn = (node.data.buttons ?? []).find(
   482→   482→        (b) => b.text.trim().toLowerCase() === input.trim().toLowerCase()
   483→   483→      );
   484→   484→      state.waiting = 'none';
   485→   485→      lastInput = '';
   486→   486→      if (btn) {
   487→   487→        // Выбор из меню можно сохранить в переменную (напр. объём контейнера)
   488→   488→        if (node.data.saveSelection) {
   489→   489→          state.vars[node.data.saveSelection] = btn.text;
   490→   490→        }
   491→   491→        current = nextNode(flow, node.id, btn.id);
   492→   492→      } else if (isMainMenuCommand(input) || isStartCommand(input)) {
   493→   493→        // «🏠 Главное меню» или /start в любом месте диалога — начать заново
   494→   494→        current = findStart(flow);
   495→   495→        if (current && current.type === 'start') {
   496→   496→          current = nextNode(flow, current.id);
   497→   497→        }
   498→   498→      } else if (assistantEnabled) {
   499→   499→        // Свободный текст вместо кнопки — отвечает ИИ-ассистент,
   500→   500→        // затем пользователю снова показывается это же меню
   501→   501→        await assistantFallback(input, node);
   502→   502→        return { messages, state, needsOperator };
   503→   503→      } else {
   504→   504→        // Нет совпадения — переспрашиваем
   505→   505→        const text = interpolate(node.data.text, state.vars) || 'Выберите вариант:';
   506→   506→        messages.push({ text, nodeId: node.id, buttons: node.data.buttons ?? [] });
   507→   507→        state.waiting = 'buttons';
   508→   508→        state.currentNodeId = node.id;
   509→   509→        state.history.push({ role: 'bot', text });
   510→   510→        return { messages, state, needsOperator };
   511→   511→      }
   512→   512→    } else {
   513→   513→      state.waiting = 'none';
   514→   514→      current = findStart(flow);
   515→   515→    }
   516→   516→  }
   517→   517→
   518→   518→  // 2. Старт нового диалога (или продолжение после ожидания)
   519→   519→  if (!current) {
   520→   520→    if (state.waiting !== 'none') {
   521→   521→      // Ждём ввода, но его не пришло — ничего не делаем
   522→   522→      return { messages, state, needsOperator };
   523→   523→    }
   524→   524→    if (input) {
   525→   525→      state.history.push({ role: 'user', text: input });
   526→   526→    }
   527→   527→    // Диалог уже передан оператору — бот молчит, пишет только человек-оператор
   528→   528→    if (needsOperator) {
   529→   529→      return { messages, state, needsOperator };
   530→   530→    }
   531→   531→    // ИИ-ассистент: свободный вопрос пользователя (в т.ч. первое сообщение);
   532→   532→    // /start всегда запускает сценарий с приветствия — детерминированный вход
   533→   533→    if (assistantEnabled && input && !isStartCommand(input)) {
   534→   534→      await assistantFallback(input);
   535→   535→      return { messages, state, needsOperator };
   536→   536→    }
   537→   537→    current = findStart(flow);
   538→   538→    if (current && current.type === 'start') {
   539→   539→      current = nextNode(flow, current.id);
   540→   540→    }
   541→   541→  }
   542→   542→
   543→   543→  // 3. Главный цикл traversal'а
   544→   544→  let steps = 0;
   545→   545→  while (current && steps < MAX_STEPS) {
   546→   546→    steps++;
   547→   547→    const node = current;
   548→   548→    switch (node.type) {
   549→   549→      case 'start': {
   550→   550→        current = nextNode(flow, node.id);
   551→   551→        break;
   552→   552→      }
   553→   553→      case 'message': {
   554→   554→        // Узел с флагом «Создать заявку»: номер попадает в {{order.number}}
   555→   555→        if (node.data.createOrder && ctx?.botId) {
   556→   556→          try {
   557→   557→            const cfg = node.data.createOrder;
   558→   558→            // Значение переменной; «{{city}}, {{street}}» — подстановка шаблона
   559→   559→            const pick = (v?: string) => {
   560→   560→              if (!v || !v.trim()) return '';
   561→   561→              if (v.includes('{{')) return interpolate(v, state.vars).trim();
   562→   562→              return (state.vars[v.trim()] ?? '').trim();
   563→   563→            };
   564→   564→            const order = await createOrder({
   565→   565→              botId: ctx.botId,
   566→   566→              conversationId: ctx.conversationId,
   567→   567→              externalUserId: ctx.externalUserId ?? null,
   568→   568→              type: cfg.type ?? 'waste',
   569→   569→              clientName: pick(cfg.nameVar) || null,
   570→   570→              phone: pick(cfg.phoneVar) || null,
   571→   571→              city: pick(cfg.cityVar) || null,
   572→   572→              address: pick(cfg.addressVar) || null,
   573→   573→              size: pick(cfg.sizeVar) || null,
   574→   574→              wishDate: pick(cfg.dateVar) || null,
   575→   575→              comment: cfg.comment ? interpolate(cfg.comment, state.vars) : null,
   576→   576→            });
   577→   577→            state.vars['order.number'] = String(order.number);
   578→   578→            state.vars['order.id'] = order.id;
   579→   579→            state.vars['order.type'] = order.type;
   580→   580→            console.log(`[flow-engine] заявка №${order.number} создана (${order.type})`);
   581→   581→          } catch (err) {
   582→   582→            console.error('[flow-engine] createOrder error:', err);
   583→   583→            state.vars['order.number'] = '—';
   584→   584→          }
   585→   585→        }
   586→   586→        const text = interpolate(node.data.text, state.vars, lastInput || undefined);
   587→   587→        if (text.trim()) {
   588→   588→          messages.push({ text, nodeId: node.id });
   589→   589→          state.history.push({ role: 'bot', text });
   590→   590→        }
   591→   591→        lastInput = '';
   592→   592→        current = nextNode(flow, node.id);
   593→   593→        break;
   594→   594→      }
   595→   595→      case 'question': {
   596→   596→        const text = interpolate(node.data.text, state.vars, lastInput || undefined);
   597→   597→        if (text.trim()) {
   598→   598→          messages.push({ text, nodeId: node.id });
   599→   599→          state.history.push({ role: 'bot', text });
   600→   600→        }
   601→   601→        state.waiting = 'input';
   602→   602→        state.currentNodeId = node.id;
   603→   603→        current = null;
   604→   604→        break;
   605→   605→      }
   606→   606→      case 'buttons': {
   607→   607→        const text = interpolate(node.data.text, state.vars, lastInput || undefined);
   608→   608→        if (text.trim()) {
   609→   609→          messages.push({
   610→   610→            text,
   611→   611→            nodeId: node.id,
   612→   612→            buttons: node.data.buttons ?? [],
   613→   613→          });
   614→   614→          state.history.push({ role: 'bot', text });
   615→   615→        }
   616→   616→        // Запоминаем последнее меню — ассистент вернёт к нему после ответа
   617→   617→        state.vars['__last_menu'] = node.id;
   618→   618→        state.waiting = 'buttons';
   619→   619→        state.currentNodeId = node.id;
   620→   620→        current = null;
   621→   621→        break;
   622→   622→      }
   623→   623→      case 'condition': {
   624→   624→        const ok = evalCondition(node.data.condition, state.vars);
   625→   625→        current = nextNode(flow, node.id, ok ? 'yes' : 'no');
   626→   626→        break;
   627→   627→      }
   628→   628→      case 'ai': {
   629→   629→        try {
   630→   630→          // AI-узел дополняется общей базой знаний ассистента бота
   631→   631→          const d = { ...node.data };
   632→   632→          if (assistantEnabled && d.useBotKnowledge !== false && assistant?.knowledge?.trim()) {
   633→   633→            d.knowledge = d.knowledge?.trim()
   634→   634→              ? `${d.knowledge.trim()}\n\n${assistant.knowledge.trim()}`
   635→   635→              : assistant.knowledge;
   636→   636→          }
   637→   637→          const raw = await runAI(d, state, lastInput);
   638→   638→          const text = interpolate(raw, state.vars);
   639→   639→          messages.push({ text, nodeId: node.id });
   640→   640→          state.history.push({ role: 'bot', text });
   641→   641→        } catch (err) {
   642→   642→          console.error('[flow-engine] AI node error:', err);
   643→   643→          const fallback =
   644→   644→            'Извините, сервис ИИ временно недоступен. Попробуйте ещё раз через минуту.';
   645→   645→          messages.push({ text: fallback, nodeId: node.id });
   646→   646→          state.history.push({ role: 'bot', text: fallback });
   647→   647→        }
   648→   648→        lastInput = '';
   649→   649→        current = nextNode(flow, node.id);
   650→   650→        break;
   651→   651→      }
   652→   652→      case 'http': {
   653→   653→        const url = interpolate(node.data.url, state.vars, lastInput || undefined);
   654→   654→        if (url.trim()) {
   655→   655→          try {
   656→   656→            const method = (node.data.method || 'GET').toUpperCase();
   657→   657→            const init: RequestInit = { method };
   658→   658→            if (method !== 'GET' && node.data.body) {
   659→   659→              init.body = interpolate(node.data.body, state.vars, lastInput || undefined);
   660→   660→              init.headers = { 'Content-Type': 'application/json' };
   661→   661→            }
   662→   662→            const res = await fetch(url, { ...init, signal: AbortSignal.timeout(10000) });
   663→   663→            state.vars['__http_status'] = String(res.status);
   664→   664→          } catch {
   665→   665→            state.vars['__http_status'] = 'error';
   666→   666→          }
   667→   667→        }
   668→   668→        lastInput = '';
   669→   669→        current = nextNode(flow, node.id);
   670→   670→        break;
   671→   671→      }
   672→   672→      case 'delay': {
   673→   673→        const sec = Math.min(Math.max(node.data.seconds ?? 1, 0), 3);
   674→   674→        if (sec > 0) await new Promise((r) => setTimeout(r, sec * 1000));
   675→   675→        current = nextNode(flow, node.id);
   676→   676→        break;
   677→   677→      }
   678→   678→      case 'handoff': {
   679→   679→        state.vars['__operator'] = 'true';
   680→   680→        needsOperator = true;
   681→   681→        const text =
   682→   682→          interpolate(node.data.text, state.vars, lastInput || undefined) ||
   683→   683→          'Соединяю вас с живым оператором, оставайтесь на линии 🙌';
   684→   684→        if (text.trim()) {
   685→   685→          messages.push({ text, nodeId: node.id });
   686→   686→          state.history.push({ role: 'bot', text });
   687→   687→        }
   688→   688→        lastInput = '';
   689→   689→        current = nextNode(flow, node.id);
   690→   690→        break;
   691→   691→      }
   692→   692→      case 'end':
   693→   693→      default: {
   694→   694→        state.waiting = 'none';
   695→   695→        state.currentNodeId = null;
   696→   696→        current = null;
   697→   697→        break;
   698→   698→      }
   699→   699→    }
   700→   700→  }
   701→   701→
   702→   702→  if (steps >= MAX_STEPS) {
   703→   703→    state.waiting = 'none';
   704→   704→    state.currentNodeId = null;
   705→   705→  }
   706→   706→
   707→   707→  return { messages, state, needsOperator };
   708→   708→}
