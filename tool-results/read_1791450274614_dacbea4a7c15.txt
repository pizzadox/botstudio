   110→export default function ChannelsView({
   111→  bot,
   112→  onBack,
   113→  onOpenEditor,
   114→}: {
   115→  bot: { id: string; name: string; status: string };
   116→  onBack: () => void;
   117→  onOpenEditor: () => void;
   118→}) {
   119→  const { toast } = useToast();
   120→  const [channels, setChannels] = useState<ChannelItem[]>([]);
   121→  const [loading, setLoading] = useState(true);
   122→  const [addOpen, setAddOpen] = useState(false);
   123→  const [type, setType] = useState<ChannelItem['type']>('telegram');
   124→  const [title, setTitle] = useState('');
   125→  const [token, setToken] = useState('');
   126→  const [phone, setPhone] = useState('');
   127→  const [busy, setBusy] = useState(false);
   128→  // IMP-FE21-09: id канала, для которого идёт проверка — повторный клик невозможен
   129→  const [testBusyId, setTestBusyId] = useState<string | null>(null);
   130→  const [demoChannel, setDemoChannel] = useState<ChannelItem | null>(null);
   131→  const [origin, setOrigin] = useState('');
   132→  // Подтверждение удаления (AlertDialog): state-target + open={!!target}
   133→  const [deleteTarget, setDeleteTarget] = useState<ChannelItem | null>(null);
   134→
   135→  const load = useCallback(async () => {
   136→    try {
   137→      const d = await api<{ channels: ChannelItem[] }>(`/api/bots/${bot.id}/channels`);
   138→      setChannels(d.channels);
   139→    } catch {
   140→      toast({ title: 'Не удалось загрузить каналы', variant: 'destructive' });
   141→    } finally {
   142→      setLoading(false);
   143→    }
   144→  }, [bot.id, toast]);
   145→
   146→  useEffect(() => {
   147→    setOrigin(window.location.origin);
   148→    load();
   149→  }, [load]);
   150→
   151→  const createChannel = async () => {
   152→    setBusy(true);
   153→    try {
   154→      await api(`/api/bots/${bot.id}/channels`, {
   155→        method: 'POST',
   156→        body: JSON.stringify({ type, title, token, phone }),
   157→      });

   159→      setTitle('');
   160→      setToken('');
   161→      setPhone('');
   162→      toast({ title: 'Канал добавлен' });

   165→      toast({ title: 'Ошибка', description: e instanceof Error ? e.message : '', variant: 'destructive' });

   167→      setBusy(false);
   168→    }
   169→  };

   174→      await api(`/api/channels/${ch.id}`, { method: 'PUT', body: JSON.stringify({ active }) });
   175→    } catch {
   176→      load();
   177→    }
   178→  };
   179→
   180→  const testChannel = async (ch: ChannelItem) => {
   181→    if (testBusyId) return;
   182→    setTestBusyId(ch.id);
   183→    toast({ title: 'Проверяем подключение…' });
   184→    try {
   185→      const d = await api<{ ok: boolean; message: string }>(`/api/channels/${ch.id}/test`, {
   186→        method: 'POST',
   187→      });
   188→      toast({ title: d.message, variant: d.ok ? 'default' : 'destructive' });
   189→      load();
   190→    } catch {
   191→      toast({ title: 'Ошибка проверки', variant: 'destructive' });
   192→    } finally {
   193→      setTestBusyId(null);
   194→    }
   195→  };
   196→
   197→  const deleteChannel = async (ch: ChannelItem) => {
   198→    try {
   199→      await api(`/api/channels/${ch.id}`, { method: 'DELETE' });
   200→      toast({ title: 'Канал удалён' });
   201→      load();
   202→    } catch {
   203→      toast({ title: 'Не удалось удалить', variant: 'destructive' });
   204→    }
   205→  };
   206→
   207→  const copy = async (text: string) => {
   208→    try {
   209→      await navigator.clipboard.writeText(text);
   210→      toast({ title: 'Скопировано в буфер' });
   211→    } catch {
   212→      toast({ title: 'Не удалось скопировать', variant: 'destructive' });
   213→    }
   214→  };
   215→
   216→  // IMP-FE21-14: готовый URL приходит с сервера (ch.webhookUrl); клиентская
   217→  // сборка из secret — фолбэк на случай, если BE ещё не выкатил поле
   218→  const webhookUrl = (ch: ChannelItem) =>
   219→    ch.webhookUrl ?? `${origin}/api/webhook/${ch.type}/${ch.secret}`;
   220→
   221→  return (
   222→    <div className="flex-1 min-h-0 overflow-y-auto space-y-6 p-4 sm:p-6 lg:p-8">
   223→      <div className="flex flex-wrap items-center justify-between gap-4">
   224→        <div className="flex items-center gap-3">
   225→          <Button variant="ghost" size="icon" className="h-8 w-8" onClick={onBack} aria-label="Назад">
   226→            <ChevronLeft className="h-4 w-4" />
   227→          </Button>
   228→          <div>
   229→            <h1 className="flex items-center gap-2 text-lg font-bold tracking-tight sm:text-2xl">
   230→              <Plug className="h-6 w-6 text-primary" /> Каналы
   231→            </h1>
   232→            <p className="text-sm text-muted-foreground">Подключение бота «{bot.name}» к мессенджерам</p>
   233→          </div>
   234→        </div>
   235→        <Button onClick={() => { setType('telegram'); setAddOpen(true); }}>
   236→          <Plus className="h-4 w-4" /> Подключить канал
   237→        </Button>
   238→      </div>
   239→
   240→      {bot.status !== 'published' && (
   241→        <Alert className="border-amber-200 bg-amber-50 dark:border-amber-500/30 dark:bg-amber-500/10">
   242→          <Info className="h-4 w-4 text-amber-600 dark:text-amber-400" />
   243→          <AlertTitle className="text-amber-800 dark:text-amber-300">Бот ещё не опубликован</AlertTitle>
   244→          <AlertDescription className="text-amber-700 dark:text-amber-300/80">
   245→            Каналы сохранят настройки, но отвечать они начнут после публикации бота в{' '}
   246→            <button className="font-medium underline underline-offset-2" onClick={onOpenEditor}>
   247→              конструкторе
   248→            </button>
   249→            .
   250→          </AlertDescription>
   251→        </Alert>
   252→      )}
   253→
   254→      {loading ? (
   255→        <div className="space-y-4" role="status" aria-live="polite">
   256→          <p className="sr-only">Загрузка каналов…</p>
   257→          <div className="grid gap-4 md:grid-cols-2 xl:grid-cols-3">
   258→            {[0, 1, 2].map((i) => (
   259→              <div key={i} className="rounded-xl border bg-card p-4">
   260→                <div className="flex items-center gap-3">
   261→                  <Skeleton className="h-10 w-10 rounded-xl" />
   262→                  <div className="min-w-0 flex-1 space-y-2">
   263→                    <Skeleton className="h-4 w-2/3" />
   264→                    <Skeleton className="h-3 w-1/3" />
   265→                  </div>
   266→                </div>
   267→                <Skeleton className="mt-4 h-9 w-full" />
   268→                <div className="mt-3 flex gap-2">
   269→                  <Skeleton className="h-8 flex-1" />
   270→                  <Skeleton className="h-8 w-9" />
   271→                </div>
   272→              </div>
   273→            ))}
   274→          </div>
   275→        </div>
   276→      ) : channels.length === 0 ? (
   277→        <Card className="border-dashed">
   278→          <CardContent className="flex flex-col items-center gap-3 py-12 text-center">
   279→            {/* IMP-FE21-15: унификация пустых состояний — h-12 w-12 rounded-xl */}
   280→            <div className="flex h-12 w-12 items-center justify-center rounded-xl bg-primary/10 text-primary">
   281→              <Plug className="h-6 w-6" aria-hidden="true" />
   282→            </div>
   283→            <div className="font-semibold">Ни один канал не подключён</div>
   284→            <p className="max-w-md text-sm text-muted-foreground">
   285→              Подключите Telegram, WhatsApp, MAX — или создайте демо-чат, чтобы общаться с ботом
   286→              прямо на сайте.
   287→            </p>
   288→            <Button onClick={() => setAddOpen(true)}>
   289→              <Plus className="h-4 w-4" /> Подключить канал
   290→            </Button>
   291→          </CardContent>
   292→        </Card>
   293→      ) : (
   294→        <div className="grid gap-4 md:grid-cols-2 xl:grid-cols-3">
   295→          {channels.map((ch) => {
   296→            const meta = CHANNEL_META[ch.type];
   297→            const Icon = meta.icon;
   298→            return (
   299→              <Card key={ch.id}>
   300→                <CardHeader className="pb-3">
   301→                  <div className="flex min-w-0 items-start justify-between gap-2">
   302→                    <div className="flex items-center gap-3 min-w-0">
   303→                      <div className={cn('flex h-10 w-10 shrink-0 items-center justify-center rounded-xl', meta.color)}>
   304→                        <Icon className="h-5 w-5" />
   305→                      </div>
   306→                      <div className="min-w-0">
   307→                        <CardTitle className="truncate text-base">{ch.title}</CardTitle>
   308→                        <CardDescription>{meta.label}</CardDescription>
   309→                      </div>
   310→                    </div>
   311→                    <div className="flex items-center gap-2">
   312→                      <Switch
   313→                        checked={ch.active}
   314→                        onCheckedChange={(v) => toggleActive(ch, v)}
   315→                        aria-label={ch.active ? `Выключить канал «${ch.title}»` : `Включить канал «${ch.title}»`}
   316→                      />
   317→                    </div>
   318→                  </div>
   319→                </CardHeader>
   320→                <CardContent className="space-y-3">
   321→                  {ch.type === 'max' ? (
   322→                    <div className="rounded-lg border bg-muted/40 p-2.5 text-[11px] leading-relaxed text-muted-foreground">
   323→                      <span className="font-medium text-foreground">Вебхук не нужен.</span> Сообщения из MAX
   324→                      принимаются автоматически (long polling). Нажмите «Проверить» — студия свяжется с MAX и
   325→                      покажет имя бота.
   326→                    </div>
   327→                  ) : (
   328→                    <div className="flex items-center gap-2 rounded-lg border bg-muted/40 p-2">
   329→                      <code className="min-w-0 flex-1 truncate text-[11px] text-muted-foreground">
   330→                        {webhookUrl(ch)}
   331→                      </code>
   332→                      <Button
   333→                        variant="ghost"
   334→                        size="icon"
   335→                        className="h-7 w-7 shrink-0"
   336→                        onClick={() => copy(webhookUrl(ch))}
   337→                        aria-label="Скопировать URL вебхука"
   338→                      >
   339→                        <Copy className="h-3.5 w-3.5" />
   340→                      </Button>
   341→                    </div>
   342→                  )}
   343→                  {ch.type === 'web' && (
   344→                    <Button variant="outline" className="w-full" onClick={() => setDemoChannel(ch)}>
   345→                      <Bot className="h-4 w-4" /> Открыть демо-чат
   346→                    </Button>
   347→                  )}
   348→                  {ch.lastStatus && (
   349→                    <div role="status" aria-live="polite" className="min-w-0">
   350→                      <span
   351→                        className={cn(
   352→                          'inline-flex max-w-full items-center gap-1.5 rounded-full border px-2 py-0.5 text-[11px] font-medium',
   353→                          ch.lastStatus.startsWith('OK')
   354→                            ? 'border-emerald-200 bg-emerald-100 text-emerald-800 dark:border-emerald-500/30 dark:bg-emerald-500/15 dark:text-emerald-300'
   355→                            : ch.lastStatus.startsWith('Ошибка')
   356→                              ? 'border-rose-200 bg-rose-100 text-rose-800 dark:border-rose-500/30 dark:bg-rose-500/15 dark:text-rose-300'
   357→                              : 'border-slate-200 bg-slate-100 text-slate-700 dark:border-slate-500/30 dark:bg-slate-500/15 dark:text-slate-300'
   358→                        )}
   359→                      >
   360→                        <span
   361→                          className={cn(
   362→                            'h-1.5 w-1.5 shrink-0 rounded-full',
   363→                            ch.lastStatus.startsWith('OK')
   364→                              ? 'bg-emerald-500'
   365→                              : ch.lastStatus.startsWith('Ошибка')
   366→                                ? 'bg-rose-500'
   367→                                : 'bg-slate-400 dark:bg-slate-500'
   368→                          )}
   369→                          aria-hidden
   370→                        />
   371→                        <span className="sr-only">Статус: </span>
   372→                        <span className="truncate" title={ch.lastStatus}>
   373→                          {ch.lastStatus}
   374→                        </span>
   375→                      </span>
   376→                    </div>
   377→                  )}
   378→                  <div className="flex gap-2">
   379→                    {/* IMP-FE21-09: на время запроса кнопка заблокирована */}
   380→                    <Button
   381→                      size="sm"
   382→                      variant="outline"
   383→                      className="flex-1"
   384→                      onClick={() => testChannel(ch)}
   385→                      disabled={testBusyId === ch.id}
   386→                    >
   387→                      {testBusyId === ch.id ? (
   388→                        <Loader2 className="h-3.5 w-3.5 animate-spin" aria-hidden="true" />
   389→                      ) : (
   390→                        <Zap className="h-3.5 w-3.5" aria-hidden="true" />
   391→                      )}
   392→                      {testBusyId === ch.id ? 'Проверяем…' : 'Проверить'}
   393→                    </Button>
   394→                    <Button
   395→                      size="sm"
   396→                      variant="outline"
   397→                      className="text-destructive hover:bg-destructive/10 hover:text-destructive"
   398→                      onClick={() => setDeleteTarget(ch)}
   399→                      aria-label={`Удалить канал «${ch.title}»`}
   400→                    >
   401→                      <Trash2 className="h-3.5 w-3.5" />
   402→                    </Button>
   403→                  </div>
   404→                  <details className="group">
   405→                    <summary className="flex cursor-pointer list-none items-center gap-1 text-xs font-medium text-primary [&::-webkit-details-marker]:hidden">
   406→                      Как подключить
   407→                      <ChevronDown
   408→                        className="h-3.5 w-3.5 transition-transform group-open:rotate-180"
   409→                        aria-hidden
   410→                      />
   411→                    </summary>
   412→                    <p className="mt-2 whitespace-pre-line rounded-lg border bg-muted/40 p-2.5 text-[11px] leading-relaxed text-muted-foreground">
   413→                      {meta.hint}
   414→                    </p>
   415→                  </details>
   416→                </CardContent>
   417→              </Card>
   418→            );
   419→          })}
   420→        </div>
   421→      )}
   422→
   423→      {/* Интеграция MyTKO (mytko.ru — «Чистая логистика») */}
   424→      <MytkoCard botId={bot.id} />
   425→
   426→      {/* Диалог добавления канала */}
   427→      <Dialog open={addOpen} onOpenChange={setAddOpen}>
   428→        <DialogContent>
   429→          <DialogHeader>
   430→            <DialogTitle>Подключить канал</DialogTitle>
   431→            <DialogDescription>
   432→              Выберите мессенджер и вставьте данные доступа.
   433→            </DialogDescription>
   434→          </DialogHeader>
   435→          {/* IMP-FE21-10: Enter отправляет форму, preventDefault обязателен */}
   436→          <form
   437→            className="space-y-4"
   438→            onSubmit={(e) => {
   439→              e.preventDefault();
   440→              createChannel();
   441→            }}
   442→          >
   443→            <div className="space-y-4">
   444→              <div className="grid grid-cols-2 gap-2 sm:grid-cols-4">
   445→                {(Object.keys(CHANNEL_META) as ChannelItem['type'][]).map((t) => {
   446→                  const meta = CHANNEL_META[t];
   447→                  const Icon = meta.icon;
   448→                  return (
   449→                    <button
   450→                      key={t}
   451→                      type="button"
   452→                      onClick={() => setType(t)}
   453→                      aria-pressed={type === t}
   454→                      className={cn(
   455→                        'flex flex-col items-center gap-1.5 rounded-xl border p-3 text-xs font-medium transition-colors focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-primary/40',
   456→                        type === t
   457→                          ? 'border-primary bg-primary/5 shadow-sm ring-1 ring-primary/30'
   458→                          : 'hover:border-primary/40 hover:bg-muted/50'
   459→                      )}
   460→                    >
   461→                      <div className={cn('flex h-8 w-8 items-center justify-center rounded-lg', meta.color)}>
   462→                        <Icon className="h-4 w-4" />

   464→                      {meta.label}
   465→                    </button>
   466→                  );
   467→                })}
   468→              </div>
   469→              <div className="space-y-2">
   470→                <Label htmlFor="ch-title" className="text-xs font-medium">Название канала</Label>
   471→                <Input
   472→                  id="ch-title"
   473→                  autoFocus
   474→                  placeholder={`${CHANNEL_META[type].label} — основной`}
   475→                  value={title}
   476→                  onChange={(e) => setTitle(e.target.value)}
   477→                />
   478→              </div>
   479→              {type !== 'web' && (
   480→                <div className="space-y-2">
   481→                  <Label htmlFor="ch-token" className="text-xs font-medium">{CHANNEL_META[type].tokenLabel}</Label>

   483→                    id="ch-token"
   484→                    type="password"

   562→function DemoChat({ secret }: { secret: string }) {
   563→  const [messages, setMessages] = useState<
   564→    { id: string; role: string; text: string; buttons?: { id: string; text: string }[] }[]
   565→  >([]);
   566→  const [input, setInput] = useState('');
   567→  const [sending, setSending] = useState(false);
   568→  const convRef = useRef<string | null>(null);
   569→  const visitorRef = useRef<string>('');
   570→  const scrollRef = useRef<HTMLDivElement>(null);
   571→  const storageKey = `botstudio_demo_conv_${secret}`;
   572→  const visitorKey = 'botstudio_visitor';
   573→
   574→  // Стабильный id посетителя: один гость = одно обращение,
   575→  // даже если conversationId был потерян
   576→  useEffect(() => {
   577→    try {
   578→      let v = window.localStorage.getItem(visitorKey);
   579→      if (!v) {
   580→        v =
   581→          typeof crypto !== 'undefined' && 'randomUUID' in crypto
   582→            ? crypto.randomUUID().replace(/-/g, '')
   583→            : Math.random().toString(36).slice(2) + Date.now().toString(36);
   584→        window.localStorage.setItem(visitorKey, v);
   585→      }
   586→      visitorRef.current = v;
   587→    } catch {
   588→      visitorRef.current = Math.random().toString(36).slice(2) + Date.now().toString(36);
   589→    }
   590→  }, [visitorKey]);
   591→
   592→  // Восстановить диалог посетителя (как реальный виджет сайта).
   593→  // IMP-FE21-23: через api() — проверяется res.ok, ошибки не глотаются молча
   594→  useEffect(() => {
   595→    let cancelled = false;
   596→    void (async () => {
   597→      try {
   598→        const saved = window.localStorage.getItem(storageKey);
   599→        if (!saved) return;
   600→        convRef.current = saved;
   601→        const d = await api<{
   602→          messages?: { id: string; role: string; text: string; buttons?: { id: string; text: string }[] }[];
   603→        }>(`/api/webhook/demo/${secret}?conversationId=${saved}`);
   604→        if (!cancelled && d.messages?.length) setMessages(d.messages);
   605→      } catch {
   606→        /* диалог не восстановился — начнётся новый при отправке */
   607→      }
   608→    })();
   609→    return () => {
   610→      cancelled = true;
   611→    };
   612→  }, [secret, storageKey]);
   613→
   614→  const scrollToBottom = () => {
   615→    requestAnimationFrame(() => {
   616→      scrollRef.current?.scrollTo({ top: scrollRef.current.scrollHeight });
   617→    });
   618→  };
   619→
   620→  useEffect(() => {
   621→    scrollToBottom();
   622→  }, [messages]);
   623→
   624→  const poll = useCallback(async () => {
   625→    if (!convRef.current) return;
   626→    try {
   627→      const d = await api<{
   628→        messages: { id: string; role: string; text: string; buttons?: { id: string; text: string }[] }[];
   629→      }>(`/api/webhook/demo/${secret}?conversationId=${convRef.current}`);
   630→      setMessages((prev) => (d.messages.length > prev.length ? d.messages : prev));
   631→    } catch {
   632→      /* ignore */
   633→    }
   634→  }, [secret]);
   635→
   636→  useEffect(() => {
   637→    const t = setInterval(poll, 4000);
   638→    return () => clearInterval(t);
   639→  }, [poll]);
   640→
   641→  const send = async (text: string) => {
   642→    const t = text.trim();
   643→    if (!t || sending) return;
   644→    setSending(true);
   645→    setInput('');
   646→    setMessages((prev) => [...prev, { id: `local-${Date.now()}`, role: 'user', text: t }]);
   647→    scrollToBottom();
   648→    try {
   649→      const d = await api<{
   650→        conversationId: string;
   651→        messages: { id: string; role: string; text: string; buttons?: { id: string; text: string }[] }[];
   652→      }>(`/api/webhook/demo/${secret}`, {
   653→        method: 'POST',
   654→        body: JSON.stringify({ text: t, conversationId: convRef.current, visitorId: visitorRef.current || undefined }),
   655→      });
   656→      convRef.current = d.conversationId;
   657→      try {
   658→        window.localStorage.setItem(storageKey, d.conversationId);
   659→      } catch {
   660→        /* ignore */
   661→      }
   662→      setMessages(d.messages);
   663→      scrollToBottom();
   664→    } catch (e) {
   665→      // IMP-FE21-23: сбой виден в чате системным сообщением, а не теряется молча
   666→      setMessages((prev) => [
   667→        ...prev,
   668→        {
   669→          id: `err-${Date.now()}`,
   670→          role: 'bot',
   671→          text: `Не удалось отправить${e instanceof Error && e.message ? `: ${e.message}` : ''}`,
   672→        },
   673→      ]);

   675→      setSending(false);
   676→    }
   677→  };
   678→
   679→  return (

   681→      <div className="flex items-center gap-3 rounded-t-xl bg-gradient-to-r from-emerald-600 to-teal-600 p-4 text-white">
   682→        <div className="flex h-10 w-10 items-center justify-center rounded-full bg-white/20">
   683→          <Bot className="h-5 w-5" />
   684→        </div>

   688→            <span className="h-1.5 w-1.5 rounded-full bg-emerald-300" /> онлайн

   691→        <Badge variant="secondary" className="text-[10px]">
   692→          демо сайта
   693→        </Badge>

   696→        {messages.length === 0 && (
   697→          <div className="flex h-full items-center justify-center text-center text-sm text-muted-foreground">
   698→            Напишите сообщение — бот ответит
   699→            <br />
   700→            как в настоящем виджете на сайте
   701→          </div>
   702→        )}
   703→        {messages.map((m) => (

   705→            <div

   719→          if (!lastButtons?.length) return null;

   750→          {sending ? <Loader2 className="h-4 w-4 animate-spin" /> : <Send className="h-4 w-4" />}
