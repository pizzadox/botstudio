   217→   217→        <span className="flex items-center gap-1.5">

   234→   234→          <span className="flex items-center gap-1.5">

   246→   246→function OrderChat({
   247→   247→  botId,

   303→   303→    }
   304→   304→  };

   306→   306→  return (

   317→   317→            Клиент может написать в боте, открыв карточку своей заявки.
   318→   318→          </div>
   319→   319→        )}

   321→   321→          const fromClient = m.role === 'user';
   322→   322→          return (
   323→   323→            <div key={m.id} className={cn('flex', fromClient ? 'justify-start' : 'justify-end')}>

   325→   325→                className={cn(

   327→   327→                  fromClient

   338→   338→                <div className="text-right text-[9px] opacity-60">
   339→   339→                  {new Date(m.createdAt).toLocaleTimeString('ru-RU', {
   340→   340→                    hour: '2-digit',
   341→   341→                    minute: '2-digit',
   342→   342→                  })}
   343→   343→                </div>
   344→   344→              </div>
   345→   345→            </div>
   346→   346→          );
   347→   347→        })}
   348→   348→      </div>

   358→   358→            hasConversation
   359→   359→              ? 'Ответить клиенту…'
   360→   360→              : 'Диалог с клиентом не привязан — чат только для чтения'
