    65→/** Цвет маркера на карте */
    66→function markerColor(order: OrderDto): string {
    67→  if (ARCHIVE_STATUSES.includes(order.status)) return 'bg-slate-400';
    68→  switch (order.type) {
    69→    case 'waste':
    70→      return 'bg-emerald-500';
    71→    case 'kgm':
    72→      return 'bg-amber-500';
    73→    default:
    74→      return 'bg-violet-500';
    75→  }
    76→}

    87→// ─── Карта OpenFreeMap (MapLibre GL, бесплатные векторные тайлы без ключа) ───
    88→
    89→function OrdersMap({
    90→  orders,
    91→  showCompleted,
    92→  placementOrderId,
    93→  selectedId,
    94→  focusOrderId,
    95→  focusTick,
    96→  onSelect,
    97→  onPlace,
    98→  onHighlight,
    99→}: {
   100→  orders: OrderDto[];
   101→  showCompleted: boolean;
   102→  placementOrderId: string | null;
   103→  selectedId: string | null;
   104→  focusOrderId: string | null;
   105→  focusTick: number;
   106→  onSelect: (id: string) => void;
   107→  onPlace: (lat: number, lng: number) => void;
   108→  /** Выделить заявку без открытия карточки (клик по списку точек) */
   109→  onHighlight: (id: string) => void;
   110→}) {
   111→  const containerRef = useRef<HTMLDivElement>(null);
   112→  const mapRef = useRef<MlMap | null>(null);
   113→  const mlRef = useRef<typeof import('maplibre-gl') | null>(null);
   114→  const markersRef = useRef<MlMarker[]>([]);
   115→  const [ready, setReady] = useState(false);
   116→  const [failed, setFailed] = useState(false);
   117→  const [listOpen, setListOpen] = useState(false);
   118→
   119→  // Инициализация карты (динамический импорт — компонент клиентский)
   120→  useEffect(() => {
   121→    let cancelled = false;
   122→    (async () => {
   123→      try {
   124→        const ml = await import('maplibre-gl');
   125→        if (cancelled || !containerRef.current) return;
   126→        // Воркер MapLibre не собирается Turbopack'ом — берём локальную копию из public/
   127→        ml.setWorkerUrl('/maplibre-gl-worker.mjs');
   128→        mlRef.current = ml;
   129→        const map = new ml.Map({
   130→          container: containerRef.current,
   131→          style: 'https://tiles.openfreemap.org/styles/liberty',
   132→          center: [32.3, 58.35], // Новгородская область (Великий Новгород — Боровичи — Валдай)
   133→          zoom: 7,
   134→        });
   135→        map.addControl(new ml.NavigationControl({ showCompass: false }), 'top-right');
   136→        map.on('load', () => {
   137→          if (!cancelled) setReady(true);
   138→        });
   139→        map.on('error', (e) => {
   140→          // Тайловая сеть недоступна — сообщаем, но карту не убиваем
   141→          console.warn('[orders map]', e?.error?.message ?? e);
   142→        });
   143→        mapRef.current = map;
   144→      } catch (err) {
   145→        console.error('[orders map] init failed', err);
   146→        if (!cancelled) setFailed(true);
   147→      }
   148→    })();
   149→    return () => {
   150→      cancelled = true;
   151→      mapRef.current?.remove();
   152→      mapRef.current = null;
   153→      setReady(false);
   154→    };
   155→  }, []);
   156→
   157→  // Маркеры заявок
   158→  const visible = orders.filter(
   159→    (o) => o.lat != null && o.lng != null && (showCompleted || ACTIVE_STATUSES.includes(o.status))
   160→  );
   161→
   162→  useEffect(() => {
   163→    const map = mapRef.current;
   164→    const ml = mlRef.current;
   165→    if (!map || !ml || !ready) return;
   166→    for (const m of markersRef.current) m.remove();
   167→    markersRef.current = [];
   168→    for (const o of visible) {
   169→      const el = document.createElement('button');
   170→      el.type = 'button';
   171→      el.title = `Заявка №${o.number} — ${ORDER_TYPE_LABELS[o.type] ?? ''}`;
   172→      // Без hover:scale — масштабирование сдвигает маркер и «прыгает»;
   173→      // подсвечиваем тенью и рамкой, размер элемента не меняется
   174→      el.className = cn(
   175→        'flex h-8 w-8 cursor-pointer items-center justify-center rounded-full border-2 text-[11px] font-bold text-white shadow-md transition-[box-shadow,filter,border-color] hover:shadow-xl hover:brightness-110',
   176→        markerColor(o),
   177→        o.id === selectedId ? 'border-primary ring-2 ring-primary/50' : 'border-white'
   178→      );
   179→      el.textContent = String(o.number);
   180→      el.addEventListener('click', (e) => {
   181→        e.stopPropagation();
   182→        onSelect(o.id);
   183→      });
   184→      const marker = new ml.Marker({ element: el })
   185→        .setLngLat([o.lng as number, o.lat as number])
   186→        .addTo(map);
   187→      markersRef.current.push(marker);
   188→    }
   189→    // зависимость — сериализованный список видимых маркеров
   190→  }, [ready, visible.map((o) => `${o.id}:${o.lat}:${o.lng}:${o.status}`).join('|'), selectedId]);
   191→
   192→  // Наведение на заявку из списка/карточки: перелетаем к точке (focusTick —
   193→  // чтобы повторный клик по той же заявке тоже срабатывал)
   194→  useEffect(() => {
   195→    const map = mapRef.current;
   196→    if (!map || !ready || !focusOrderId) return;
   197→    const o = visible.find((x) => x.id === focusOrderId);
   198→    if (!o || o.lat == null || o.lng == null) return;
   199→    map.flyTo({ center: [o.lng, o.lat], zoom: Math.max(map.getZoom(), 15), duration: 800 });
   200→  }, [ready, focusOrderId, focusTick]);
   201→
   202→  // Во время режима указания точки список скрываем (баннер занимает то же место)
   203→  useEffect(() => {
   204→    if (placementOrderId) setListOpen(false);
   205→  }, [placementOrderId]);
   206→
   207→  // Переход к точке из списка: перелёт + выделение, без открытия карточки
   208→  const jumpToListPoint = useCallback(
   209→    (o: OrderDto) => {
   210→      const map = mapRef.current;
   211→      if (!map || o.lat == null || o.lng == null) return;
   212→      onHighlight(o.id);
   213→      map.flyTo({ center: [o.lng, o.lat], zoom: Math.max(map.getZoom(), 15), duration: 800 });
   214→      setListOpen(false);
   215→    },
   216→    [onHighlight]
   217→  );
   218→
   219→  // Режим указания точки: клик по карте → координаты заявки
   220→  const handleMapClick = useCallback(
   221→    (lat: number, lng: number) => onPlace(lat, lng),
   222→    [onPlace]
   223→  );
   224→  useEffect(() => {
   225→    const map = mapRef.current;
   226→    if (!map || !ready || !placementOrderId) return;
   227→    const onClick = (e: { lngLat: { lat: number; lng: number } }) =>
   228→      handleMapClick(e.lngLat.lat, e.lngLat.lng);
   229→    map.on('click', onClick);
   230→    map.getCanvas().style.cursor = 'crosshair';
   231→    return () => {
   232→      map.off('click', onClick);
   233→      map.getCanvas().style.cursor = '';
   234→    };
   235→  }, [ready, placementOrderId, handleMapClick]);
   236→
   237→  if (failed) {
   238→    return (
   239→      <div className="flex h-full min-h-[420px] flex-col items-center justify-center gap-3 rounded-xl border bg-muted/30 text-center text-muted-foreground">
   240→        <MapPin className="h-10 w-10 opacity-30" />
   241→        <p className="max-w-xs text-sm">
   242→          Не удалось загрузить карту OpenFreeMap. Проверьте доступ к tiles.openfreemap.org и
   243→          обновите страницу.
   244→        </p>
   245→      </div>
   246→    );
   247→  }
   248→
   249→  return (
   250→    <div className="relative h-full min-h-[380px] overflow-hidden rounded-xl border bg-muted/30">
   251→      <div ref={containerRef} className="h-full w-full" aria-label="Карта заявок" />
   252→      {!ready && (
   253→        <div className="absolute inset-0 flex items-center justify-center gap-2 text-sm text-muted-foreground">
   254→          <Loader2 className="h-4 w-4 animate-spin" /> Загружаю карту OpenFreeMap…
   255→        </div>
   256→      )}
   257→      {/* Легенда (компактная, не перекрывает атрибуцию и контролы) */}
   258→      <div className="absolute bottom-2.5 left-2.5 z-10 flex flex-wrap items-center gap-x-2.5 gap-y-0.5 rounded-lg border bg-background/95 px-2.5 py-1.5 text-[10px] leading-tight shadow-sm backdrop-blur sm:bottom-3 sm:left-3 sm:text-[11px]">
   259→        <span className="flex items-center gap-1.5">
   260→          <span className="h-2.5 w-2.5 rounded-full bg-emerald-500" /> отходы
   261→        </span>
   262→        <span className="flex items-center gap-1.5">
   263→          <span className="h-2.5 w-2.5 rounded-full bg-amber-500" /> КГМ
   264→        </span>
   265→        <span className="flex items-center gap-1.5">
   266→          <span className="h-2.5 w-2.5 rounded-full bg-slate-400" /> выполненные
   267→        </span>
   268→      </div>
   269→      {/* Счётчик маркеров + кнопка разворачиваемого списка точек */}
   270→      <div className="absolute left-2.5 top-2.5 z-10 flex items-start gap-2 sm:left-3 sm:top-3">
   271→        <div className="rounded-lg border bg-background/95 px-2.5 py-1.5 text-xs shadow-sm backdrop-blur">
   272→          <MapPin className="mr-1 inline h-3 w-3 text-primary" />
   273→          На карте: <b>{visible.length}</b>
   274→        </div>
   275→        {!placementOrderId && visible.length > 0 && (
   276→          <button
   277→            type="button"
   278→            onClick={() => setListOpen((v) => !v)}
   279→            aria-expanded={listOpen}
   280→            className={cn(
   281→              'flex items-center gap-1 rounded-lg border bg-background/95 px-2.5 py-1.5 text-xs shadow-sm backdrop-blur transition-colors hover:bg-muted',
   282→              listOpen && 'bg-muted'
   283→            )}
   284→          >
   285→            <List className="h-3 w-3 text-primary" />
   286→            Список
   287→            <ChevronDown
   288→              className={cn('h-3 w-3 transition-transform duration-200', listOpen && 'rotate-180')}
   289→            />
   290→          </button>
   291→        )}
   292→      </div>
   293→      {/* Разворачиваемый список точек: клик — перелёт к маркеру */}
   294→      {listOpen && (
   295→        <div className="absolute inset-x-2.5 top-12 z-20 flex max-h-[62%] flex-col overflow-hidden rounded-xl border bg-background/95 shadow-lg backdrop-blur sm:inset-x-auto sm:left-3 sm:top-12 sm:w-[340px]">
   296→          <div className="flex items-center justify-between border-b px-3 py-2 text-[11px] font-medium text-muted-foreground">
   297→            <span>Точки на карте ({visible.length}) — нажмите для перехода</span>
   298→            <button
   299→              type="button"
   300→              onClick={() => setListOpen(false)}
   301→              className="rounded px-1 text-[11px] transition-colors hover:bg-muted hover:text-foreground"

   306→          <ul className="min-h-0 flex-1 divide-y overflow-y-auto">
   307→            {visible.map((o) => (
   308→              <li key={o.id}>
   309→                <button
   310→                  type="button"
   311→                  onClick={() => jumpToListPoint(o)}
   312→                  className={cn(
   313→                    'flex w-full items-center gap-2 px-2.5 py-2 text-left transition-colors hover:bg-muted/70',
   314→                    o.id === selectedId && 'bg-primary/5'
   315→                  )}
   316→                >
   317→                  <span

   328→                    </span>
   329→                    <span className="block truncate text-[10px] text-muted-foreground">
   330→                      {[o.city, o.clientName, o.wishDate].filter(Boolean).join(' · ') || '—'}

   332→                  </span>
   333→                  <Badge
   334→                    variant="outline"
   335→                    className={cn('h-4 shrink-0 px-1 text-[9px]', ORDER_STATUS_BADGES[o.status])}

   337→                    {ORDER_STATUS_LABELS[o.status] ?? o.status}

   345→      {placementOrderId && (
   346→        <div className="absolute inset-x-2.5 top-12 z-20 flex items-center justify-between gap-2 rounded-lg border border-primary/40 bg-primary px-3 py-2 text-xs font-medium text-primary-foreground shadow-md sm:inset-x-auto sm:right-auto sm:left-3 sm:top-12">
   347→          <span className="flex items-center gap-1.5">

   349→            {placementOrderId}
   350→          </span>
   351→        </div>
   352→      )}
   353→    </div>
   354→  );
   355→}

   359→function OrderChat({
   360→  botId,
   361→  orderId,
   362→  hasConversation,
   363→}: {
   364→  botId: string;
   365→  orderId: string;
   366→  hasConversation: boolean;
   367→}) {
   368→  const { toast } = useToast();
   369→  const [messages, setMessages] = useState<OrderMessageDto[]>([]);
   370→  const [text, setText] = useState('');
   371→  const [sending, setSending] = useState(false);
   372→  const scrollRef = useRef<HTMLDivElement>(null);
   373→
   374→  const load = useCallback(async () => {
   375→    try {
   376→      const d = await api<{ messages: OrderMessageDto[] }>(
   377→        `/api/bots/${botId}/orders/${orderId}/messages`
   378→      );
   379→      setMessages(d.messages);
   380→    } catch {
   381→      /* polling errors ignored */
   382→    }
   383→  }, [botId, orderId]);
   384→
   385→  useEffect(() => {
   386→    setMessages([]);
   387→    load();
   388→    const t = setInterval(load, 3000);
   389→    return () => clearInterval(t);
   390→  }, [load]);
   391→
   392→  useEffect(() => {
   393→    requestAnimationFrame(() => {
   394→      scrollRef.current?.scrollTo({ top: scrollRef.current.scrollHeight });
   395→    });
   396→  }, [messages.length]);
   397→
   398→  const send = async () => {
   399→    const t = text.trim();
   400→    if (!t) return;
   401→    setSending(true);
   402→    setText('');
   403→    try {
   404→      const d = await api<{ message: OrderMessageDto; delivered: boolean; deliveryError?: string }>(
   405→        `/api/bots/${botId}/orders/${orderId}/messages`,
   406→        { method: 'POST', body: JSON.stringify({ text: t }) }
   407→      );
   408→      setMessages((prev) => [...prev, d.message]);
   409→      if (!d.delivered && d.deliveryError) {
   410→        toast({ title: 'Не доставлено в мессенджер', description: d.deliveryError, variant: 'destructive' });
   411→      }
   412→    } catch {
   413→      toast({ title: 'Не удалось отправить', variant: 'destructive' });
   414→    } finally {
   415→      setSending(false);
   416→    }
   417→  };
   418→
   419→  return (
   420→    <div className="flex min-h-0 flex-1 flex-col">
   421→      <div className="flex items-center gap-2 border-b px-3 py-2 text-xs text-muted-foreground">
   422→        <MessageSquare className="h-3.5 w-3.5" />
   423→        Чат с клиентом по заявке — сообщения доставляются в его мессенджер
   424→      </div>
   425→      <div ref={scrollRef} className="min-h-0 flex-1 space-y-2 overflow-y-auto bg-muted/30 p-3">
   426→        {messages.length === 0 && (
   427→          <div className="flex h-full items-center justify-center text-center text-xs text-muted-foreground">
   428→            Сообщений по заявке пока нет.
   429→            <br />
   430→            Клиент может написать в боте, открыв карточку своей заявки.
   431→          </div>
   432→        )}
   433→        {messages.map((m) => {
   434→          const fromClient = m.role === 'user';
   435→          return (
   436→            <div key={m.id} className={cn('flex', fromClient ? 'justify-start' : 'justify-end')}>
   437→              <div
   438→                className={cn(
   439→                  'max-w-[85%] space-y-0.5 rounded-2xl px-3 py-1.5 text-sm leading-snug',
   440→                  fromClient
   441→                    ? 'rounded-bl-md border bg-card'
   442→                    : 'rounded-br-md bg-primary text-primary-foreground'
   443→                )}
   444→              >
   445→                {m.nodeId === '__operator' && (
   446→                  <div className="flex items-center gap-1 text-[10px] opacity-70">
   447→                    <Headset className="h-3 w-3" /> оператор
   448→                  </div>
   449→                )}
   450→                <div className="whitespace-pre-wrap">{m.text}</div>
   451→                <div className="text-right text-[9px] opacity-60">
   452→                  {new Date(m.createdAt).toLocaleTimeString('ru-RU', {
   453→                    hour: '2-digit',
   454→                    minute: '2-digit',
   455→                  })}
   456→                </div>
   457→              </div>
   458→            </div>
   459→          );
   460→        })}
   461→      </div>
   462→      <form
   463→        className="flex items-center gap-2 border-t p-2.5 pb-[max(0.625rem,env(safe-area-inset-bottom))]"
   464→        onSubmit={(e) => {
   465→          e.preventDefault();
   466→          send();
   467→        }}
   468→      >
   469→        <Input
   470→          placeholder={
   471→            hasConversation
   472→              ? 'Ответить клиенту…'
   473→              : 'Диалог с клиентом не привязан — чат только для чтения'
   474→          }
   475→          value={text}
   476→          disabled={!hasConversation}
   477→          onChange={(e) => setText(e.target.value)}
   478→        />
   479→        <Button
   480→          type="submit"
   481→          size="icon"
   482→          className="h-10 w-10 shrink-0"
   483→          disabled={sending || !hasConversation || !text.trim()}
   484→          aria-label="Отправить"
   485→        >
   486→          {sending ? <Loader2 className="h-4 w-4 animate-spin" /> : <Send className="h-4 w-4" />}
   487→        </Button>
   488→      </form>
   489→    </div>
   490→  );
   491→}

   495→function OrderDetailDialog({
   496→  botId,
   497→  orderId,
   498→  open,

   502→  onShowOnMap,

   504→}: {

   506→  orderId: string | null;
   507→  open: boolean;
   508→  onClose: () => void;
   509→  onSwitchOrder: (id: string) => void;
   510→  onPlaceRequest: (orderId: string) => void;
   511→  onShowOnMap: (orderId: string) => void;
   512→  onDataChanged: () => void;
   513→}) {
   514→  const { toast } = useToast();
   515→  const [order, setOrder] = useState<OrderDto | null>(null);
   516→  const [clientOrders, setClientOrders] = useState<OrderDto[]>([]);
   517→  const [loading, setLoading] = useState(false);
   518→  const [assignee, setAssignee] = useState('');
   519→  const [address, setAddress] = useState('');
   520→  const [wishDate, setWishDate] = useState('');
   521→  const [comment, setComment] = useState('');
   522→  const [saving, setSaving] = useState(false);
   523→  const [geoLoading, setGeoLoading] = useState(false);
   524→  const [confirmGeo, setConfirmGeo] = useState(false);
   525→  const [confirmDelete, setConfirmDelete] = useState(false);
   526→
   527→  const load = useCallback(async () => {
   528→    if (!orderId) return;
   529→    setLoading(true);
   530→    try {
   531→      const d = await api<{ order: OrderDto; clientOrders: OrderDto[] }>(
   532→        `/api/bots/${botId}/orders/${orderId}`
   533→      );
   534→      setOrder(d.order);
   535→      setClientOrders(d.clientOrders);
   536→      setAssignee(d.order.assignee ?? '');
   537→      setAddress(d.order.address ?? '');
   538→      setWishDate(d.order.wishDate ?? '');
   539→      setComment(d.order.comment ?? '');
   540→    } catch {
   541→      toast({ title: 'Не удалось загрузить заявку', variant: 'destructive' });
   542→    } finally {
   543→      setLoading(false);
   544→    }
   545→  }, [botId, orderId, toast]);
   546→
   547→  useEffect(() => {
   548→    if (open && orderId) load();
   549→    if (!open) {
   550→      setOrder(null);
   551→      setClientOrders([]);
   552→      setConfirmDelete(false);
   553→      setConfirmGeo(false);
   554→    }
   555→  }, [open, orderId, load]);
   556→
   557→  const patch = async (data: Record<string, unknown>, successMsg?: string) => {
   558→    if (!orderId) return;
   559→    setSaving(true);
   560→    try {
   561→      const d = await api<{ order: OrderDto }>(`/api/bots/${botId}/orders/${orderId}`, {
   562→        method: 'PATCH',
   563→        body: JSON.stringify(data),
   564→      });
   565→      setOrder(d.order);
   566→      onDataChanged();
   567→      if (successMsg) toast({ title: successMsg });
   568→    } catch {
   569→      toast({ title: 'Ошибка сохранения', variant: 'destructive' });
   570→    } finally {
   571→      setSaving(false);
   572→    }

   576→   * Геокодирование адреса. Ручная точка защищена: если координаты ставили
   577→   * вручную — сначала подтверждение; если адрес не нашёлся — прежние
   578→   * координаты остаются (не «слетают»).
   579→   */

   581→    if (!orderId || !order) return;
   582→    setConfirmGeo(false);
   583→    setGeoLoading(true);
   584→    try {
   585→      const d = await api<{ order: OrderDto; geo?: { ok: boolean; lat?: number; lng?: number } }>(
   586→        `/api/bots/${botId}/orders/${orderId}`,
   587→        { method: 'PATCH', body: JSON.stringify({ geocode: true }) }
   588→      );
   589→      setOrder(d.order);
   590→      onDataChanged();
   591→      if (d.geo?.ok) {
   592→        toast({
   593→          title: `Координаты обновлены: ${d.geo.lat?.toFixed(5)}, ${d.geo.lng?.toFixed(5)}`,
   594→        });
   595→      } else if (d.order.lat != null) {
   596→        toast({
   597→          title: 'Адрес не найден на карте',
   598→          description: 'Прежние координаты сохранены без изменений',
   599→        });
   600→      } else {
   601→        toast({
   602→          title: 'Адрес не найден на карте',
   603→          description: 'Уточните адрес или укажите точку кнопкой «Указать на карте»',
   604→          variant: 'destructive',
   605→        });
   606→      }
   607→    } catch {
   608→      toast({ title: 'Ошибка геокодирования', variant: 'destructive' });
   609→    } finally {
   610→      setGeoLoading(false);
   611→    }
   612→  };
   613→
   614→  const remove = async () => {
   615→    if (!orderId) return;
   616→    try {
   617→      await api(`/api/bots/${botId}/orders/${orderId}`, { method: 'DELETE' });
   618→      toast({ title: `Заявка удалена` });
   619→      onDataChanged();
   620→      onClose();
   621→    } catch {
   622→      toast({ title: 'Не удалось удалить', variant: 'destructive' });
   623→    }
   624→  };
   625→
   626→  const hasManualPoint = order?.geoSource === 'manual';
   627→
   628→  return (
   629→    <Dialog open={open} onOpenChange={(v) => !v && onClose()}>
   630→      <DialogContent className="grid h-[100dvh] w-screen max-w-none grid-rows-[auto_1fr] gap-0 overflow-hidden rounded-none border-0 p-0 sm:h-[92dvh] sm:w-[calc(100%-2rem)] sm:max-w-4xl sm:rounded-xl sm:border md:h-[85vh]">
   631→        <DialogHeader className="border-b px-4 py-3 pr-14 md:pl-5 md:pr-16">
   632→          <DialogTitle className="flex min-w-0 flex-wrap items-center gap-x-2 gap-y-1 text-base">
   633→            <span className="shrink-0 text-lg">{order ? ORDER_TYPE_ICONS[order.type] : '📋'}</span>
   634→            <span className="min-w-0 flex-1 truncate">
   635→              {order ? `Заявка №${order.number} — ${ORDER_TYPE_LABELS[order.type] ?? ''}` : 'Заявка'}
   636→            </span>
   637→            {order && (
   638→              <Badge
   639→                variant="outline"
   640→                className={cn('shrink-0 text-[10px]', ORDER_STATUS_BADGES[order.status])}
   641→              >
   642→                {ORDER_STATUS_LABELS[order.status] ?? order.status}
   643→              </Badge>
   644→            )}
   645→          </DialogTitle>
   646→          <DialogDescription className="sr-only">
   647→            Карточка заявки: данные клиента, статус и встроенный чат
   648→          </DialogDescription>
   649→        </DialogHeader>
   650→
   651→        {!order && loading ? (
   652→          <div className="flex items-center justify-center gap-2 text-muted-foreground">
   653→            <Loader2 className="h-5 w-5 animate-spin" /> Загрузка…
   654→          </div>
   655→        ) : !order ? (
   656→          <div className="flex items-center justify-center text-sm text-muted-foreground">
   657→            Заявка не найдена
   658→          </div>
   659→        ) : (
   660→          <div className="min-h-0 overflow-y-auto pb-[max(1rem,env(safe-area-inset-bottom))] md:grid md:grid-cols-2 md:overflow-hidden md:pb-0">
   661→            {/* Левая колонка: клиент + данные заявки */}
   662→            <div className="min-h-0 space-y-4 border-b p-4 md:border-b-0 md:border-r md:overflow-y-auto">
   663→              {/* Клиент */}
   664→              <div className="rounded-xl border bg-muted/30 p-3">
   665→                <div className="flex items-center justify-between gap-2">
   666→                  <h3 className="flex items-center gap-1.5 text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   667→                    <User className="h-3.5 w-3.5" /> Карточка клиента

   670→                    <span
   671→                      className={cn(
   672→                        'shrink-0 rounded-full px-2 py-0.5 text-[10px] font-medium',

   675→                    >
   676→                      {SOURCE_LABELS[order.conversation.source] ?? order.conversation.source}

   678→                  )}
   679→                </div>
   680→                <div className="mt-2 space-y-1 text-sm">
   681→                  <div className="break-words font-medium">

   778→                    <Button
   779→                      size="sm"
   780→                      variant="outline"
   781→                      className="h-9 shrink-0"
   782→                      disabled={saving}
   783→                      onClick={() => patch({ assignee }, 'Исполнитель назначен')}
   784→                    >
   785→                      Назначить
   786→                    </Button>
   787→                  </div>
   788→                </div>

   802→                        <SelectItem value="__none">Город не указан</SelectItem>
   803→                        {SERVICE_CITIES.map((c) => (
   804→                          <SelectItem key={c} value={c}>
   805→                            {c}
   806→                          </SelectItem>
   807→                        ))}
   808→                      </SelectContent>
   809→                    </Select>
   810→                    <Input
   811→                      className="h-9"
   812→                      value={address}
   813→                      placeholder="Улица, дом"
   814→                      onChange={(e) => setAddress(e.target.value)}
   815→                      onBlur={() => {
   816→                        if (address !== (order.address ?? '')) patch({ address }, 'Адрес обновлён');

   818→                      disabled={saving}
   819→                    />
   820→                  </div>
   821→                  <div className="flex flex-wrap gap-2 pt-1">
   822→                    <Button
   823→                      size="sm"
   824→                      variant="outline"
   825→                      className="h-8"
   826→                      onClick={() => onPlaceRequest(order.id)}
   827→                    >
   828→                      <Crosshair className="h-3.5 w-3.5" />
   829→                      {order.lat != null ? 'Перенести точку' : 'Указать на карте'}
   830→                    </Button>
   831→                    {order.lat != null && (
   832→                      <Button
   833→                        size="sm"
   834→                        variant="outline"
   835→                        className="h-8"
   836→                        onClick={() => onShowOnMap(order.id)}
   837→                      >
   838→                        <MapPin className="h-3.5 w-3.5" /> Показать на карте

   847→                        onClick={() => setConfirmGeo(true)}

   849→                        <Globe className="h-3.5 w-3.5" /> Геокодировать

   851→                    ) : (
   852→                      <Button
   853→                        size="sm"
   854→                        variant="outline"
   855→                        className="h-8"
   856→                        disabled={geoLoading || saving || !address.trim()}
   857→                        onClick={runGeocode}
   858→                      >
   859→                        {geoLoading ? (
   860→                          <Loader2 className="h-3.5 w-3.5 animate-spin" />
   861→                        ) : (
   862→                          <Globe className="h-3.5 w-3.5" />
   863→                        )}
   864→                        Геокодировать
   865→                      </Button>
   866→                    )}
   867→                  </div>
   868→                  {confirmGeo && (
   869→                    <div className="flex flex-col gap-2 rounded-lg border border-amber-300/70 bg-amber-50 px-2.5 py-2 text-xs text-amber-800 sm:flex-row sm:items-center">
   870→                      <span className="min-w-0 flex-1">
   871→                        Точка установлена вручную. Пересчитать координаты по адресу?
   872→                      </span>
   873→                      <span className="flex shrink-0 gap-1.5 sm:ml-auto">
   874→                        <Button size="sm" className="h-7" onClick={runGeocode}>

   877→                        <Button
   878→                          size="sm"
   879→                          variant="ghost"
   880→                          className="h-7"
   881→                          onClick={() => setConfirmGeo(false)}
   882→                        >
   883→                          Оставить
   884→                        </Button>
   885→                      </span>
   886→                    </div>
   887→                  )}
   888→                  {order.lat != null ? (

   890→                      Координаты: {order.lat.toFixed(5)}, {order.lng?.toFixed(5)}

   895→                    <div className="pt-1 text-[11px] text-amber-600">

   919→                    Сохранить комментарий
   920→                  </Button>
   921→                </div>
   922→                <div className="flex flex-wrap items-center justify-between gap-2 border-t pt-2 text-[11px] text-muted-foreground">

   929→                      </Button>
   930→                      <Button
   931→                        size="sm"
   932→                        variant="ghost"
   933→                        className="h-7"
   934→                        onClick={() => setConfirmDelete(false)}
   935→                      >
   936→                        Нет
   937→                      </Button>
   938→                    </span>
   939→                  ) : (
   940→                    <button
   941→                      className="flex items-center gap-1 text-muted-foreground transition-colors hover:text-destructive"
   942→                      onClick={() => setConfirmDelete(true)}
   943→                    >
   944→                      <Trash2 className="h-3 w-3" /> удалить
   945→                    </button>
   946→                  )}
   947→                </div>
   948→              </div>
   949→            </div>
   950→
   951→            {/* Правая колонка: встроенный чат с пользователем.

   953→                на десктопе — занимает всю высоту колонки. */}
   954→            <div className="flex h-[46dvh] min-h-[300px] flex-col md:h-auto md:min-h-0 md:overflow-hidden">
   955→              <OrderChat
   956→                botId={botId}
   957→                orderId={order.id}
   958→                hasConversation={!!order.conversationId}
   959→              />
   960→            </div>
   961→          </div>
   962→        )}
   963→      </DialogContent>
   964→    </Dialog>
   965→  );
   966→}

   968→// ─── Диалог создания заявки оператором ───────────────────────────────────────
   969→
   970→function NewOrderDialog({
   971→  botId,
   972→  open,
   973→  onClose,
   974→  onCreated,
   975→}: {
   976→  botId: string;
   977→  open: boolean;
   978→  onClose: () => void;
   979→  onCreated: () => void;
   980→}) {
   981→  const { toast } = useToast();
   982→  const [saving, setSaving] = useState(false);
   983→  const OTHER_CITY = '__other';
   984→  const [form, setForm] = useState({
   985→    type: 'waste',
   986→    clientName: '',
   987→    phone: '',
   988→    city: cityButtons()[0] ?? '',
   989→    cityOther: '',
   990→    street: '',
   991→    size: '',
   992→    wishDate: '',
   993→    comment: '',
   994→  });
   995→
   996→  const submit = async () => {
   997→    const city = (form.city === OTHER_CITY ? form.cityOther : form.city).trim();
   998→    const address = composeAddress(city, form.street);
   999→    if (!address.trim() && !form.clientName.trim()) {
  1000→      toast({ title: 'Укажите хотя бы адрес или имя клиента', variant: 'destructive' });
  1001→      return;
  1002→    }
  1003→    setSaving(true);
  1004→    try {
  1005→      await api(`/api/bots/${botId}/orders`, {
  1006→        method: 'POST',
  1007→        body: JSON.stringify({
  1008→          type: form.type,
  1009→          clientName: form.clientName,
  1010→          phone: form.phone,
  1011→          city,
  1012→          address,
  1013→          size: form.size,
  1014→          wishDate: form.wishDate,
  1015→          comment: form.comment,
  1016→        }),
  1017→      });
  1018→      toast({ title: 'Заявка создана — координаты подбираются автоматически' });
  1019→      setForm({
