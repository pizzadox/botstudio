    72→    case 'waste':
    73→      return 'bg-emerald-500';
    74→    case 'kgm':
    75→      return 'bg-amber-500';
    76→    default:
    77→      return 'bg-violet-500';
    78→  }
    79→}

    90→// ─── Карта OpenFreeMap (MapLibre GL, бесплатные векторные тайлы без ключа) ───
    91→
    92→function OrdersMap({
    93→  orders,
    94→  showCompleted,
    95→  mytkoEnabled,
    96→  placementOrderId,
    97→  selectedId,
    98→  focusOrderId,
    99→  focusTick,
   100→  onSelect,
   101→  onPlace,
   102→  onHighlight,
   103→}: {
   104→  orders: OrderDto[];
   105→  showCompleted: boolean;
   106→  mytkoEnabled: boolean;
   107→  placementOrderId: string | null;
   108→  selectedId: string | null;
   109→  focusOrderId: string | null;
   110→  focusTick: number;
   111→  onSelect: (id: string) => void;
   112→  onPlace: (lat: number, lng: number) => void;
   113→  /** Выделить заявку без открытия карточки (клик по списку точек) */
   114→  onHighlight: (id: string) => void;
   115→}) {
   116→  const containerRef = useRef<HTMLDivElement>(null);
   117→  const mapRef = useRef<MlMap | null>(null);
   118→  const mlRef = useRef<typeof import('maplibre-gl') | null>(null);
   119→  /** маркеры по id заявки — чтобы сдвигать их при «spiderfy» совпадающих точек */
   120→  const markersRef = useRef<Map<string, MlMarker>>(new Map());
   121→  /** исходные координаты заявки (без сдвига) — для пересчёта раскладки на каждом move */
   122→  const baseCoordsRef = useRef<Map<string, { lat: number; lng: number }>>(new Map());
   123→  const [ready, setReady] = useState(false);
   124→  const [failed, setFailed] = useState(false);
   125→  const [listOpen, setListOpen] = useState(false);
   126→
   127→  // Инициализация карты (динамический импорт — компонент клиентский)
   128→  useEffect(() => {
   129→    let cancelled = false;
   130→    (async () => {
   131→      try {
   132→        const ml = await import('maplibre-gl');
   133→        if (cancelled || !containerRef.current) return;
   134→        // Воркер MapLibre не собирается Turbopack'ом — берём локальную копию из public/
   135→        ml.setWorkerUrl('/maplibre-gl-worker.mjs');
   136→        mlRef.current = ml;
   137→        const map = new ml.Map({
   138→          container: containerRef.current,
   139→          style: 'https://tiles.openfreemap.org/styles/liberty',
   140→          center: [32.3, 58.35], // Новгородская область (Великий Новгород — Боровичи — Валдай)
   141→          zoom: 7,
   142→        });
   143→        map.addControl(new ml.NavigationControl({ showCompass: false }), 'top-right');
   144→        map.on('load', () => {
   145→          if (!cancelled) setReady(true);
   146→        });
   147→        map.on('error', (e) => {
   148→          // Тайловая сеть недоступна — сообщаем, но карту не убиваем
   149→          console.warn('[orders map]', e?.error?.message ?? e);
   150→        });
   151→        mapRef.current = map;
   152→      } catch (err) {
   153→        console.error('[orders map] init failed', err);
   154→        if (!cancelled) setFailed(true);
   155→      }
   156→    })();
   157→    return () => {
   158→      cancelled = true;
   159→      mapRef.current?.remove();
   160→      mapRef.current = null;
   161→      setReady(false);
   162→    };
   163→  }, []);
   164→
   165→  // Маркеры заявок
   166→  const visible = orders.filter(
   167→    (o) => o.lat != null && o.lng != null && (showCompleted || ACTIVE_STATUSES.includes(o.status))
   168→  );
   169→
   170→  /**
   171→   * «Spiderfy» совпадающих точек: несколько заявок с одним адресом геокодятся
   172→   * в идентичные координаты и ложатся маркерами друг на друга — видна только
   173→   * одна. Группируем их по округлённым координатам и раскладываем кружком
   174→   * с ПОСТОЯННЫМ пиксельным радиусом (пересчёт на каждом move/zoom), чтобы
   175→   * все точки были видны и кликабельны на любом масштабе карты.
   176→   */
   177→  const spreadMarkers = useCallback(() => {
   178→    const map = mapRef.current;
   179→    if (!map) return;
   180→
   181→    // группы совпадающих координат (6 знаков ≈ 0.1 м — попадают и «почти равные»)
   182→    const groups = new Map<string, string[]>();
   183→    for (const [id, c] of baseCoordsRef.current) {
   184→      const key = `${c.lat.toFixed(6)}|${c.lng.toFixed(6)}`;
   185→      const g = groups.get(key);
   186→      if (g) g.push(id);
   187→      else groups.set(key, [id]);
   188→    }
   189→
   190→    const zoom = map.getZoom();
   191→    // пикселей на градус долготы в Web Mercator (тайл MapLibre = 512px!)
   192→    const pxPerDegLng = 512 * Math.pow(2, zoom) / 360;
   193→    const SPREAD_PX = 44; // диаметр раскладки вокруг истинной точки
   194→
   195→    for (const ids of groups.values()) {
   196→      const n = ids.length;
   197→      ids.forEach((id, i) => {
   198→        const marker = markersRef.current.get(id);
   199→        const base = baseCoordsRef.current.get(id);
   200→        if (!marker || !base) return;
   201→        if (n === 1) {
   202→          marker.setLngLat([base.lng, base.lat]);
   203→          return;
   204→        }
   205→        // равномерно по окружности; при 2 точках — слева и справа от истинной
   206→        const angle = (2 * Math.PI * i) / n + (n === 2 ? Math.PI / 2 : Math.PI / 6);
   207→        const dxPx = Math.cos(angle) * (SPREAD_PX / 2);
   208→        const dyPx = Math.sin(angle) * (SPREAD_PX / 2);
   209→        const dLng = dxPx / pxPerDegLng;
   210→        const dLat = (dyPx / pxPerDegLng) * Math.cos((base.lat * Math.PI) / 180);
   211→        marker.setLngLat([base.lng + dLng, base.lat + dLat]);
   212→      });
   213→    }
   214→  }, []);
   215→
   216→  useEffect(() => {
   217→    const map = mapRef.current;
   218→    const ml = mlRef.current;
   219→    if (!map || !ml || !ready) return;
   220→    for (const m of markersRef.current.values()) m.remove();
   221→    markersRef.current = new Map();
   222→    baseCoordsRef.current = new Map();
   223→    for (const o of visible) {
   224→      const el = document.createElement('button');
   225→      el.type = 'button';
   226→      el.title = `Заявка №${o.number} — ${ORDER_TYPE_LABELS[o.type] ?? ''}`;
   227→      // Без hover:scale — масштабирование сдвигает маркер и «прыгает»;
   228→      // подсвечиваем тенью и рамкой, размер элемента не меняется
   229→      el.className = cn(
   230→        'flex h-8 w-8 cursor-pointer items-center justify-center rounded-full border-2 text-[11px] font-bold text-white shadow-md transition-[box-shadow,filter,border-color] hover:shadow-xl hover:brightness-110',
   231→        markerColor(o),
   232→        o.id === selectedId ? 'border-primary ring-2 ring-primary/50' : 'border-white'
   233→      );
   234→      el.textContent = String(o.number);
   235→      el.addEventListener('click', (e) => {
   236→        e.stopPropagation();
   237→        onSelect(o.id);
   238→      });
   239→      const marker = new ml.Marker({ element: el })
   240→        .setLngLat([o.lng as number, o.lat as number])
   241→        .addTo(map);
   242→      markersRef.current.set(o.id, marker);
   243→      baseCoordsRef.current.set(o.id, { lat: o.lat as number, lng: o.lng as number });
   244→    }
   245→    // раскладываем совпадающие точки и обновляем раскладку при движении/зуме
   246→    spreadMarkers();
   247→    map.on('move', spreadMarkers);
   248→    return () => {
   249→      map.off('move', spreadMarkers);
   250→    };
   251→    // зависимость — сериализованный список видимых маркеров
   252→  }, [ready, visible.map((o) => `${o.id}:${o.lat}:${o.lng}:${o.status}`).join('|'), selectedId, spreadMarkers]);
   253→
   254→  // Наведение на заявку из списка/карточки: перелетаем к точке (focusTick —
   255→  // чтобы повторный клик по той же заявке тоже срабатывал)
   256→  useEffect(() => {
   257→    const map = mapRef.current;
   258→    if (!map || !ready || !focusOrderId) return;
   259→    const o = visible.find((x) => x.id === focusOrderId);
   260→    if (!o || o.lat == null || o.lng == null) return;
   261→    map.flyTo({ center: [o.lng, o.lat], zoom: Math.max(map.getZoom(), 15), duration: 800 });
   262→  }, [ready, focusOrderId, focusTick]);
   263→
   264→  // Во время режима указания точки список скрываем (баннер занимает то же место)
   265→  useEffect(() => {
   266→    if (placementOrderId) setListOpen(false);
   267→  }, [placementOrderId]);
   268→
   269→  // Переход к точке из списка: перелёт + выделение, без открытия карточки
   270→  const jumpToListPoint = useCallback(
   271→    (o: OrderDto) => {
   272→      const map = mapRef.current;
   273→      if (!map || o.lat == null || o.lng == null) return;
   274→      onHighlight(o.id);
   275→      map.flyTo({ center: [o.lng, o.lat], zoom: Math.max(map.getZoom(), 15), duration: 800 });
   276→      setListOpen(false);
   277→    },
   278→    [onHighlight]
   279→  );
   280→
   281→  // Режим указания точки: клик по карте → координаты заявки
   282→  const handleMapClick = useCallback(
   283→    (lat: number, lng: number) => onPlace(lat, lng),
   284→    [onPlace]
   285→  );
   286→  useEffect(() => {
   287→    const map = mapRef.current;
   288→    if (!map || !ready || !placementOrderId) return;
   289→    const onClick = (e: { lngLat: { lat: number; lng: number } }) =>
   290→      handleMapClick(e.lngLat.lat, e.lngLat.lng);
   291→    map.on('click', onClick);
   292→    map.getCanvas().style.cursor = 'crosshair';
   293→    return () => {
   294→      map.off('click', onClick);
   295→      map.getCanvas().style.cursor = '';
   296→    };
   297→  }, [ready, placementOrderId, handleMapClick]);
   298→
   299→  if (failed) {
   300→    return (
   301→      <div className="flex h-full min-h-[420px] flex-col items-center justify-center gap-3 rounded-xl border bg-muted/30 text-center text-muted-foreground">
   302→        <MapPin className="h-10 w-10 opacity-30" />
   303→        <p className="max-w-xs text-sm">
   304→          Не удалось загрузить карту OpenFreeMap. Проверьте доступ к tiles.openfreemap.org и
   305→          обновите страницу.
   306→        </p>
   307→      </div>
   308→    );
   309→  }
   310→
   311→  return (
   312→    <div className="relative h-full min-h-[380px] overflow-hidden rounded-xl border bg-muted/30">
   313→      <div ref={containerRef} className="h-full w-full" aria-label="Карта заявок" />
   314→      {!ready && (
   315→        <div className="absolute inset-0 flex items-center justify-center gap-2 text-sm text-muted-foreground">
   316→          <Loader2 className="h-4 w-4 animate-spin" /> Загружаю карту OpenFreeMap…
   317→        </div>
   318→      )}
   319→      {/* Легенда (компактная, не перекрывает атрибуцию и контролы) */}
   320→      <div className="absolute bottom-2.5 left-2.5 z-10 flex flex-wrap items-center gap-x-2.5 gap-y-0.5 rounded-lg border bg-background/95 px-2.5 py-1.5 text-[10px] leading-tight shadow-sm backdrop-blur sm:bottom-3 sm:left-3 sm:text-[11px]">
   321→        <span className="flex items-center gap-1.5">
   322→          <span className="h-2.5 w-2.5 rounded-full bg-emerald-500" /> отходы
   323→        </span>
   324→        <span className="flex items-center gap-1.5">
   325→          <span className="h-2.5 w-2.5 rounded-full bg-amber-500" /> КГМ
   326→        </span>
   327→        <span className="flex items-center gap-1.5">
   328→          <span className="h-2.5 w-2.5 rounded-full bg-slate-400" /> выполненные
   329→        </span>
   330→      </div>
   331→      {/* Счётчик маркеров + кнопка разворачиваемого списка точек */}
   332→      <div className="absolute left-2.5 top-2.5 z-10 flex items-start gap-2 sm:left-3 sm:top-3">
   333→        <div
   334→          role="status"
   335→          aria-label={`Точек на карте: ${visible.length}`}
   336→          className="rounded-lg border bg-background/95 px-2.5 py-1.5 text-xs shadow-sm backdrop-blur"
   337→        >
   338→          <MapPin className="mr-1 inline h-3 w-3 text-primary" aria-hidden />
   339→          На карте: <b className="tabular-nums">{visible.length}</b>
   340→        </div>
   341→        {!placementOrderId && visible.length > 0 && (
   342→          <button
   343→            type="button"
   344→            onClick={() => setListOpen((v) => !v)}
   345→            aria-expanded={listOpen}
   346→            className={cn(
   347→              'flex items-center gap-1 rounded-lg border bg-background/95 px-2.5 py-1.5 text-xs shadow-sm backdrop-blur transition-colors hover:bg-muted',
   348→              listOpen && 'bg-muted'
   349→            )}
   350→          >
   351→            <List className="h-3 w-3 text-primary" />
   352→            Список
   353→            <ChevronDown
   354→              className={cn('h-3 w-3 transition-transform duration-200', listOpen && 'rotate-180')}
   355→            />
   356→          </button>
   357→        )}
   358→      </div>
   359→      {/* Разворачиваемый список точек: клик — перелёт к маркеру */}
   360→      {listOpen && (
   361→        <div className="absolute inset-x-2.5 top-12 z-20 flex max-h-[62%] flex-col overflow-hidden rounded-xl border bg-background/95 shadow-lg backdrop-blur sm:inset-x-auto sm:left-3 sm:top-12 sm:w-[340px]">
   362→          <div className="flex items-center justify-between border-b px-3 py-2 text-[11px] font-medium text-muted-foreground">
   363→            <span>Точки на карте ({visible.length}) — нажмите для перехода</span>
   364→            <button
   365→              type="button"
   366→              onClick={() => setListOpen(false)}
   367→              className="rounded px-1 text-[11px] transition-colors hover:bg-muted hover:text-foreground"
   368→            >
   369→              Скрыть
   370→            </button>
   371→          </div>
   372→          <ul className="min-h-0 flex-1 divide-y overflow-y-auto">
   373→            {visible.map((o) => (
   374→              <li key={o.id}>
   375→                <button
   376→                  type="button"
   377→                  onClick={() => jumpToListPoint(o)}
   378→                  className={cn(
   379→                    'flex w-full items-center gap-2 px-2.5 py-2.5 text-left transition-colors hover:bg-muted/70 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-inset focus-visible:ring-ring',
   380→                    o.id === selectedId && 'bg-primary/5'
   381→                  )}
   382→                  aria-label={`Перейти к заявке №${o.number} — ${o.address || o.city || 'без адреса'}`}
   383→                >
   384→                  <span
   385→                    className={cn(
   386→                      'flex h-6 w-6 shrink-0 items-center justify-center rounded-full text-[10px] font-bold text-white',
   387→                      markerColor(o)
   388→                    )}
   389→                  >
   390→                    {o.number}
   391→                  </span>
   392→                  <span className="min-w-0 flex-1">
   393→                    <span className="block truncate text-xs font-medium">
   394→                      {ORDER_TYPE_ICONS[o.type]} {o.address || o.city || 'Без адреса'}
   395→                    </span>
   396→                    <span className="block truncate text-[10px] text-muted-foreground">
   397→                      {[o.city, o.clientName, o.wishDate].filter(Boolean).join(' · ') || '—'}
   398→                    </span>
   399→                  </span>
   400→                  <Badge
   401→                    variant="outline"
   402→                    className={cn('h-4 shrink-0 px-1 text-[9px]', ORDER_STATUS_BADGES[o.status])}
   403→                  >
   404→                    {ORDER_STATUS_LABELS[o.status] ?? o.status}
   405→                  </Badge>
   406→                  {mytkoEnabled && <MytkoBadge status={o.mytkoStatus} />}
   407→                </button>
   408→              </li>
   409→            ))}
   410→          </ul>
   411→        </div>
   412→      )}
   413→      {placementOrderId && (
   414→        <div className="absolute inset-x-2.5 top-12 z-20 flex items-center justify-between gap-2 rounded-lg border border-primary/40 bg-primary px-3 py-2 text-xs font-medium text-primary-foreground shadow-md sm:inset-x-auto sm:right-auto sm:left-3 sm:top-12">
   415→          <span className="flex items-center gap-1.5">
   416→            <Crosshair className="h-3.5 w-3.5 shrink-0" /> Кликните по карте — точка заявки №
   417→            {placementOrderId}
   418→          </span>
   419→        </div>
   420→      )}
   421→    </div>
   422→  );
   423→}

   427→function OrderChat({
   428→  botId,
   429→  orderId,
   430→  hasConversation,
   431→}: {
   432→  botId: string;
   433→  orderId: string;
   434→  hasConversation: boolean;
   435→}) {
   436→  const { toast } = useToast();
   437→  const [messages, setMessages] = useState<OrderMessageDto[]>([]);
   438→  const [text, setText] = useState('');
   439→  const [sending, setSending] = useState(false);
   440→  const scrollRef = useRef<HTMLDivElement>(null);
   441→
   442→  const load = useCallback(async () => {
   443→    try {
   444→      const d = await api<{ messages: OrderMessageDto[] }>(
   445→        `/api/bots/${botId}/orders/${orderId}/messages`
   446→      );
   447→      setMessages(d.messages);
   448→    } catch {
   449→      /* polling errors ignored */
   450→    }
   451→  }, [botId, orderId]);
   452→
   453→  useEffect(() => {
   454→    setMessages([]);
   455→    load();
   456→    const t = setInterval(load, 3000);
   457→    return () => clearInterval(t);
   458→  }, [load]);
   459→
   460→  useEffect(() => {
   461→    requestAnimationFrame(() => {
   462→      scrollRef.current?.scrollTo({ top: scrollRef.current.scrollHeight });
   463→    });
   464→  }, [messages.length]);
   465→
   466→  const send = async () => {
   467→    const t = text.trim();
   468→    if (!t) return;
   469→    setSending(true);
   470→    setText('');
   471→    try {
   472→      const d = await api<{ message: OrderMessageDto; delivered: boolean; deliveryError?: string }>(

   494→      <div ref={scrollRef} aria-live="polite" className="min-h-0 flex-1 space-y-2 overflow-y-auto bg-muted/30 p-3">
   495→        {messages.length === 0 && (
   496→          <div className="flex h-full flex-col items-center justify-center gap-2 px-4 text-center text-muted-foreground">
   497→            <div className="flex h-10 w-10 items-center justify-center rounded-xl bg-primary/10 text-primary" aria-hidden>
   498→              <MessageSquare className="h-5 w-5" />
   499→            </div>
   500→            <p className="text-xs">
   501→              Сообщений по заявке пока нет.
   502→              <br />
   503→              Клиент может написать в боте, открыв карточку своей заявки.
   504→            </p>
   505→          </div>
   506→        )}
   507→        {messages.map((m) => {
   508→          const fromClient = m.role === 'user';
   509→          return (
   510→            <div key={m.id} className={cn('flex', fromClient ? 'justify-start' : 'justify-end')}>
   511→              <div
   512→                className={cn(
   513→                  'max-w-[85%] space-y-0.5 rounded-2xl px-3 py-1.5 text-sm leading-snug',

   521→                    <Headset className="h-3 w-3" /> оператор
   522→                  </div>
   523→                )}
   524→                <div className="whitespace-pre-wrap">{m.text}</div>
   525→                <div className="text-right text-[10px] opacity-60">
   526→                  {new Date(m.createdAt).toLocaleTimeString('ru-RU', {
   527→                    hour: '2-digit',
   528→                    minute: '2-digit',
   529→                  })}
   530→                </div>
   531→              </div>
   532→            </div>
   533→          );
   534→        })}
   535→      </div>
   536→      <form
   537→        className="flex items-center gap-2 border-t p-2.5 pb-[max(0.625rem,env(safe-area-inset-bottom))]"
   538→        onSubmit={(e) => {
   539→          e.preventDefault();
   540→          send();
   541→        }}
   542→      >
   543→        <Input
   544→          aria-label="Ответ клиенту"
   545→          placeholder={
   546→            hasConversation
   547→              ? 'Ответить клиенту…'
   548→              : 'Диалог с клиентом не привязан — чат только для чтения'
   549→          }
   550→          value={text}
   551→          disabled={!hasConversation}
   552→          onChange={(e) => setText(e.target.value)}
   553→        />
   554→        <Button
   555→          type="submit"
   556→          size="icon"
   557→          className="h-10 w-10 shrink-0"
   558→          disabled={sending || !hasConversation || !text.trim()}
   559→          aria-label="Отправить"
   560→        >
   561→          {sending ? <Loader2 className="h-4 w-4 animate-spin" /> : <Send className="h-4 w-4" />}
   562→        </Button>
   563→      </form>
   564→    </div>
   565→  );
   566→}

   570→function OrderDetailDialog({
   571→  botId,
   572→  orderId,
   573→  open,
   574→  onClose,
   575→  onSwitchOrder,
   576→  onPlaceRequest,
   577→  onShowOnMap,
   578→  onDataChanged,
   579→  mytkoEnabled,
   580→}: {
   581→  botId: string;
   582→  orderId: string | null;
   583→  open: boolean;
   584→  onClose: () => void;

   587→  onShowOnMap: (orderId: string) => void;
   588→  onDataChanged: () => void;
   589→  mytkoEnabled: boolean;
   590→}) {
   591→  const { toast } = useToast();

   597→  const [wishDate, setWishDate] = useState('');
   598→  const [comment, setComment] = useState('');

   600→  const [geoLoading, setGeoLoading] = useState(false);
   601→  const [confirmGeo, setConfirmGeo] = useState(false);
   602→  const [confirmDelete, setConfirmDelete] = useState(false);
   603→  const [mytkoBusy, setMytkoBusy] = useState(false);
   604→
   605→  const load = useCallback(async () => {
   606→    if (!orderId) return;
   607→    setLoading(true);
   608→    try {
   609→      const d = await api<{ order: OrderDto; clientOrders: OrderDto[] }>(
   610→        `/api/bots/${botId}/orders/${orderId}`
   611→      );

   621→      setLoading(false);

   640→        method: 'PATCH',
   641→        body: JSON.stringify(data),
   642→      });

   644→      onDataChanged();
   645→      if (successMsg) toast({ title: successMsg });
   646→    } catch {
   647→      toast({ title: 'Ошибка сохранения', variant: 'destructive' });
   648→    } finally {
   649→      setSaving(false);
   650→    }
   651→  };
   652→
   653→  /**
   654→   * Геокодирование адреса. Ручная точка защищена: если координаты ставили
   655→   * вручную — сначала подтверждение; если адрес не нашёлся — прежние
   656→   * координаты остаются (не «слетают»).
   657→   */
   658→  const runGeocode = async () => {
   659→    if (!orderId || !order) return;
   660→    setConfirmGeo(false);
   661→    setGeoLoading(true);
   662→    try {
   663→      const d = await api<{ order: OrderDto; geo?: { ok: boolean; lat?: number; lng?: number } }>(
   664→        `/api/bots/${botId}/orders/${orderId}`,
   665→        { method: 'PATCH', body: JSON.stringify({ geocode: true }) }
   666→      );
   667→      setOrder(d.order);
   668→      onDataChanged();
   669→      if (d.geo?.ok) {
   670→        toast({
   671→          title: `Координаты обновлены: ${d.geo.lat?.toFixed(5)}, ${d.geo.lng?.toFixed(5)}`,
   672→        });
   673→      } else if (d.order.lat != null) {
   674→        toast({
   675→          title: 'Адрес не найден на карте',
   676→          description: 'Прежние координаты сохранены без изменений',
   677→        });
   678→      } else {
   679→        toast({
   680→          title: 'Адрес не найден на карте',
   681→          description: 'Уточните адрес или укажите точку кнопкой «Указать на карте»',

   687→    } finally {
   688→      setGeoLoading(false);
   689→    }
   690→  };
   691→
   692→  const remove = async () => {
   693→    if (!orderId) return;
   694→    try {
   695→      await api(`/api/bots/${botId}/orders/${orderId}`, { method: 'DELETE' });

   698→      onClose();

   709→      const d = await api<{ ok: boolean; order: OrderDto }>(
   710→        `/api/bots/${botId}/orders/${orderId}/mytko`,
   711→        { method: 'POST' }
   712→      );
   713→      setOrder(d.order);
   714→      onDataChanged();
   715→      if (d.order.mytkoStatus === 'synced') {
   716→        toast({ title: 'Синхронизировано с MyTKO ✅', description: d.order.mytkoInfo ?? undefined });
   717→      } else {
   718→        toast({
   719→          title: 'Ошибка синхронизации с MyTKO',
   720→          description: d.order.mytkoError ?? undefined,
   721→          variant: 'destructive',
   722→        });
   723→      }
   724→    } catch (e) {
   725→      toast({ title: 'Ошибка синхронизации', description: e instanceof Error ? e.message : '', variant: 'destructive' });
   726→    } finally {
   727→      setMytkoBusy(false);
   728→    }
   729→  };
   730→
   731→  const hasManualPoint = order?.geoSource === 'manual';
   732→
   733→  return (
   734→    <Dialog open={open} onOpenChange={(v) => !v && onClose()}>
   735→      <DialogContent className="grid h-[100dvh] w-screen max-w-none grid-rows-[auto_1fr] gap-0 overflow-hidden rounded-none border-0 p-0 sm:h-[92dvh] sm:w-[calc(100%-2rem)] sm:max-w-4xl sm:rounded-xl sm:border md:h-[85vh]">
   736→        <DialogHeader className="border-b px-4 py-3 pr-14 md:pl-5 md:pr-16">
   737→          <DialogTitle className="flex min-w-0 flex-wrap items-center gap-x-2 gap-y-1 text-base">
   738→            <span className="shrink-0 text-lg">{order ? ORDER_TYPE_ICONS[order.type] : '📋'}</span>
   739→            <span className="min-w-0 flex-1 truncate">
   740→              {order ? `Заявка №${order.number} — ${ORDER_TYPE_LABELS[order.type] ?? ''}` : 'Заявка'}
   741→            </span>
   742→            {order && (
   743→              <Badge
   744→                variant="outline"
   745→                className={cn('shrink-0 text-[10px]', ORDER_STATUS_BADGES[order.status])}
   746→              >
   747→                {ORDER_STATUS_LABELS[order.status] ?? order.status}

   751→          </DialogTitle>
   752→          <DialogDescription className="sr-only">

   762→          <div className="flex items-center justify-center text-sm text-muted-foreground">
   763→            Заявка не найдена
   764→          </div>
   765→        ) : (
   766→          <div className="min-h-0 overflow-y-auto pb-[max(1rem,env(safe-area-inset-bottom))] md:grid md:grid-cols-2 md:overflow-hidden md:pb-0">

   944→                        onClick={() => onShowOnMap(order.id)}
