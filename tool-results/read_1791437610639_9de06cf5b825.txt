    76→    76→    default:
    77→    77→      return 'bg-violet-500';
    78→    78→  }
    79→    79→}

    91→    91→
    92→    92→function OrdersMap({
    93→    93→  orders,
    94→    94→  showCompleted,
    95→    95→  mytkoEnabled,
    96→    96→  placementOrderId,
    97→    97→  selectedId,
    98→    98→  focusOrderId,
    99→    99→  focusTick,
   100→   100→  onSelect,
   101→   101→  onPlace,
   102→   102→  onHighlight,
   103→   103→}: {
   104→   104→  orders: OrderDto[];
   105→   105→  showCompleted: boolean;
   106→   106→  mytkoEnabled: boolean;
   107→   107→  placementOrderId: string | null;
   108→   108→  selectedId: string | null;
   109→   109→  focusOrderId: string | null;
   110→   110→  focusTick: number;
   111→   111→  onSelect: (id: string) => void;
   112→   112→  onPlace: (lat: number, lng: number) => void;
   113→   113→  /** Выделить заявку без открытия карточки (клик по списку точек) */
   114→   114→  onHighlight: (id: string) => void;
   115→   115→}) {
   116→   116→  const containerRef = useRef<HTMLDivElement>(null);
   117→   117→  const mapRef = useRef<MlMap | null>(null);
   118→   118→  const mlRef = useRef<typeof import('maplibre-gl') | null>(null);
   119→   119→  /** маркеры по id заявки — чтобы сдвигать их при «spiderfy» совпадающих точек */
   120→   120→  const markersRef = useRef<Map<string, MlMarker>>(new Map());
   121→   121→  /** исходные координаты заявки (без сдвига) — для пересчёта раскладки на каждом move */
   122→   122→  const baseCoordsRef = useRef<Map<string, { lat: number; lng: number }>>(new Map());
   123→   123→  const [ready, setReady] = useState(false);
   124→   124→  const [failed, setFailed] = useState(false);
   125→   125→  const [listOpen, setListOpen] = useState(false);
   126→   126→
   127→   127→  // Инициализация карты (динамический импорт — компонент клиентский)
   128→   128→  useEffect(() => {
   129→   129→    let cancelled = false;
   130→   130→    (async () => {
   131→   131→      try {
   132→   132→        const ml = await import('maplibre-gl');
   133→   133→        if (cancelled || !containerRef.current) return;
   134→   134→        // Воркер MapLibre не собирается Turbopack'ом — берём локальную копию из public/
   135→   135→        ml.setWorkerUrl('/maplibre-gl-worker.mjs');
   136→   136→        mlRef.current = ml;
   137→   137→        const map = new ml.Map({
   138→   138→          container: containerRef.current,
   139→   139→          style: 'https://tiles.openfreemap.org/styles/liberty',
   140→   140→          center: [32.3, 58.35], // Новгородская область (Великий Новгород — Боровичи — Валдай)
   141→   141→          zoom: 7,
   142→   142→        });
   143→   143→        map.addControl(new ml.NavigationControl({ showCompass: false }), 'top-right');
   144→   144→        map.on('load', () => {
   145→   145→          if (!cancelled) setReady(true);
   146→   146→        });
   147→   147→        map.on('error', (e) => {
   148→   148→          // Тайловая сеть недоступна — сообщаем, но карту не убиваем
   149→   149→          console.warn('[orders map]', e?.error?.message ?? e);
   150→   150→        });
   151→   151→        mapRef.current = map;
   152→   152→      } catch (err) {
   153→   153→        console.error('[orders map] init failed', err);
   154→   154→        if (!cancelled) setFailed(true);
   155→   155→      }
   156→   156→    })();
   157→   157→    return () => {
   158→   158→      cancelled = true;
   159→   159→      mapRef.current?.remove();
   160→   160→      mapRef.current = null;
   161→   161→      setReady(false);
   162→   162→    };
   163→   163→  }, []);
   164→   164→
   165→   165→  // Маркеры заявок
   166→   166→  const visible = orders.filter(
   167→   167→    (o) => o.lat != null && o.lng != null && (showCompleted || ACTIVE_STATUSES.includes(o.status))
   168→   168→  );
   169→   169→
   170→   170→  /**
   171→   171→   * «Spiderfy» совпадающих точек: несколько заявок с одним адресом геокодятся
   172→   172→   * в идентичные координаты и ложатся маркерами друг на друга — видна только
   173→   173→   * одна. Группируем их по округлённым координатам и раскладываем кружком
   174→   174→   * с ПОСТОЯННЫМ пиксельным радиусом (пересчёт на каждом move/zoom), чтобы
   175→   175→   * все точки были видны и кликабельны на любом масштабе карты.
   176→   176→   */
   177→   177→  const spreadMarkers = useCallback(() => {
   178→   178→    const map = mapRef.current;
   179→   179→    if (!map) return;
   180→   180→
   181→   181→    // группы совпадающих координат (6 знаков ≈ 0.1 м — попадают и «почти равные»)
   182→   182→    const groups = new Map<string, string[]>();
   183→   183→    for (const [id, c] of baseCoordsRef.current) {
   184→   184→      const key = `${c.lat.toFixed(6)}|${c.lng.toFixed(6)}`;
   185→   185→      const g = groups.get(key);
   186→   186→      if (g) g.push(id);
   187→   187→      else groups.set(key, [id]);
   188→   188→    }
   189→   189→
   190→   190→    const zoom = map.getZoom();
   191→   191→    // пикселей на градус долготы в Web Mercator (тайл MapLibre = 512px!)
   192→   192→    const pxPerDegLng = 512 * Math.pow(2, zoom) / 360;
   193→   193→    const SPREAD_PX = 44; // диаметр раскладки вокруг истинной точки
   194→   194→
   195→   195→    for (const ids of groups.values()) {
   196→   196→      const n = ids.length;
   197→   197→      ids.forEach((id, i) => {
   198→   198→        const marker = markersRef.current.get(id);
   199→   199→        const base = baseCoordsRef.current.get(id);
   200→   200→        if (!marker || !base) return;
   201→   201→        if (n === 1) {
   202→   202→          marker.setLngLat([base.lng, base.lat]);
   203→   203→          return;
   204→   204→        }
   205→   205→        // равномерно по окружности; при 2 точках — слева и справа от истинной
   206→   206→        const angle = (2 * Math.PI * i) / n + (n === 2 ? Math.PI / 2 : Math.PI / 6);
   207→   207→        const dxPx = Math.cos(angle) * (SPREAD_PX / 2);
   208→   208→        const dyPx = Math.sin(angle) * (SPREAD_PX / 2);
   209→   209→        const dLng = dxPx / pxPerDegLng;
   210→   210→        const dLat = (dyPx / pxPerDegLng) * Math.cos((base.lat * Math.PI) / 180);
   211→   211→        marker.setLngLat([base.lng + dLng, base.lat + dLat]);
   212→   212→      });
   213→   213→    }
   214→   214→  }, []);
   215→   215→
   216→   216→  useEffect(() => {
   217→   217→    const map = mapRef.current;
   218→   218→    const ml = mlRef.current;
   219→   219→    if (!map || !ml || !ready) return;
   220→   220→    for (const m of markersRef.current.values()) m.remove();
   221→   221→    markersRef.current = new Map();
   222→   222→    baseCoordsRef.current = new Map();
   223→   223→    for (const o of visible) {
   224→   224→      const el = document.createElement('button');
   225→   225→      el.type = 'button';
   226→   226→      el.title = `Заявка №${o.number} — ${ORDER_TYPE_LABELS[o.type] ?? ''}`;
   227→   227→      // Без hover:scale — масштабирование сдвигает маркер и «прыгает»;
   228→   228→      // подсвечиваем тенью и рамкой, размер элемента не меняется
   229→   229→      el.className = cn(
   230→   230→        'flex h-8 w-8 cursor-pointer items-center justify-center rounded-full border-2 text-[11px] font-bold text-white shadow-md transition-[box-shadow,filter,border-color] hover:shadow-xl hover:brightness-110',
   231→   231→        markerColor(o),
   232→   232→        o.id === selectedId ? 'border-primary ring-2 ring-primary/50' : 'border-white'
   233→   233→      );
   234→   234→      el.textContent = String(o.number);
   235→   235→      el.addEventListener('click', (e) => {
   236→   236→        e.stopPropagation();
   237→   237→        onSelect(o.id);
   238→   238→      });
   239→   239→      const marker = new ml.Marker({ element: el })
   240→   240→        .setLngLat([o.lng as number, o.lat as number])
   241→   241→        .addTo(map);
   242→   242→      markersRef.current.set(o.id, marker);
   243→   243→      baseCoordsRef.current.set(o.id, { lat: o.lat as number, lng: o.lng as number });
   244→   244→    }
   245→   245→    // раскладываем совпадающие точки и обновляем раскладку при движении/зуме
   246→   246→    spreadMarkers();
   247→   247→    map.on('move', spreadMarkers);
   248→   248→    return () => {
   249→   249→      map.off('move', spreadMarkers);
   250→   250→    };
   251→   251→    // зависимость — сериализованный список видимых маркеров
   252→   252→  }, [ready, visible.map((o) => `${o.id}:${o.lat}:${o.lng}:${o.status}`).join('|'), selectedId, spreadMarkers]);
   253→   253→
   254→   254→  // Наведение на заявку из списка/карточки: перелетаем к точке (focusTick —
   255→   255→  // чтобы повторный клик по той же заявке тоже срабатывал)
   256→   256→  useEffect(() => {
   257→   257→    const map = mapRef.current;
   258→   258→    if (!map || !ready || !focusOrderId) return;
   259→   259→    const o = visible.find((x) => x.id === focusOrderId);
   260→   260→    if (!o || o.lat == null || o.lng == null) return;
   261→   261→    map.flyTo({ center: [o.lng, o.lat], zoom: Math.max(map.getZoom(), 15), duration: 800 });
   262→   262→  }, [ready, focusOrderId, focusTick]);
   263→   263→
   264→   264→  // Во время режима указания точки список скрываем (баннер занимает то же место)
   265→   265→  useEffect(() => {
   266→   266→    if (placementOrderId) setListOpen(false);
   267→   267→  }, [placementOrderId]);
   268→   268→
   269→   269→  // Переход к точке из списка: перелёт + выделение, без открытия карточки
   270→   270→  const jumpToListPoint = useCallback(
   271→   271→    (o: OrderDto) => {
   272→   272→      const map = mapRef.current;
   273→   273→      if (!map || o.lat == null || o.lng == null) return;
   274→   274→      onHighlight(o.id);
   275→   275→      map.flyTo({ center: [o.lng, o.lat], zoom: Math.max(map.getZoom(), 15), duration: 800 });
   276→   276→      setListOpen(false);
   277→   277→    },
   278→   278→    [onHighlight]
   279→   279→  );
   280→   280→
   281→   281→  // Режим указания точки: клик по карте → координаты заявки
   282→   282→  const handleMapClick = useCallback(
   283→   283→    (lat: number, lng: number) => onPlace(lat, lng),
   284→   284→    [onPlace]
   285→   285→  );
   286→   286→  useEffect(() => {
   287→   287→    const map = mapRef.current;
   288→   288→    if (!map || !ready || !placementOrderId) return;
   289→   289→    const onClick = (e: { lngLat: { lat: number; lng: number } }) =>
   290→   290→      handleMapClick(e.lngLat.lat, e.lngLat.lng);
   291→   291→    map.on('click', onClick);
   292→   292→    map.getCanvas().style.cursor = 'crosshair';
   293→   293→    return () => {
   294→   294→      map.off('click', onClick);
   295→   295→      map.getCanvas().style.cursor = '';
   296→   296→    };
   297→   297→  }, [ready, placementOrderId, handleMapClick]);
   298→   298→
   299→   299→  if (failed) {
   300→   300→    return (
   301→   301→      <div className="flex h-full min-h-[420px] flex-col items-center justify-center gap-3 rounded-xl border bg-muted/30 text-center text-muted-foreground">
   302→   302→        <MapPin className="h-10 w-10 opacity-30" />
   303→   303→        <p className="max-w-xs text-sm">
   304→   304→          Не удалось загрузить карту OpenFreeMap. Проверьте доступ к tiles.openfreemap.org и
   305→   305→          обновите страницу.
   306→   306→        </p>
   307→   307→      </div>
   308→   308→    );
   309→   309→  }
   310→   310→
   311→   311→  return (
   312→   312→    <div className="relative h-full min-h-[380px] overflow-hidden rounded-xl border bg-muted/30">
   313→   313→      <div ref={containerRef} className="h-full w-full" aria-label="Карта заявок" />
   314→   314→      {!ready && (
   315→   315→        <div className="absolute inset-0 flex items-center justify-center gap-2 text-sm text-muted-foreground">
   316→   316→          <Loader2 className="h-4 w-4 animate-spin" /> Загружаю карту OpenFreeMap…
   317→   317→        </div>
   318→   318→      )}
   319→   319→      {/* Легенда (компактная, не перекрывает атрибуцию и контролы) */}
   320→   320→      <div className="absolute bottom-2.5 left-2.5 z-10 flex flex-wrap items-center gap-x-2.5 gap-y-0.5 rounded-lg border bg-background/95 px-2.5 py-1.5 text-[10px] leading-tight shadow-sm backdrop-blur sm:bottom-3 sm:left-3 sm:text-[11px]">
   321→   321→        <span className="flex items-center gap-1.5">
   322→   322→          <span className="h-2.5 w-2.5 rounded-full bg-emerald-500" /> отходы
   323→   323→        </span>
   324→   324→        <span className="flex items-center gap-1.5">
   325→   325→          <span className="h-2.5 w-2.5 rounded-full bg-amber-500" /> КГМ
   326→   326→        </span>
   327→   327→        <span className="flex items-center gap-1.5">
   328→   328→          <span className="h-2.5 w-2.5 rounded-full bg-slate-400" /> выполненные
   329→   329→        </span>
   330→   330→      </div>
   331→   331→      {/* Счётчик маркеров + кнопка разворачиваемого списка точек */}
   332→   332→      <div className="absolute left-2.5 top-2.5 z-10 flex items-start gap-2 sm:left-3 sm:top-3">
   333→   333→        <div
   334→   334→          role="status"
   335→   335→          aria-label={`Точек на карте: ${visible.length}`}
   336→   336→          className="rounded-lg border bg-background/95 px-2.5 py-1.5 text-xs shadow-sm backdrop-blur"
   337→   337→        >
   338→   338→          <MapPin className="mr-1 inline h-3 w-3 text-primary" aria-hidden />
   339→   339→          На карте: <b className="tabular-nums">{visible.length}</b>
   340→   340→        </div>
   341→   341→        {!placementOrderId && visible.length > 0 && (
   342→   342→          <button
   343→   343→            type="button"
   344→   344→            onClick={() => setListOpen((v) => !v)}
   345→   345→            aria-expanded={listOpen}
   346→   346→            className={cn(
   347→   347→              'flex items-center gap-1 rounded-lg border bg-background/95 px-2.5 py-1.5 text-xs shadow-sm backdrop-blur transition-colors hover:bg-muted',
   348→   348→              listOpen && 'bg-muted'
   349→   349→            )}
   350→   350→          >
   351→   351→            <List className="h-3 w-3 text-primary" />
   352→   352→            Список
   353→   353→            <ChevronDown
   354→   354→              className={cn('h-3 w-3 transition-transform duration-200', listOpen && 'rotate-180')}
   355→   355→            />
   356→   356→          </button>
   357→   357→        )}
   358→   358→      </div>
   359→   359→      {/* Разворачиваемый список точек: клик — перелёт к маркеру */}
   360→   360→      {listOpen && (
   361→   361→        <div className="absolute inset-x-2.5 top-12 z-20 flex max-h-[62%] flex-col overflow-hidden rounded-xl border bg-background/95 shadow-lg backdrop-blur sm:inset-x-auto sm:left-3 sm:top-12 sm:w-[340px]">
   362→   362→          <div className="flex items-center justify-between border-b px-3 py-2 text-[11px] font-medium text-muted-foreground">
   363→   363→            <span>Точки на карте ({visible.length}) — нажмите для перехода</span>
   364→   364→            <button
   365→   365→              type="button"
   366→   366→              onClick={() => setListOpen(false)}
   367→   367→              className="rounded px-1 text-[11px] transition-colors hover:bg-muted hover:text-foreground"
   368→   368→            >
   369→   369→              Скрыть
   370→   370→            </button>
   371→   371→          </div>
   372→   372→          <ul className="min-h-0 flex-1 divide-y overflow-y-auto">
   373→   373→            {visible.map((o) => (
   374→   374→              <li key={o.id}>
   375→   375→                <button
   376→   376→                  type="button"
   377→   377→                  onClick={() => jumpToListPoint(o)}
   378→   378→                  className={cn(
   379→   379→                    'flex w-full items-center gap-2 px-2.5 py-2.5 text-left transition-colors hover:bg-muted/70 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-inset focus-visible:ring-ring',
   380→   380→                    o.id === selectedId && 'bg-primary/5'
   381→   381→                  )}
   382→   382→                  aria-label={`Перейти к заявке №${o.number} — ${o.address || o.city || 'без адреса'}`}
   383→   383→                >
   384→   384→                  <span
   385→   385→                    className={cn(
   386→   386→                      'flex h-6 w-6 shrink-0 items-center justify-center rounded-full text-[10px] font-bold text-white',
   387→   387→                      markerColor(o)
   388→   388→                    )}
   389→   389→                  >
