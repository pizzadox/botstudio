   100→/**
   101→ * Дедуп-слияние сообщений поллинга (IMP-F03): добавляем только новые id,
   102→ * сортируем по createdAt. Если ничего нового — возвращаем prev как есть
   103→ * (ссылочная стабильность убирает лишние ре-рендеры).
   104→ */
   105→function mergeMessages(prev: ChatMessage[], incoming: ChatMessage[]): ChatMessage[] {
   106→  const seen = new Set(prev.map((m) => m.id));
   107→  const add = incoming.filter((m) => !seen.has(m.id));
   108→  if (!add.length) return prev;
   109→  return [...prev, ...add].sort((a, b) => (a.createdAt > b.createdAt ? 1 : -1));
   110→}

   153→function timeAgo(iso: string): string {
   154→  const diff = Date.now() - new Date(iso).getTime();
   155→  const m = Math.floor(diff / 60000);
   156→  if (m < 1) return 'только что';
   157→  if (m < 60) return `${m} мин назад`;
   158→  const h = Math.floor(m / 60);
   159→  if (h < 24) return `${h} ч назад`;
   160→  return `${Math.floor(h / 24)} дн назад`;
   161→}

   413→/** Панель заявок клиента: просмотр и быстрое изменение статуса/исполнителя */
   414→function ConversationOrdersPanel({
   415→  conversationId,
   416→  botId,
   417→  refreshTick,
   418→}: {
   419→  conversationId: string;
   420→  botId: string;
   421→  refreshTick: number;
   422→}) {
   423→  const { toast } = useToast();
   424→  const [open, setOpen] = useState(false);
   425→  const [orders, setOrders] = useState<ConvOrder[]>([]);
   426→  const [mytkoEnabled, setMytkoEnabled] = useState(false);
   427→  const [loaded, setLoaded] = useState(false);
   428→  const [savingId, setSavingId] = useState<string | null>(null);
   429→
   430→  const load = useCallback(async () => {
   431→    try {
   432→      const d = await api<{ orders: ConvOrder[]; mytkoEnabled?: boolean }>(
   433→        `/api/conversations/${conversationId}/orders`
   434→      );
   435→      setOrders(d.orders);
   436→      setMytkoEnabled(!!d.mytkoEnabled);
   437→    } catch {
   438→      /* ignore */
   439→    } finally {
   440→      setLoaded(true);
   441→    }
   442→  }, [conversationId]);
   443→
   444→  useEffect(() => {
   445→    load();
   446→  }, [load, refreshTick]);
   447→
   448→  const patch = async (orderId: string, data: Record<string, unknown>) => {
   449→    setSavingId(orderId);
   450→    try {
   451→      const d = await api<{ order: { status: string; assignee: string | null } }>(
   452→        `/api/bots/${botId}/orders/${orderId}`,
   453→        { method: 'PATCH', body: JSON.stringify(data) }
   454→      );
   455→      setOrders((prev) =>
   456→        prev.map((o) =>
   457→          o.id === orderId ? { ...o, status: d.order.status, assignee: d.order.assignee } : o
   458→        )
   459→      );
   460→      toast({ title: 'Заявка обновлена' });
   461→    } catch {
   462→      toast({ title: 'Не удалось обновить заявку', variant: 'destructive' });
   463→    } finally {
   464→      setSavingId(null);
   465→    }
   466→  };
   467→
   468→  return (
   469→    <div className="border-b bg-background px-4 py-2">
   470→      <button
   471→        className="flex w-full items-center gap-1.5 rounded text-left text-xs font-medium text-muted-foreground transition-colors hover:text-foreground focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring focus-visible:ring-offset-2"
   472→        onClick={() => setOpen((v) => !v)}
   473→        aria-expanded={open}
   474→        aria-controls="conversation-orders-list"
   475→      >
   476→        {open ? <ChevronDown className="h-3.5 w-3.5" /> : <ChevronRight className="h-3.5 w-3.5" />}
   477→        <Truck className="h-3.5 w-3.5" />
   478→        Заявки клиента <span className="tabular-nums">({orders.length})</span>
   479→      </button>
   480→      <div id="conversation-orders-list">
   481→        {open && loaded && orders.length > 0 && (
   482→          <div className="mt-2 max-h-56 space-y-1.5 overflow-y-auto pr-1">
   483→            {orders.map((o) => (
   484→              <div key={o.id} className="rounded-lg border p-2">
   485→                <div className="flex flex-wrap items-center gap-x-2 gap-y-1">
   486→                  <span className="text-xs font-semibold tabular-nums">№{o.number}</span>
   487→                  <Badge
   488→                    variant="outline"
   489→                    className={cn('h-4 px-1 text-[9px]', ORDER_STATUS_BADGES[o.status])}
   490→                  >
   491→                    {ORDER_STATUS_LABELS[o.status] ?? o.status}
   492→                  </Badge>
   493→                  {mytkoEnabled && <MytkoBadge status={o.mytkoStatus} />}
   494→                  <span className="min-w-0 flex-1 truncate text-xs text-muted-foreground">
   495→                    {ORDER_TYPE_LABELS[o.type] ?? o.type}
   496→                    {o.address ? ` · ${o.address}` : ''}
   497→                  </span>
   498→                </div>
   499→                <div className="mt-1.5 flex flex-wrap items-center gap-1.5">
   500→                  <Select
   501→                    value={o.status}
   502→                    onValueChange={(v) => patch(o.id, { status: v })}
   503→                    disabled={savingId === o.id}
   504→                  >
   505→                    <SelectTrigger className="h-7 w-[130px] text-[11px]" aria-label="Статус заявки">
   506→                      <SelectValue />
   507→                    </SelectTrigger>
   508→                    <SelectContent>
   509→                      {Object.entries(ORDER_STATUS_LABELS).map(([k, v]) => (
   510→                        <SelectItem key={k} value={k} className="text-xs">
   511→                          {v}
   512→                        </SelectItem>
   513→                      ))}
   514→                    </SelectContent>
   515→                  </Select>
   516→                  <Input
   517→                    className="h-7 w-[120px] flex-1 text-[11px] sm:w-[140px]"
   518→                    defaultValue={o.assignee ?? ''}
   519→                    placeholder="Исполнитель…"
   520→                    aria-label="Исполнитель"
   521→                    onBlur={(e) => {
   522→                      if (e.target.value !== (o.assignee ?? '')) patch(o.id, { assignee: e.target.value });
   523→                    }}
   524→                  />
   525→                  {o.wishDate && (

   532→        )}
   533→        {open && loaded && orders.length === 0 && (

   540→          </div>
