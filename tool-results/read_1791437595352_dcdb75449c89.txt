   105→function timeAgo(iso: string): string {
   106→  const diff = Date.now() - new Date(iso).getTime();
   107→  const m = Math.floor(diff / 60000);
   108→  if (m < 1) return 'только что';
   109→  if (m < 60) return `${m} мин назад`;
   110→  const h = Math.floor(m / 60);
   111→  if (h < 24) return `${h} ч назад`;
   112→  return `${Math.floor(h / 24)} дн назад`;
   113→}

   162→/** Панель заявок клиента: просмотр и быстрое изменение статуса/исполнителя */
   163→function ConversationOrdersPanel({
   164→  conversationId,
   165→  botId,
   166→  refreshTick,
   167→}: {
   168→  conversationId: string;
   169→  botId: string;
   170→  refreshTick: number;
   171→}) {
   172→  const { toast } = useToast();
   173→  const [open, setOpen] = useState(false);
   174→  const [orders, setOrders] = useState<ConvOrder[]>([]);
   175→  const [mytkoEnabled, setMytkoEnabled] = useState(false);
   176→  const [loaded, setLoaded] = useState(false);
   177→  const [savingId, setSavingId] = useState<string | null>(null);
   178→
   179→  const load = useCallback(async () => {
   180→    try {
   181→      const d = await api<{ orders: ConvOrder[]; mytkoEnabled?: boolean }>(
   182→        `/api/conversations/${conversationId}/orders`
   183→      );
   184→      setOrders(d.orders);
   185→      setMytkoEnabled(!!d.mytkoEnabled);
   186→    } catch {
   187→      /* ignore */
   188→    } finally {
   189→      setLoaded(true);
   190→    }
   191→  }, [conversationId]);
   192→
   193→  useEffect(() => {
   194→    load();
   195→  }, [load, refreshTick]);
   196→
   197→  const patch = async (orderId: string, data: Record<string, unknown>) => {
   198→    setSavingId(orderId);
   199→    try {
   200→      const d = await api<{ order: { status: string; assignee: string | null } }>(
   201→        `/api/bots/${botId}/orders/${orderId}`,
   202→        { method: 'PATCH', body: JSON.stringify(data) }
   203→      );
   204→      setOrders((prev) =>
   205→        prev.map((o) =>
   206→          o.id === orderId ? { ...o, status: d.order.status, assignee: d.order.assignee } : o
   207→        )
   208→      );
   209→      toast({ title: 'Заявка обновлена' });
   210→    } catch {
   211→      toast({ title: 'Не удалось обновить заявку', variant: 'destructive' });
   212→    } finally {
   213→      setSavingId(null);
   214→    }
   215→  };
   216→
   217→  return (
   218→    <div className="border-b bg-background px-4 py-2">
   219→      <button
   220→        className="flex w-full items-center gap-1.5 rounded text-left text-xs font-medium text-muted-foreground transition-colors hover:text-foreground focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring focus-visible:ring-offset-2"
   221→        onClick={() => setOpen((v) => !v)}
   222→        aria-expanded={open}
   223→        aria-controls="conversation-orders-list"
   224→      >
   225→        {open ? <ChevronDown className="h-3.5 w-3.5" /> : <ChevronRight className="h-3.5 w-3.5" />}
   226→        <Truck className="h-3.5 w-3.5" />
   227→        Заявки клиента <span className="tabular-nums">({orders.length})</span>
   228→      </button>
   229→      <div id="conversation-orders-list">
   230→        {open && loaded && orders.length > 0 && (
   231→          <div className="mt-2 max-h-56 space-y-1.5 overflow-y-auto pr-1">
   232→            {orders.map((o) => (
   233→              <div key={o.id} className="rounded-lg border p-2">
   234→                <div className="flex flex-wrap items-center gap-x-2 gap-y-1">
   235→                  <span className="text-xs font-semibold tabular-nums">№{o.number}</span>
   236→                  <Badge
   237→                    variant="outline"
   238→                    className={cn('h-4 px-1 text-[9px]', ORDER_STATUS_BADGES[o.status])}
   239→                  >
   240→                    {ORDER_STATUS_LABELS[o.status] ?? o.status}
   241→                  </Badge>
   242→                  {mytkoEnabled && <MytkoBadge status={o.mytkoStatus} />}
   243→                  <span className="min-w-0 flex-1 truncate text-xs text-muted-foreground">
   244→                    {ORDER_TYPE_LABELS[o.type] ?? o.type}
   245→                    {o.address ? ` · ${o.address}` : ''}
   246→                  </span>
   247→                </div>
   248→                <div className="mt-1.5 flex flex-wrap items-center gap-1.5">
   249→                  <Select
   250→                    value={o.status}
   251→                    onValueChange={(v) => patch(o.id, { status: v })}
   252→                    disabled={savingId === o.id}
   253→                  >
   254→                    <SelectTrigger className="h-7 w-[130px] text-[11px]" aria-label="Статус заявки">
   255→                      <SelectValue />
   256→                    </SelectTrigger>
   257→                    <SelectContent>
   258→                      {Object.entries(ORDER_STATUS_LABELS).map(([k, v]) => (
   259→                        <SelectItem key={k} value={k} className="text-xs">
   260→                          {v}
   261→                        </SelectItem>
   262→                      ))}
   263→                    </SelectContent>
   264→                  </Select>
   265→                  <Input
   266→                    className="h-7 w-[120px] flex-1 text-[11px] sm:w-[140px]"
   267→                    defaultValue={o.assignee ?? ''}
   268→                    placeholder="Исполнитель…"
   269→                    aria-label="Исполнитель"
   270→                    onBlur={(e) => {
   271→                      if (e.target.value !== (o.assignee ?? '')) patch(o.id, { assignee: e.target.value });
   272→                    }}
   273→                  />
   274→                  {o.wishDate && (
   275→                    <span className="text-[11px] text-muted-foreground">🗓 {o.wishDate}</span>
   276→                  )}
   277→                </div>
   278→              </div>
   279→            ))}
   280→          </div>
   281→        )}
   282→        {open && loaded && orders.length === 0 && (
   283→          <p className="mt-1.5 text-[11px] text-muted-foreground">У клиента пока нет заявок.</p>
   284→        )}
   285→        {open && !loaded && (
   286→          <div className="mt-2 space-y-1.5" role="status" aria-label="Загрузка заявок клиента">
   287→            <Skeleton className="h-[52px] w-full rounded-lg" />
   288→            <Skeleton className="h-[52px] w-full rounded-lg" />
   289→          </div>
   290→        )}
   291→      </div>
   292→    </div>
   293→  );
   294→}

   296→export default function InboxView({
   297→  bot,
   298→  onBack,
   299→  focusConversationId,
   300→  onFocusConsumed,
   301→}: {
   302→  bot: { id: string; name: string };
   303→  onBack: () => void;
   304→  /** id диалога из уведомления — сразу открыть его */
   305→  focusConversationId?: string | null;
   306→  onFocusConsumed?: () => void;
   307→}) {
   308→  const { toast } = useToast();
   309→  const [conversations, setConversations] = useState<ConversationListItem[]>([]);
   310→  const [selectedId, setSelectedId] = useState<string | null>(null);
   311→  const [messages, setMessages] = useState<ChatMessage[]>([]);
   312→  const [contact, setContact] = useState<string | null>(null);
   313→  const [selectedSource, setSelectedSource] = useState('web');
   314→  const [needsOperator, setNeedsOperator] = useState(false);
   315→  const [status, setStatus] = useState('open');
   316→  const [reply, setReply] = useState('');
   317→  const [loading, setLoading] = useState(true);
   318→  const [sending, setSending] = useState(false);
   319→  const [ordersTick, setOrdersTick] = useState(0);
   320→  /** Фильтр списка: все / нужен оператор / новые / закрытые */
   321→  const [filter, setFilter] = useState<'all' | 'operator' | 'unread' | 'closed'>('all');
   322→  const [query, setQuery] = useState('');
   323→  const scrollRef = useRef<HTMLDivElement>(null);
   324→
   325→  const loadList = useCallback(async () => {
   326→    try {
   327→      const d = await api<{ conversations: ConversationListItem[] }>(
   328→        `/api/bots/${bot.id}/conversations`
   329→      );
   330→      setConversations(d.conversations);
   331→    } catch {
   332→      /* ignore polling errors */
   333→    } finally {
   334→      setLoading(false);
   335→    }
   336→  }, [bot.id]);
   337→
   338→  const loadConversation = useCallback(async (id: string) => {
   339→    try {
   340→      const d = await api<{
   341→        conversation: { contact: string | null; source: string; needsOperator: boolean; status: string };
   342→        messages: ChatMessage[];
   343→      }>(`/api/conversations/${id}`);
   344→      setMessages(d.messages);
   345→      setContact(d.conversation.contact);
   346→      setSelectedSource(d.conversation.source);
   347→      setNeedsOperator(d.conversation.needsOperator);
   348→      setStatus(d.conversation.status);
   349→    } catch {
   350→      /* ignore */
   351→    }
   352→  }, []);
   353→
   354→  /** Открыть диалог: выделяем и сразу помечаем прочитанным (сервер запоминает operatorReadAt) */
   355→  const openConversation = useCallback((id: string) => {
   356→    setSelectedId(id);
   357→    setConversations((prev) => prev.map((c) => (c.id === id ? { ...c, unread: false } : c)));
   358→  }, []);
   359→
   360→  useEffect(() => {
   361→    loadList();
   362→  }, [loadList]);
   363→
   364→  // Переход из уведомления: сразу открываем нужный диалог
   365→  useEffect(() => {
   366→    if (!focusConversationId) return;
   367→    openConversation(focusConversationId);
   368→    onFocusConsumed?.();
   369→  }, [focusConversationId, openConversation, onFocusConsumed]);
   370→
   371→  useEffect(() => {
   372→    const t = setInterval(loadList, 6000);
   373→    return () => clearInterval(t);
   374→  }, [loadList]);
   375→
   376→  useEffect(() => {
   377→    if (!selectedId) return;
   378→    loadConversation(selectedId);
   379→    const t = setInterval(() => loadConversation(selectedId), 4000);
   380→    return () => clearInterval(t);
   381→  }, [selectedId, loadConversation]);
   382→
   383→  useEffect(() => {
   384→    requestAnimationFrame(() => {
   385→      scrollRef.current?.scrollTo({ top: scrollRef.current.scrollHeight });
   386→    });
   387→  }, [messages]);
   388→
   389→  const sendReply = async () => {
   390→    const t = reply.trim();
   391→    if (!t || !selectedId) return;
   392→    setSending(true);
   393→    setReply('');
   394→    try {
   395→      const d = await api<{ message: ChatMessage }>(`/api/conversations/${selectedId}/operator`, {
   396→        method: 'POST',
   397→        body: JSON.stringify({ text: t }),
   398→      });
   399→      setMessages((prev) => [...prev, d.message]);
   400→    } catch {
   401→      toast({ title: 'Не удалось отправить', variant: 'destructive' });
   402→    } finally {
   403→      setSending(false);
   404→    }
   405→  };
   406→
   407→  const closeConversation = async () => {
   408→    if (!selectedId) return;
   409→    try {
   410→      await api(`/api/conversations/${selectedId}/close`, { method: 'POST' });
   411→      setNeedsOperator(false);
   412→      setStatus('closed');
   413→      toast({ title: 'Обращение закрыто' });
   414→      loadList();
   415→    } catch {
   416→      toast({ title: 'Ошибка', variant: 'destructive' });
   417→    }
   418→  };
   419→
   420→  const selected = conversations.find((c) => c.id === selectedId) ?? null;
   421→
   422→  const operatorCount = useMemo(
   423→    () => conversations.filter((c) => c.needsOperator && c.status === 'open').length,
   424→    [conversations]
   425→  );
   426→  const unreadCount = useMemo(
   427→    () => conversations.filter((c) => c.unread && c.status === 'open').length,
   428→    [conversations]
   429→  );
   430→  const closedCount = useMemo(
   431→    () => conversations.filter((c) => c.status === 'closed').length,
   432→    [conversations]
   433→  );
   434→
   435→  /** Сортировка: нуждаются в операторе → непрочитанные → по времени; плюс поиск и фильтр */
   436→  const visibleList = useMemo(() => {
   437→    const q = query.trim().toLowerCase();
   438→    let list = conversations.filter((c) => {
   439→      if (filter === 'operator') return c.needsOperator && c.status === 'open';
   440→      if (filter === 'unread') return c.unread && c.status === 'open';
   441→      if (filter === 'closed') return c.status === 'closed';
   442→      return true;
   443→    });
   444→    if (q) {
   445→      list = list.filter(
   446→        (c) =>
   447→          (c.contact ?? '').toLowerCase().includes(q) ||
   448→          (c.lastMessage?.text ?? '').toLowerCase().includes(q)
   449→      );
   450→    }
   451→    const weight = (c: ConversationListItem) =>
   452→      c.needsOperator && c.status === 'open' ? 0 : c.unread && c.status === 'open' ? 1 : 2;
   453→    return [...list].sort((a, b) => {
   454→      const w = weight(a) - weight(b);
   455→      if (w !== 0) return w;
   456→      return new Date(b.updatedAt).getTime() - new Date(a.updatedAt).getTime();
   457→    });
   458→  }, [conversations, filter, query]);
   459→
   460→  return (
   461→    <div className="flex min-h-0 flex-1 flex-col">
   462→      <div className="flex items-center gap-2 border-b bg-background px-3 py-2.5 sm:gap-3 sm:px-4 sm:py-3">
   463→        <Button
   464→          variant="ghost"
   465→          size="icon"
   466→          className="h-8 w-8 shrink-0 sm:h-9 sm:w-9"
   467→          onClick={onBack}
   468→          aria-label="Назад"
   469→        >
   470→          <ChevronLeft className="h-4 w-4" />
   471→        </Button>
   472→        <div className="flex min-w-0 flex-1 items-center gap-2">
   473→          <Inbox className="h-5 w-5 shrink-0 text-primary" aria-hidden />
   474→          <h1 className="shrink-0 text-lg font-bold tracking-tight sm:text-2xl">Входящие</h1>
   475→          <Badge variant="secondary" className="min-w-0 max-w-[110px] truncate sm:max-w-[180px]">
   476→            {bot.name}
   477→          </Badge>
   478→          {operatorCount > 0 && (
   479→            <Badge variant="destructive" className="shrink-0 animate-pulse tabular-nums">
   480→              <Headset className="mr-1 h-3 w-3" />
   481→              <span className="hidden sm:inline">оператор: </span>
   482→              {operatorCount}
   483→            </Badge>
   484→          )}
   485→          {unreadCount > 0 && (
   486→            <Badge className="shrink-0 bg-primary/15 text-primary tabular-nums hover:bg-primary/15">
   487→              <span className="hidden sm:inline">новые: </span>
   488→              {unreadCount}
   489→            </Badge>
   490→          )}
   491→        </div>
   492→      </div>
   493→
   494→      <div className="grid min-h-0 flex-1 md:grid-cols-[340px_1fr]">
   495→        {/* Список диалогов */}
   496→        <div className={cn('flex min-h-0 flex-col border-r', selected && 'hidden md:flex')}>
   497→          {/* Фильтры: все / нужен оператор / новые / закрытые + поиск */}
   498→          <div className="flex flex-wrap items-center gap-1.5 border-b bg-background px-2.5 py-2 sm:px-3">
   499→            {(
   500→              [
   501→                { key: 'all', label: 'Все', count: conversations.length },
   502→                { key: 'operator', label: 'Требуют внимания', count: operatorCount },
   503→                { key: 'unread', label: 'Новые', count: unreadCount },
   504→                { key: 'closed', label: 'Закрытые', count: closedCount },
   505→              ] as const
   506→            ).map((f) => (
   507→              <button
   508→                key={f.key}
   509→                type="button"
   510→                onClick={() => setFilter(f.key)}
   511→                aria-pressed={filter === f.key}
   512→                className={cn(
   513→                  'h-8 rounded-full px-3 text-xs font-medium transition-colors',
