     1→'use client';
     2→
     3→import { useCallback, useEffect, useRef, useState } from 'react';
     4→import {
     5→  BookOpen,
     6→  Bot,
     7→  BrainCircuit,
     8→  ChevronLeft,
     9→  Loader2,
    10→  Pencil,
    11→  Plus,
    12→  RotateCcw,
    13→  Save,
    14→  Send,
    15→  Sparkles,
    16→  Trash2,
    17→  X,
    18→} from 'lucide-react';
    19→import { api } from '@/lib/client-api';
    20→import type { AiConfig, KnowledgeItemDto } from '@/lib/studio-types';
    21→import type { EngineState } from '@/lib/flow-types';
    22→import { Badge } from '@/components/ui/badge';
    23→import { Button } from '@/components/ui/button';
    24→import {
    25→  Card,
    26→  CardContent,
    27→  CardDescription,
    28→  CardHeader,
    29→  CardTitle,
    30→} from '@/components/ui/card';
    31→import { Input } from '@/components/ui/input';
    32→import { Label } from '@/components/ui/label';
    33→import { Skeleton } from '@/components/ui/skeleton';
    34→import { Switch } from '@/components/ui/switch';
    35→import { Textarea } from '@/components/ui/textarea';
    36→import { useToast } from '@/hooks/use-toast';
    37→import { cn } from '@/lib/utils';
    38→
    39→const DEFAULT_PROMPT =
    40→  'Ты — дружелюбный ИИ-ассистент техподдержки. Отвечай кратко, вежливо и по делу, на языке пользователя. Опирайся на базу знаний; если ответа нет — честно скажи об этом и предложи позвать оператора.';
    41→
    42→interface ChatItem {
    43→  id: string;
    44→  role: 'user' | 'bot';
    45→  text: string;
    46→  buttons?: { id: string; text: string }[];
    47→}
    48→
    49→export default function AiAssistantView({
    50→  bot,
    51→  onBack,
    52→  onOpenInbox,
    53→}: {
    54→  bot: { id: string; name: string; status: string };
    55→  onBack: () => void;
    56→  onOpenInbox: () => void;
    57→}) {
    58→  const { toast } = useToast();
    59→
    60→  // ── Настройки ассистента ──
    61→  const [cfg, setCfg] = useState<AiConfig>({ enabled: false, prompt: '', reaskMenu: true });
    62→  const [savedCfg, setSavedCfg] = useState<AiConfig>({ enabled: false, prompt: '', reaskMenu: true });
    63→  const [loaded, setLoaded] = useState(false);
    64→  const [savingCfg, setSavingCfg] = useState(false);
    65→
    66→  // ── База знаний ──
    67→  const [knowledge, setKnowledge] = useState<KnowledgeItemDto[]>([]);
    68→  const [editing, setEditing] = useState<{ id: string | null; title: string; content: string } | null>(null);
    69→  const [kbBusy, setKbBusy] = useState(false);
    70→
    71→  // ── Тест-чат ──
    72→  const [chat, setChat] = useState<ChatItem[]>([]);
    73→  const [chatState, setChatState] = useState<EngineState | null>(null);
    74→  const [input, setInput] = useState('');
    75→  const [sending, setSending] = useState(false);
    76→  const [handedOff, setHandedOff] = useState<string | null>(null);
    77→  const scrollRef = useRef<HTMLDivElement>(null);
    78→
    79→  const dirty = JSON.stringify(cfg) !== JSON.stringify(savedCfg);
    80→
    81→  const load = useCallback(async () => {
    82→    try {
    83→      const d = await api<{ aiConfig: AiConfig; knowledge: KnowledgeItemDto[] }>(
    84→        `/api/bots/${bot.id}/ai`
    85→      );
    86→      const c: AiConfig = {
    87→        enabled: d.aiConfig.enabled === true,
    88→        prompt: d.aiConfig.prompt ?? '',
    89→        reaskMenu: d.aiConfig.reaskMenu !== false,
    90→      };
    91→      setCfg(c);
    92→      setSavedCfg(c);
    93→      setKnowledge(d.knowledge);
    94→    } catch {
    95→      toast({ title: 'Не удалось загрузить настройки ассистента', variant: 'destructive' });
    96→    } finally {
    97→      setLoaded(true);
    98→    }
    99→  }, [bot.id, toast]);
   100→
   101→  useEffect(() => {
   102→    load();
   103→  }, [load]);
   104→
   105→  useEffect(() => {
   106→    scrollRef.current?.scrollTo({ top: scrollRef.current.scrollHeight });
   107→  }, [chat, sending]);
   108→
   109→  const saveCfg = async () => {
   110→    setSavingCfg(true);
   111→    try {
   112→      const d = await api<{ aiConfig: AiConfig }>(`/api/bots/${bot.id}/ai`, {
   113→        method: 'PUT',
   114→        body: JSON.stringify(cfg),
   115→      });
   116→      const c: AiConfig = {
   117→        enabled: d.aiConfig.enabled === true,
   118→        prompt: d.aiConfig.prompt ?? '',
   119→        reaskMenu: d.aiConfig.reaskMenu !== false,
   120→      };
   121→      setSavedCfg(c);
   122→      setCfg(c);
   123→      toast({
   124→        title: c.enabled ? 'ИИ-ассистент включён' : 'Настройки сохранены',
   125→        description: c.enabled
   126→          ? 'Бот отвечает на свободные вопросы во всех каналах.'
   127→          : 'Ассистент работает только по сценарию.',
   128→      });
   129→    } catch {
   130→      toast({ title: 'Ошибка сохранения', variant: 'destructive' });
   131→    } finally {
   132→      setSavingCfg(false);
   133→    }
   134→  };
   135→
   136→  // ── База знаний: CRUD ──
   137→  const saveItem = async () => {
   138→    if (!editing) return;
   139→    if (!editing.title.trim() || !editing.content.trim()) {
   140→      toast({ title: 'Укажите заголовок и содержимое', variant: 'destructive' });
   141→      return;
   142→    }
   143→    setKbBusy(true);
   144→    try {
   145→      if (editing.id) {
   146→        const d = await api<{ item: KnowledgeItemDto }>(`/api/knowledge/${editing.id}`, {
   147→          method: 'PUT',
   148→          body: JSON.stringify({ title: editing.title, content: editing.content }),
   149→        });
   150→        setKnowledge((prev) => prev.map((k) => (k.id === d.item.id ? d.item : k)));
   151→      } else {
   152→        const d = await api<{ item: KnowledgeItemDto }>(`/api/bots/${bot.id}/knowledge`, {
   153→          method: 'POST',
   154→          body: JSON.stringify({ title: editing.title, content: editing.content }),
   155→        });
   156→        setKnowledge((prev) => [...prev, d.item]);
   157→      }
   158→      setEditing(null);
   159→    } catch {
   160→      toast({ title: 'Не удалось сохранить запись', variant: 'destructive' });
   161→    } finally {
   162→      setKbBusy(false);
   163→    }
   164→  };
   165→
   166→  const deleteItem = async (id: string) => {
   167→    setKbBusy(true);
   168→    try {
   169→      await api(`/api/knowledge/${id}`, { method: 'DELETE' });
   170→      setKnowledge((prev) => prev.filter((k) => k.id !== id));
   171→      if (editing?.id === id) setEditing(null);
   172→    } catch {
   173→      toast({ title: 'Не удалось удалить запись', variant: 'destructive' });
   174→    } finally {
   175→      setKbBusy(false);
   176→    }
   177→  };
   178→
   179→  // ── Тест-чат ──
   180→  const send = async (text: string) => {
   181→    const t = text.trim();
   182→    if (!t || sending) return;
   183→    setSending(true);
   184→    setInput('');
   185→    setChat((prev) => [...prev, { id: `u-${Date.now()}`, role: 'user', text: t }]);
   186→    try {
   187→      const d = await api<{
   188→        messages: { text: string; nodeId: string; buttons?: { id: string; text: string }[] }[];
   189→        state: EngineState;
   190→        needsOperator: boolean;
   191→        conversationId?: string;
   192→      }>(`/api/bots/${bot.id}/simulate`, {
   193→        method: 'POST',
   194→        body: JSON.stringify({ input: t, state: chatState, conversationId: handedOff ?? undefined }),
   195→      });
   196→      setChatState(d.state);
   197→      setChat((prev) => [
   198→        ...prev,
   199→        ...d.messages.map((m, i) => ({
   200→          id: `b-${Date.now()}-${i}`,
   201→          role: 'bot' as const,
   202→          text: m.text,
   203→          buttons: m.buttons,
   204→        })),
   205→      ]);
   206→      if (d.needsOperator) setHandedOff(d.conversationId ?? 'operator');
   207→    } catch {
   208→      setChat((prev) => [
   209→        ...prev,
   210→        { id: `e-${Date.now()}`, role: 'bot', text: 'Ошибка: не удалось получить ответ. Попробуйте ещё раз.' },
   211→      ]);
   212→    } finally {
   213→      setSending(false);
   214→    }
   215→  };
   216→
   217→  const resetChat = () => {
   218→    setChat([]);
   219→    setChatState(null);
   220→    setHandedOff(null);
   221→    setInput('');
   222→  };
   223→
   224→  const kbSuggestions = knowledge.slice(0, 3).map((k) => k.title);
   225→
   226→  if (!loaded) {
   227→    return (
   228→      <div
   229→        className="min-h-0 flex-1 space-y-6 overflow-y-auto p-4 sm:p-6 lg:p-8"
   230→        role="status"
   231→        aria-label="Загрузка настроек ассистента"
   232→      >
   233→        <div className="flex items-center gap-3">
   234→          <Skeleton className="h-8 w-8 rounded-lg" />
   235→          <div className="min-w-0 flex-1 space-y-2">
   236→            <Skeleton className="h-7 w-56" />
   237→            <Skeleton className="h-4 w-80 max-w-full" />
   238→          </div>
   239→        </div>
   240→        <div className="grid gap-6 xl:grid-cols-5">
   241→          <div className="space-y-6 xl:col-span-3">
   242→            <Skeleton className="h-64 rounded-xl" />
   243→            <Skeleton className="h-72 rounded-xl" />
   244→          </div>
   245→          <Skeleton className="h-[480px] rounded-xl xl:col-span-2" />
   246→        </div>
   247→      </div>
   248→    );
   249→  }
   250→
   251→  return (
   252→    <div className="min-h-0 flex-1 space-y-6 overflow-y-auto p-4 sm:p-6 lg:p-8">
   253→      {/* Заголовок */}
   254→      <div className="flex flex-wrap items-center justify-between gap-4">
   255→        <div className="flex items-center gap-3">
   256→          <Button variant="ghost" size="icon" className="h-8 w-8" onClick={onBack} aria-label="Назад">
   257→            <ChevronLeft className="h-4 w-4" />
   258→          </Button>
   259→          <div className="min-w-0">
   260→            <h1 className="flex items-center gap-2 text-2xl font-bold tracking-tight">
   261→              <Sparkles className="h-6 w-6 shrink-0 text-primary" aria-hidden /> ИИ-ассистент
   262→            </h1>
   263→            <p className="truncate text-sm text-muted-foreground">
   264→              Нейросеть отвечает на свободные вопросы клиентов бота «{bot.name}»
   265→            </p>
   266→          </div>
   267→        </div>
   268→        {bot.status !== 'published' && (
   269→          <Badge
   270→            variant="outline"
   271→            className="border-amber-200 bg-amber-50 text-amber-800 dark:border-amber-500/20 dark:bg-amber-500/10 dark:text-amber-300"
   272→          >
   273→            Бот не опубликован — ассистент работает только в тестах
   274→          </Badge>
   275→        )}
   276→      </div>
   277→
   278→      <div className="grid gap-6 xl:grid-cols-5">
   279→        {/* Левая колонка: настройки + база знаний */}
   280→        <div className="space-y-6 xl:col-span-3">
   281→          {/* Настройки */}
   282→          <Card>
   283→            <CardHeader className="pb-3">
   284→              <CardTitle className="flex items-center gap-2 text-lg">
   285→                <BrainCircuit className="h-5 w-5 shrink-0 text-primary" aria-hidden /> Настройки ассистента
   286→              </CardTitle>
   287→              <CardDescription>
   288→                Когда сценарию нечего ответить (свободный текст вместо кнопки, вопрос вне сценария),
   289→                на вопрос отвечает ИИ с учётом базы знаний и истории диалога.
   290→              </CardDescription>
   291→            </CardHeader>
   292→            <CardContent className="space-y-4">
   293→              <div className="flex items-center justify-between gap-3 rounded-lg border p-3">
   294→                <div className="min-w-0">
   295→                  <Label htmlFor="ai-enabled" className="text-sm font-medium">
   296→                    Включить ИИ-ассистента
   297→                  </Label>
   298→                  <p className="text-[11px] text-muted-foreground">
   299→                    Клиент может задать любой вопрос — бот ответит нейросетью
   300→                  </p>
   301→                </div>
   302→                <Switch
   303→                  id="ai-enabled"
   304→                  checked={cfg.enabled}
   305→                  onCheckedChange={(v) => setCfg((c) => ({ ...c, enabled: v }))}
   306→                />
   307→              </div>
   308→
   309→              {cfg.enabled && (
   310→                <>
   311→                  <div className="space-y-1.5">
   312→                    <Label htmlFor="ai-prompt">Личность и тон ассистента</Label>
   313→                    <Textarea
   314→                      id="ai-prompt"
   315→                      rows={4}
   316→                      placeholder={DEFAULT_PROMPT}
   317→                      value={cfg.prompt}
   318→                      onChange={(e) => setCfg((c) => ({ ...c, prompt: e.target.value }))}
   319→                    />
   320→                    <div className="flex flex-wrap items-center justify-between gap-2">
   321→                      <p className="text-[11px] text-muted-foreground">
   322→                        Инструкция для нейросети: кто вы, как отвечать, чего избегать.
   323→                      </p>
   324→                      <Button
   325→                        variant="ghost"
   326→                        size="sm"
   327→                        className="h-7 text-[11px] text-muted-foreground"
   328→                        onClick={() => setCfg((c) => ({ ...c, prompt: DEFAULT_PROMPT }))}
   329→                      >
   330→                        <RotateCcw className="h-3 w-3" /> Шаблон
   331→                      </Button>
   332→                    </div>
   333→                  </div>
   334→
   335→                  <div className="flex items-center justify-between gap-3 rounded-lg border p-3">
   336→                    <div className="min-w-0">
   337→                      <Label htmlFor="ai-reask" className="text-sm font-medium">
   338→                        Показывать меню после ответа
   339→                      </Label>
   340→                      <p className="text-[11px] text-muted-foreground">
   341→                        После ответа ИИ пользователь снова видит кнопки сценария
   342→                      </p>
   343→                    </div>
   344→                    <Switch
   345→                      id="ai-reask"
   346→                      checked={cfg.reaskMenu}
   347→                      onCheckedChange={(v) => setCfg((c) => ({ ...c, reaskMenu: v }))}
   348→                    />
   349→                  </div>
   350→
   351→                  <div className="rounded-lg border bg-muted/40 p-3 text-[11px] leading-relaxed text-muted-foreground">
   352→                    <b className="text-foreground">Как это работает:</b> вопрос клиента не попадает
   353→                    на кнопку сценария → ассистент отвечает по базе знаний → клиент возвращается в
   354→                    меню. Просит живого человека → диалог уходит оператору. В блоках «ИИ-ответ»
   355→                    конструктора тоже используется эта база знаний.
   356→                  </div>
   357→                </>
   358→              )}
   359→
   360→              <div className="flex flex-wrap items-center gap-3">
   361→                <Button onClick={saveCfg} disabled={savingCfg || !dirty}>
   362→                  {savingCfg ? <Loader2 className="h-4 w-4 animate-spin" /> : <Save className="h-4 w-4" />}
   363→                  Сохранить настройки
   364→                </Button>
   365→                {dirty && (
   366→                  <span className="text-xs text-amber-600 dark:text-amber-400">
   367→                    Есть несохранённые изменения
   368→                  </span>
   369→                )}
   370→              </div>
   371→            </CardContent>
   372→          </Card>
   373→
   374→          {/* База знаний */}
   375→          <Card>
   376→            <CardHeader className="pb-3">
   377→              <div className="flex items-center justify-between gap-2">
   378→                <div className="min-w-0">
   379→                  <CardTitle className="flex items-center gap-2 text-lg">
   380→                    <BookOpen className="h-5 w-5 shrink-0 text-primary" aria-hidden /> База знаний
   381→                  </CardTitle>
   382→                  <CardDescription>
   383→                    Факты о компании: графики, цены, доставка, возвраты, контакты…
   384→                  </CardDescription>
   385→                </div>
   386→                <Button
   387→                  size="sm"
   388→                  variant="outline"
   389→                  onClick={() => setEditing({ id: null, title: '', content: '' })}
   390→                  disabled={kbBusy}
   391→                >
   392→                  <Plus className="h-4 w-4" /> Добавить
   393→                </Button>
   394→              </div>
   395→            </CardHeader>
   396→            <CardContent className="space-y-3">
   397→              {knowledge.length === 0 && !editing && (
   398→                <div className="flex flex-col items-center gap-3 rounded-lg border border-dashed p-6 text-center">
   399→                  <div className="flex h-12 w-12 items-center justify-center rounded-xl bg-primary/10 text-primary">
   400→                    <BookOpen className="h-6 w-6" aria-hidden />
   401→                  </div>
   402→                  <div>
   403→                    <p className="text-sm font-medium text-foreground">Записей пока нет</p>
   404→                    <p className="mt-1 max-w-[320px] text-sm text-muted-foreground">
   405→                      Добавьте ответы на частые вопросы — ассистент будет опираться на них при
   406→                      ответах.
   407→                    </p>
   408→                  </div>
   409→                  <Button
   410→                    size="sm"
   411→                    variant="outline"
   412→                    disabled={kbBusy}
   413→                    onClick={() => setEditing({ id: null, title: '', content: '' })}
   414→                  >
   415→                    <Plus className="h-4 w-4" /> Добавить запись
   416→                  </Button>
   417→                </div>
   418→              )}
   419→
   420→              {editing && (
   421→                <div className="space-y-3 rounded-lg border border-primary/40 bg-primary/5 p-4">
   422→                  <div className="flex items-center justify-between gap-2">
   423→                    <Label className="text-sm font-medium">
   424→                      {editing.id ? 'Изменить запись' : 'Новая запись'}
   425→                    </Label>
   426→                    <Button
   427→                      variant="ghost"
   428→                      size="icon"
   429→                      className="h-7 w-7"
   430→                      onClick={() => setEditing(null)}
   431→                      aria-label="Отменить редактирование"
   432→                    >
   433→                      <X className="h-4 w-4" />
   434→                    </Button>
   435→                  </div>
   436→                  <Input
   437→                    placeholder="Заголовок: например «График работы»"
   438→                    aria-label="Заголовок записи"
   439→                    value={editing.title}
   440→                    onChange={(e) => setEditing({ ...editing, title: e.target.value })}
   441→                  />
   442→                  <Textarea
   443→                    rows={4}
   444→                    placeholder="Содержимое: пн–пт с 9:00 до 18:00 МСК…"
   445→                    aria-label="Содержимое записи"
   446→                    value={editing.content}
   447→                    onChange={(e) => setEditing({ ...editing, content: e.target.value })}
   448→                  />
   449→                  <div className="flex gap-2">
   450→                    <Button size="sm" onClick={saveItem} disabled={kbBusy}>
   451→                      {kbBusy ? <Loader2 className="h-4 w-4 animate-spin" /> : <Save className="h-4 w-4" />}
   452→                      Сохранить
   453→                    </Button>
   454→                    <Button size="sm" variant="outline" onClick={() => setEditing(null)}>
   455→                      Отмена
   456→                    </Button>
   457→                  </div>
   458→                </div>
   459→              )}
   460→
   461→              {knowledge.length > 0 && (
   462→                <div
   463→                  className="max-h-96 overflow-y-auto rounded-lg border"
   464→                  role="list"
   465→                  aria-label="Записи базы знаний"
   466→                >
   467→                  <div className="divide-y">
   468→                    {knowledge.map((k) => (
   469→                      <div
   470→                        key={k.id}
   471→                        role="listitem"
   472→                        className={cn(
   473→                          'group flex items-start justify-between gap-2 p-3',
   474→                          editing?.id === k.id && 'hidden'
   475→                        )}
   476→                      >
   477→                        <div className="min-w-0">
   478→                          <div className="truncate text-sm font-medium">{k.title}</div>
   479→                          <p className="mt-0.5 line-clamp-2 whitespace-pre-line text-xs text-muted-foreground">
   480→                            {k.content}
   481→                          </p>
   482→                        </div>
   483→                        <div className="flex shrink-0 gap-1 opacity-70 transition-opacity focus-within:opacity-100 group-hover:opacity-100">
   484→                          <Button
   485→                            variant="ghost"
   486→                            size="icon"
   487→                            className="h-8 w-8"
   488→                            aria-label={`Изменить запись «${k.title}»`}
   489→                            onClick={() =>
   490→                              setEditing({ id: k.id, title: k.title, content: k.content })
   491→                            }
   492→                          >
   493→                            <Pencil className="h-3.5 w-3.5" />
   494→                          </Button>
   495→                          <Button
   496→                            variant="ghost"
   497→                            size="icon"
   498→                            className="h-8 w-8 text-muted-foreground hover:text-destructive"
   499→                            aria-label={`Удалить запись «${k.title}»`}
   500→                            onClick={() => deleteItem(k.id)}
   501→                          >
   502→                            <Trash2 className="h-3.5 w-3.5" />
   503→                          </Button>
   504→                        </div>
   505→                      </div>
   506→                    ))}
   507→                  </div>
   508→                </div>
   509→              )}
   510→            </CardContent>
   511→          </Card>
   512→        </div>
   513→
   514→        {/* Правая колонка: тест-чат */}
   515→        <Card className="xl:col-span-2">
