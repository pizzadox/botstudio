    69→      return 'bg-amber-500';
    70→    default:
    71→      return 'bg-violet-500';
    72→  }

    84→// ─── Карта OpenFreeMap (MapLibre GL, бесплатные векторные тайлы без ключа) ───

    86→function OrdersMap({
    87→  orders,
    88→  showCompleted,
    89→  placementOrderId,
    90→  onSelect,
    91→  onPlace,
    92→}: {
    93→  orders: OrderDto[];
    94→  showCompleted: boolean;
    95→  placementOrderId: string | null;
    96→  onSelect: (id: string) => void;
    97→  onPlace: (lat: number, lng: number) => void;
    98→}) {
    99→  const containerRef = useRef<HTMLDivElement>(null);
   100→  const mapRef = useRef<MlMap | null>(null);
   101→  const mlRef = useRef<typeof import('maplibre-gl') | null>(null);
   102→  const markersRef = useRef<MlMarker[]>([]);
   103→  const [ready, setReady] = useState(false);
   104→  const [failed, setFailed] = useState(false);
   105→  const [zoomKey, setZoomKey] = useState(0);
   106→
   107→  // Инициализация карты (динамический импорт — компонент клиентский)
   108→  useEffect(() => {
   109→    let cancelled = false;
   110→    (async () => {
   111→      try {
   112→        const ml = await import('maplibre-gl');
   113→        if (cancelled || !containerRef.current) return;
   114→        // Воркер MapLibre не собирается Turbopack'ом — берём локальную копию из public/
   115→        ml.setWorkerUrl('/maplibre-gl-worker.mjs');
   116→        mlRef.current = ml;
   117→        const map = new ml.Map({
   118→          container: containerRef.current,
   119→          style: 'https://tiles.openfreemap.org/styles/liberty',
   120→          center: [37.6176, 55.7558], // Москва — стартовый вид
   121→          zoom: 8.5,
   122→        });
   123→        map.addControl(new ml.NavigationControl({ showCompass: false }), 'top-right');
   124→        map.on('load', () => {
   125→          if (!cancelled) setReady(true);
   126→        });
   127→        map.on('error', (e) => {
   128→          // Тайловая сеть недоступна — сообщаем, но карту не убиваем
   129→          console.warn('[orders map]', e?.error?.message ?? e);
   130→        });
   131→        mapRef.current = map;
   132→      } catch (err) {
   133→        console.error('[orders map] init failed', err);
   134→        if (!cancelled) setFailed(true);
   135→      }
   136→    })();
   137→    return () => {
   138→      cancelled = true;
   139→      mapRef.current?.remove();
   140→      mapRef.current = null;
   141→      setReady(false);
   142→    };
   143→  }, []);
   144→
   145→  // Маркеры заявок
   146→  const visible = orders.filter(
   147→    (o) => o.lat != null && o.lng != null && (showCompleted || ACTIVE_STATUSES.includes(o.status))
   148→  );
   149→
   150→  useEffect(() => {
   151→    const map = mapRef.current;
   152→    const ml = mlRef.current;
   153→    if (!map || !ml || !ready) return;
   154→    for (const m of markersRef.current) m.remove();
   155→    markersRef.current = [];
   156→    for (const o of visible) {
   157→      const el = document.createElement('button');
   158→      el.type = 'button';
   159→      el.title = `Заявка №${o.number} — ${ORDER_TYPE_LABELS[o.type] ?? ''}`;
   160→      el.className = cn(
   161→        'flex h-8 w-8 cursor-pointer items-center justify-center rounded-full border-2 border-white text-[11px] font-bold text-white shadow-lg transition-transform hover:scale-110',
   162→        markerColor(o)
   163→      );
   164→      el.textContent = String(o.number);
   165→      el.addEventListener('click', (e) => {
   166→        e.stopPropagation();
   167→        onSelect(o.id);
   168→      });
   169→      const marker = new ml.Marker({ element: el })
   170→        .setLngLat([o.lng as number, o.lat as number])
   171→        .addTo(map);
   172→      markersRef.current.push(marker);
   173→    }
   174→    // зависимость — сериализованный список видимых маркеров
   175→  }, [ready, zoomKey, visible.map((o) => `${o.id}:${o.lat}:${o.lng}:${o.status}`).join('|')]);
   176→
   177→  // Режим указания точки: клик по карте → координаты заявки
   178→  const handleMapClick = useCallback(
   179→    (lat: number, lng: number) => onPlace(lat, lng),
   180→    [onPlace]
   181→  );
   182→  useEffect(() => {
   183→    const map = mapRef.current;
   184→    if (!map || !ready || !placementOrderId) return;
   185→    const onClick = (e: { lngLat: { lat: number; lng: number } }) =>
   186→      handleMapClick(e.lngLat.lat, e.lngLat.lng);
   187→    map.on('click', onClick);
   188→    map.getCanvas().style.cursor = 'crosshair';
   189→    return () => {
   190→      map.off('click', onClick);
   191→      map.getCanvas().style.cursor = '';
   192→    };
   193→  }, [ready, placementOrderId, handleMapClick]);
   194→
   195→  if (failed) {
   196→    return (
   197→      <div className="flex h-full min-h-[420px] flex-col items-center justify-center gap-3 rounded-xl border bg-muted/30 text-center text-muted-foreground">
   198→        <MapPin className="h-10 w-10 opacity-30" />
   199→        <p className="max-w-xs text-sm">
   200→          Не удалось загрузить карту OpenFreeMap. Проверьте доступ к tiles.openfreemap.org и
   201→          обновите страницу.
   202→        </p>
   203→      </div>
   204→    );
   205→  }
   206→
   207→  return (
   208→    <div className="relative h-full min-h-[420px] overflow-hidden rounded-xl border bg-muted/30">
   209→      <div ref={containerRef} className="h-full w-full" aria-label="Карта заявок" />
   210→      {!ready && (
   211→        <div className="absolute inset-0 flex items-center justify-center gap-2 text-sm text-muted-foreground">
   212→          <Loader2 className="h-4 w-4 animate-spin" /> Загружаю карту OpenFreeMap…
   213→        </div>
   214→      )}
   215→      {/* Легенда */}
   216→      <div className="absolute bottom-3 left-3 flex flex-col gap-1 rounded-lg border bg-background/95 px-3 py-2 text-[11px] shadow-sm backdrop-blur">
   217→        <span className="flex items-center gap-1.5">
   218→          <span className="h-2.5 w-2.5 rounded-full bg-emerald-500" /> вывоз отходов
   219→        </span>
   220→        <span className="flex items-center gap-1.5">
   221→          <span className="h-2.5 w-2.5 rounded-full bg-amber-500" /> КГМ
   222→        </span>
   223→        <span className="flex items-center gap-1.5">
   224→          <span className="h-2.5 w-2.5 rounded-full bg-slate-400" /> выполненные
   225→        </span>
   226→      </div>
   227→      {/* Счётчик маркеров */}
   228→      <div className="absolute left-3 top-3 rounded-lg border bg-background/95 px-3 py-1.5 text-xs shadow-sm backdrop-blur">
   229→        <MapPin className="mr-1 inline h-3 w-3 text-primary" />
   230→        На карте: <b>{visible.length}</b>
   231→      </div>
   232→      {placementOrderId && (
   233→        <div className="absolute inset-x-3 top-3 z-10 flex items-center justify-between gap-2 rounded-lg border border-primary/40 bg-primary/95 px-3 py-2 text-xs text-primary-foreground shadow-md md:right-auto">
   234→          <span className="flex items-center gap-1.5">
   235→            <Crosshair className="h-3.5 w-3.5" /> Кликните по карте — точка заявки №
   236→            {placementOrderId}
   237→          </span>
   238→        </div>
   239→      )}
   240→    </div>
   241→  );
   242→}

   246→function OrderChat({
   247→  botId,
   248→  orderId,
   249→  hasConversation,
   250→}: {
   251→  botId: string;
   252→  orderId: string;
   253→  hasConversation: boolean;
   254→}) {
   255→  const { toast } = useToast();
   256→  const [messages, setMessages] = useState<OrderMessageDto[]>([]);
   257→  const [text, setText] = useState('');
   258→  const [sending, setSending] = useState(false);
   259→  const scrollRef = useRef<HTMLDivElement>(null);
   260→
   261→  const load = useCallback(async () => {
   262→    try {
   263→      const d = await api<{ messages: OrderMessageDto[] }>(
   264→        `/api/bots/${botId}/orders/${orderId}/messages`
   265→      );
   266→      setMessages(d.messages);
   267→    } catch {
   268→      /* polling errors ignored */
   269→    }
   270→  }, [botId, orderId]);
   271→
   272→  useEffect(() => {
   273→    setMessages([]);
   274→    load();
   275→    const t = setInterval(load, 3000);
   276→    return () => clearInterval(t);
   277→  }, [load]);
   278→
   279→  useEffect(() => {
   280→    requestAnimationFrame(() => {
   281→      scrollRef.current?.scrollTo({ top: scrollRef.current.scrollHeight });
   282→    });
   283→  }, [messages.length]);
   284→
   285→  const send = async () => {
   286→    const t = text.trim();
   287→    if (!t) return;
   288→    setSending(true);
   289→    setText('');
   290→    try {
   291→      const d = await api<{ message: OrderMessageDto; delivered: boolean; deliveryError?: string }>(
   292→        `/api/bots/${botId}/orders/${orderId}/messages`,
   293→        { method: 'POST', body: JSON.stringify({ text: t }) }
   294→      );
   295→      setMessages((prev) => [...prev, d.message]);
   296→      if (!d.delivered && d.deliveryError) {
   297→        toast({ title: 'Не доставлено в мессенджер', description: d.deliveryError, variant: 'destructive' });
   298→      }
   299→    } catch {
   300→      toast({ title: 'Не удалось отправить', variant: 'destructive' });
   301→    } finally {
   302→      setSending(false);
   303→    }
   304→  };
   305→
   306→  return (
   307→    <div className="flex min-h-0 flex-1 flex-col">
   308→      <div className="flex items-center gap-2 border-b px-3 py-2 text-xs text-muted-foreground">
   309→        <MessageSquare className="h-3.5 w-3.5" />
   310→        Чат с клиентом по заявке — сообщения доставляются в его мессенджер
   311→      </div>
   312→      <div ref={scrollRef} className="min-h-0 flex-1 space-y-2 overflow-y-auto bg-muted/30 p-3">
   313→        {messages.length === 0 && (
   314→          <div className="flex h-full items-center justify-center text-center text-xs text-muted-foreground">
   315→            Сообщений по заявке пока нет.
   316→            <br />
   317→            Клиент может написать в боте, открыв карточку своей заявки.
   318→          </div>
   319→        )}
   320→        {messages.map((m) => {
   321→          const fromClient = m.role === 'user';
   322→          return (
   323→            <div key={m.id} className={cn('flex', fromClient ? 'justify-start' : 'justify-end')}>
   324→              <div
   325→                className={cn(
   326→                  'max-w-[85%] space-y-0.5 rounded-2xl px-3 py-1.5 text-sm leading-snug',
   327→                  fromClient
   328→                    ? 'rounded-bl-md border bg-card'
   329→                    : 'rounded-br-md bg-primary text-primary-foreground'
   330→                )}
   331→              >
   332→                {m.nodeId === '__operator' && (
   333→                  <div className="flex items-center gap-1 text-[10px] opacity-70">
   334→                    <Headset className="h-3 w-3" /> оператор
   335→                  </div>
   336→                )}
   337→                <div className="whitespace-pre-wrap">{m.text}</div>
   338→                <div className="text-right text-[9px] opacity-60">
   339→                  {new Date(m.createdAt).toLocaleTimeString('ru-RU', {
   340→                    hour: '2-digit',
   341→                    minute: '2-digit',
   342→                  })}
   343→                </div>
   344→              </div>
   345→            </div>
   346→          );
   347→        })}
   348→      </div>
   349→      <form
   350→        className="flex items-center gap-2 border-t p-2.5"
   351→        onSubmit={(e) => {
   352→          e.preventDefault();
   353→          send();
   354→        }}
   355→      >
   356→        <Input
   357→          placeholder={
   358→            hasConversation
   359→              ? 'Ответить клиенту…'
   360→              : 'Диалог с клиентом не привязан — чат только для чтения'
   361→          }
   362→          value={text}
   363→          disabled={!hasConversation}
   364→          onChange={(e) => setText(e.target.value)}
   365→        />
   366→        <Button
   367→          type="submit"
   368→          size="icon"
   369→          className="h-10 w-10 shrink-0"
   370→          disabled={sending || !hasConversation || !text.trim()}
   371→          aria-label="Отправить"
   372→        >
   373→          {sending ? <Loader2 className="h-4 w-4 animate-spin" /> : <Send className="h-4 w-4" />}
   374→        </Button>
   375→      </form>
   376→    </div>
   377→  );
   378→}

   382→function OrderDetailDialog({
   383→  botId,
   384→  orderId,
   385→  open,
   386→  onClose,
   387→  onSwitchOrder,
   388→  onPlaceRequest,
   389→  onDataChanged,
   390→}: {
   391→  botId: string;
   392→  orderId: string | null;
   393→  open: boolean;
   394→  onClose: () => void;
   395→  onSwitchOrder: (id: string) => void;
   396→  onPlaceRequest: (orderId: string) => void;
   397→  onDataChanged: () => void;
   398→}) {
   399→  const { toast } = useToast();
   400→  const [order, setOrder] = useState<OrderDto | null>(null);
   401→  const [clientOrders, setClientOrders] = useState<OrderDto[]>([]);
   402→  const [loading, setLoading] = useState(false);
   403→  const [assignee, setAssignee] = useState('');
   404→  const [wishDate, setWishDate] = useState('');
   405→  const [comment, setComment] = useState('');
   406→  const [saving, setSaving] = useState(false);
   407→  const [confirmDelete, setConfirmDelete] = useState(false);
   408→
   409→  const load = useCallback(async () => {
   410→    if (!orderId) return;
   411→    setLoading(true);
   412→    try {
   413→      const d = await api<{ order: OrderDto; clientOrders: OrderDto[] }>(
   414→        `/api/bots/${botId}/orders/${orderId}`
   415→      );
   416→      setOrder(d.order);
   417→      setClientOrders(d.clientOrders);
   418→      setAssignee(d.order.assignee ?? '');
   419→      setWishDate(d.order.wishDate ?? '');
   420→      setComment(d.order.comment ?? '');
   421→    } catch {
   422→      toast({ title: 'Не удалось загрузить заявку', variant: 'destructive' });
   423→    } finally {
   424→      setLoading(false);
   425→    }
   426→  }, [botId, orderId, toast]);
   427→
   428→  useEffect(() => {
   429→    if (open && orderId) load();
   430→    if (!open) {
   431→      setOrder(null);
   432→      setClientOrders([]);
   433→      setConfirmDelete(false);
   434→    }
   435→  }, [open, orderId, load]);
   436→
   437→  const patch = async (data: Record<string, unknown>, successMsg?: string) => {
   438→    if (!orderId) return;
   439→    setSaving(true);
   440→    try {
   441→      const d = await api<{ order: OrderDto }>(`/api/bots/${botId}/orders/${orderId}`, {
   442→        method: 'PATCH',
   443→        body: JSON.stringify(data),
   444→      });
   445→      setOrder(d.order);
   446→      onDataChanged();
   447→      if (successMsg) toast({ title: successMsg });
   448→    } catch {
   449→      toast({ title: 'Ошибка сохранения', variant: 'destructive' });
   450→    } finally {
   451→      setSaving(false);
   452→    }
   453→  };
   454→
   455→  const remove = async () => {
   456→    if (!orderId) return;
   457→    try {
   458→      await api(`/api/bots/${botId}/orders/${orderId}`, { method: 'DELETE' });
   459→      toast({ title: `Заявка удалена` });
   460→      onDataChanged();
   461→      onClose();
   462→    } catch {
   463→      toast({ title: 'Не удалось удалить', variant: 'destructive' });
   464→    }
   465→  };
   466→
   467→  return (
   468→    <Dialog open={open} onOpenChange={(v) => !v && onClose()}>
   469→      <DialogContent className="grid h-[90dvh] max-w-4xl grid-rows-[auto_1fr] gap-0 overflow-hidden p-0 md:h-[85vh]">
   470→        <DialogHeader className="border-b px-4 py-3 md:px-5">
   471→          <DialogTitle className="flex items-center gap-2 text-base">
   472→            <span className="text-lg">{order ? ORDER_TYPE_ICONS[order.type] : '📋'}</span>
   473→            {order ? `Заявка №${order.number} — ${ORDER_TYPE_LABELS[order.type] ?? ''}` : 'Заявка'}
   474→          </DialogTitle>
   475→          <DialogDescription className="sr-only">
   476→            Карточка заявки: данные клиента, статус и встроенный чат
   477→          </DialogDescription>
   478→        </DialogHeader>
   479→
   480→        {!order && loading ? (
   481→          <div className="flex items-center justify-center gap-2 text-muted-foreground">
   482→            <Loader2 className="h-5 w-5 animate-spin" /> Загрузка…
   483→          </div>
   484→        ) : !order ? (
   485→          <div className="flex items-center justify-center text-sm text-muted-foreground">
   486→            Заявка не найдена
   487→          </div>
   488→        ) : (
   489→          <div className="grid min-h-0 overflow-y-auto md:grid-cols-2 md:overflow-hidden">
   490→            {/* Левая колонка: клиент + данные заявки */}
   491→            <div className="min-h-0 space-y-4 border-b p-4 md:border-b-0 md:border-r md:overflow-y-auto">
   492→              {/* Клиент */}
   493→              <div className="rounded-xl border bg-muted/30 p-3">
   494→                <div className="flex items-center justify-between">
   495→                  <h3 className="flex items-center gap-1.5 text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   496→                    <User className="h-3.5 w-3.5" /> Карточка клиента
   497→                  </h3>
   498→                  {order.conversation && (
   499→                    <span
   500→                      className={cn(
   501→                        'rounded-full px-2 py-0.5 text-[10px] font-medium',
   502→                        SOURCE_COLORS[order.conversation.source] ?? 'bg-muted text-muted-foreground'
   503→                      )}
   504→                    >
   505→                      {SOURCE_LABELS[order.conversation.source] ?? order.conversation.source}
   506→                    </span>
   507→                  )}
   508→                </div>
   509→                <div className="mt-2 space-y-1 text-sm">
   510→                  <div className="font-medium">
   511→                    {order.clientName || order.conversation?.contact || 'Клиент без имени'}
   512→                  </div>
   513→                  {order.phone && (
   514→                    <a
   515→                      href={`tel:${order.phone.replace(/[^\d+]/g, '')}`}
   516→                      className="flex w-fit items-center gap-1.5 text-primary hover:underline"
   517→                    >
   518→                      <Phone className="h-3.5 w-3.5" /> {order.phone}
   519→                    </a>
   520→                  )}
   521→                  {order.conversation?.externalUserId && (
   522→                    <div className="text-xs text-muted-foreground">
